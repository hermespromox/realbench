<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Underwater Ecosystem</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            overflow: hidden;
            background-color: #001a33;
        }
        canvas {
            display: block;
            width: 100vw;
            height: 100vh;
        }
    </style>
</head>
<body>
    <canvas id="canvas"></canvas>
    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');

        let width, height;
        let fish = [];
        let bubbles = [];
        let particles = [];
        let jellyfish = [];
        let seaweed = [];
        let schools = [];

        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
            initEcosystem();
        }

        class Fish {
            constructor(isSchoolMember = false, schoolId = null) {
                this.init(isSchoolMember, schoolId);
            }

            init(isSchoolMember = false, schoolId = null) {
                this.isSchoolMember = isSchoolMember;
                this.schoolId = schoolId;
                this.size = isSchoolMember ? Math.random() * 3 + 2 : Math.random() * 8 + 5;
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.speed = isSchoolMember ? Math.random() * 1 + 1 : Math.random() * 1.5 + 0.5;
                this.angle = Math.random() * Math.PI * 2;
                this.turnSpeed = (Math.random() - 0.5) * 0.05;
                this.color = this.getRandomColor(isSchoolMember);
                this.phase = Math.random() * Math.PI * 2;
            }

            getRandomColor(isSchoolMember) {
                if (isSchoolMember) {
                    // Schools share a similar color palette
                    const baseHue = 200 + Math.random() * 40; 
                    return `hsl(${baseHue}, 70%, 60%)`;
                }
                return `hsl(${Math.random() * 360}, 60%, ${40 + Math.random() * 30}%)`;
            }

            update() {
                this.angle += this.turnSpeed;
                if (Math.random() < 0.02) this.turnSpeed = (Math.random() - 0.5) * 0.05;

                this.x += Math.cos(this.angle) * this.speed;
                this.y += Math.sin(this.angle) * this.speed;

                // Wrap around screen
                if (this.x < -50) this.x = width + 50;
                if (this.x > width + 50) this.x = -50;
                if (this.y < -50) this.y = height + 50;
                if (this.y > height + 50) this.y = -50;

                this.phase += 0.1;
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.angle);
                
                ctx.fillStyle = this.color;
                // Body
                ctx.beginPath();
                ctx.ellipse(0, 0, this.size * 2, this.size, 0, 0, Math.PI * 2);
                ctx.fill();
                
                // Tail (wiggling)
                const tailWiggle = Math.sin(this.phase) * 0.5;
                ctx.beginPath();
                ctx.moveTo(-this.size * 1.5, 0);
                ctx.lineTo(-this.size * 3, tailWiggle * this.size);
                ctx.lineTo(-this.size * 3, -tailWiggle * this.size);
                ctx.closePath();
                ctx.fill();
                
                ctx.restore();
            }
        }

        class Jellyfish {
            constructor() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = Math.random() * 15 + 15;
                this.speed = Math.random() * 0.3 + 0.2;
                this.angle = Math.random() * Math.PI * 2;
                this.phase = Math.random() * Math.PI * 2;
                this.color = `hsla(${180 + Math.random() * 40}, 50%, 80%, 0.4)`;
            }

            update() {
                this.phase += 0.03;
                const pulse = Math.sin(this.phase) * 0.5 + 0.5;
                this.y -= this.speed * (pulse + 0.5);
                this.x += Math.sin(this.phase * 0.5) * 0.5;

                if (this.y < -100) this.y = height + 100;
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.fillStyle = this.color;
                
                const pulse = Math.sin(this.phase) * 5;
                
                // Bell
                ctx.beginPath();
                ctx.arc(0, 0, this.size + pulse, Math.PI, 0);
                ctx.fill();
                
                // Tentacles
                ctx.strokeStyle = this.color;
                ctx.lineWidth = 2;
                for (let i = -2; i <= 2; i++) {
                    ctx.beginPath();
                    ctx.moveTo(i * 5, 0);
                    ctx.bezierCurveTo(
                        i * 5 + Math.sin(this.phase) * 5, 10,
                        i * 5 - Math.sin(this.phase) * 5, 20,
                        i * 5, 30 + pulse
                    );
                    ctx.stroke();
                }
                ctx.restore();
            }
        }

        class Bubble {
            constructor() {
                this.init();
            }
            init() {
                this.x = Math.random() * width;
                this.y = height + 20;
                this.size = Math.random() * 3 + 1;
                this.speed = Math.random() * 1 + 1;
                this.drift = Math.random() * 2 - 1;
            }
            update() {
                this.y -= this.speed;
                this.x += Math.sin(this.y * 0.05) * 0.5;
                if (this.y < -20) this.init();
            }
            draw() {
                ctx.strokeStyle = 'rgba(255, 255, 255, 0.3)';
                ctx.lineWidth = 1;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.stroke();
            }
        }

        class Seaweed {
            constructor(x) {
                this.x = x;
                this.height = Math.random() * 100 + 150;
                this.segments = 8;
                this.phase = Math.random() * Math.PI * 2;
                this.color = `hsl(${120 + Math.random() * 20}, 40%, ${20 + Math.random() * 20}%)`;
            }
            update() {
                this.phase += 0.02;
            }
            draw() {
                ctx.strokeStyle = this.color;
                ctx.lineWidth = 10;
                ctx.lineCap = 'round';
                ctx.beginPath();
                ctx.moveTo(this.x, height);
                
                for (let i = 1; i <= this.segments; i++) {
                    const y = height - (i * (this.height / this.segments));
                    const offset = Math.sin(this.phase + i * 0.5) * 15;
                    ctx.lineTo(this.x + offset, y);
                }
                ctx.stroke();
            }
        }

        class Particle {
            constructor() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = Math.random() * 2;
                this.speedX = Math.random() * 0.5 - 0.25;
                this.speedY = Math.random() * 0.5 - 0.25;
            }
            update() {
                this.x += this.speedX;
                this.y += this.speedY;
                if (this.x < 0) this.x = width;
                if (this.x > width) this.x = 0;
                if (this.y < 0) this.y = height;
                if (this.y > height) this.y = 0;
            }
            draw() {
                ctx.fillStyle = 'rgba(255, 255, 255, 0.2)';
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        function initEcosystem() {
            fish = [];
            bubbles = [];
            particles = [];
            jellyfish = [];
            seaweed = [];
            
            // Solo fish
            for (let i = 0; i < 25; i++) {
                fish.push(new Fish());
            }

            // Schools of fish
            for (let s = 0; s < 4; s++) {
                const schoolId = s;
                const schoolSize = 15 + Math.floor(Math.random() * 10);
                const baseHue = 180 + Math.random() * 60;
                for (let i = 0; i < schoolSize; i++) {
                    const f = new Fish(true, schoolId);
                    f.color = `hsl(${baseHue}, 70%, 60%)`;
                    fish.push(f);
                }
            }

            for (let i = 0; i < 60; i++) particles.push(new Particle());
            for (let i = 0; i < 30; i++) bubbles.push(new Bubble());
            for (let i = 0; i < 6; i++) jellyfish.push(new Jellyfish());
            for (let i = 0; i < 20; i++) seaweed.push(new Seaweed(Math.random() * width));
        }

        function drawBackground() {
            const grad = ctx.createLinearGradient(0, 0, 0, height);
            grad.addColorStop(0, '#004d80');
            grad.addColorStop(1, '#001a33');
            ctx.fillStyle = grad;
            ctx.fillRect(0, 0, width, height);

            // Light rays
            ctx.save();
            ctx.globalCompositeOperation = 'overlay';
            ctx.fillStyle = 'rgba(255, 255, 255, 0.1)';
            const time = Date.now() * 0.001;
            for (let i = 0; i < 8; i++) {
                ctx.beginPath();
                const xOffset = Math.sin(time * 0.2 + i) * 100;
                ctx.moveTo(width * 0.2 * i + xOffset, 0);
                ctx.lineTo(width * 0.2 * i + 100 + xOffset, 0);
                ctx.lineTo(width * 0.2 * i - 200 + xOffset * 2, height);
                ctx.lineTo(width * 0.2 * i - 400 + xOffset * 2, height);
                ctx.closePath();
                ctx.fill();
            }
            ctx.restore();
        }

        function animate() {
            drawBackground();

            seaweed.forEach(s => {
                s.update();
                s.draw();
            });

            particles.forEach(p => {
                p.update();
                p.draw();
            });

            bubbles.forEach(b => {
                b.update();
                b.draw();
            });

            jellyfish.forEach(j => {
                j.update();
                j.draw();
            });

            fish.forEach(f => {
                f.update();
                f.draw();
            });

            requestAnimationFrame(animate);
        }

        window.addEventListener('resize', resize);
        resize();
        animate();
    </script>
</body>
</html>
