   const sf=(j+1)/wd.segs;
   const sway=wd.swA*Math.pow(sf,1.3)*(0.72*Math.sin(wd.f1*OM*t+wd.p1+j*0.3)+0.38*Math.sin(wd.f2*OM*t+wd.p2+j*0.55));

   ctx.lineWidth=Math.max(0.8,wd.baseW*(1-sf*0.82));

function drawSchools(t){
 for(const s of schools){
  for(const m of s.members){
   const tt=t-m.lag;
   const A=schoolAnchor(s,tt);
   const th=m.a0+m.swF*OM*t+m.swP;
   const rr=m.r*s.spread;
   const x=A.x+Math.cos(th)*rr+m.jA*Math.sin(m.jF*OM*t+m.jP);
   const y=A.y+Math.sin(th)*rr*0.55+m.jA2*Math.sin(m.jF2*OM*t+m.jP2);
   let h=Math.atan2(A.dy,A.dx);
   h+=m.hA*Math.sin(m.hF*OM*t+m.hP);
   const tail=m.tailF*OM*t+m.tailP;
   const col={h:s.hue[0]+m.lOff*2,s:s.hue[1],l:s.hue[2]+m.lOff};
   ctx.save();
   ctx.translate(x,y);
   ctx.rotate(h);
   if(Math.cos(h)<0)ctx.scale(1,-1);
   ctx.globalAlpha=0.92;
   paintFish(m.size,col,tail,false,0.15);
   ctx.restore();
  }
 }
}

  const y=f.fy*H+f.bobA*Math.sin(f.bobF*OM*t+f.bobP)+f.bobA2*Math.sin(f.bobF2*OM*t+f.bobP2);
  const dvx=f.dir*(v+f.wobA*f.size*f.wobF*OM*Math.cos(wobPh));
  const dvy=f.bobA*f.bobF*OM*Math.cos(f.bobF*OM*t+f.bobP)+f.bobA2*f.bobF2*OM*Math.cos(f.bobF2*OM*t+f.bobP2);
  const pitch=clamp(Math.atan2(dvy,Math.abs(dvx)),-0.5,0.5);
  const tail=f.tailF*OM*t+f.tailP;
  ctx.save();
  ctx.translate(x,y);
  if(f.dir<0)ctx.scale(-1,1);
  ctx.rotate(pitch);
  paintFish(f.size,f.col,tail,f.stripes,f.depth);
  ctx.restore();
 }
}

function drawBubbles(t){
 const span=H+24;
 for(const b of bubbles){
  const v=span*b.k/T;
  const y=mod(b.y0*span-v*t,span)-12;
  if(y<-b.r*2||y>H+b.r*2)continue;
  const x=b.x0*W+b.wobA*Math.sin(b.wobF*OM*t+b.wobP);
  const fadeT=clamp((y-26)/50,0,1);
  const a=0.55*Math.min(1,fadeT+0.15);
  if(a<=0.02)continue;
  ctx.beginPath();ctx.arc(x,y,b.r,0,TAU);
  ctx.fillStyle=`rgba(190,225,250,${(a*0.16).toFixed(3)})`;
  ctx.fill();
  ctx.strokeStyle=`rgba(215,240,255,${a.toFixed(3)})`;
  ctx.lineWidth=1;ctx.stroke();
  ctx.beginPath();ctx.arc(x-b.r*0.32,y-b.r*0.32,Math.max(0.5,b.r*0.3),0,TAU);
  ctx.fillStyle=`rgba(240,250,255,${(a*0.8).toFixed(3)})`;
  ctx.fill();
 }
}

function drawParticles(t){
 for(const p of parts){
  const x=mod(p.x0*W+(W*p.nx/T)*t+p.wA*Math.sin(p.wF*OM*t+p.wP),W);
  const y=mod(p.y0*H+(H*p.ny/T)*t,H);
  const a=p.a*(0.55+0.45*Math.sin(p.pF*OM*t+p.pP));
  if(a<=0.01)continue;
  ctx.fillStyle=`rgba(205,232,255,${a.toFixed(3)})`;
  ctx.beginPath();ctx.arc(x,y,p.size,0,TAU);ctx.fill();
 }
}

function vignette(){
 const g=ctx.createRadialGradient(W*0.5,H*0.45,Math.min(W,H)*0.35,W*0.5,H*0.5,Math.max(W,H)*0.75);
 g.addColorStop(0,'rgba(1,8,16,0)');
 g.addColorStop(1,'rgba(1,8,16,0.5)');
 ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
}

function draw(t){
 drawBackground();
 drawParticles(t);
 drawRays(t,1);
 drawSurface(t);
 drawFloor(t);
 drawWeeds(weedsBack,t,0.7);
 for(const j of jellies)drawJelly(j,t);
 drawSchools(t);
 drawFishes(t);
 drawWeeds(weedsFront,t,0.95);
 drawBubbles(t);
 drawRays(t,0.35);
 vignette();
}

const start=performance.now();
function frame(now){
 const t=((now-start)/1000)%T;
 draw(t);
 requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
})();
