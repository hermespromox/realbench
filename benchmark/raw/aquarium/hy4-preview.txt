<script>
const canvas = document.getElementById('ocean');
const ctx = canvas.getContext('2d');
let width, height;

function resize() {
    let dpr = window.devicePixelRatio || 1;
    width = window.innerWidth;
    height = window.innerHeight;
    canvas.width = width * dpr;
    canvas.height = height * dpr;
    ctx.scale(dpr, dpr);
}
window.addEventListener('resize', resize);
resize();

let time = 0;
let lastTime = 0;

class LightRay {
    constructor() {
        this.x = Math.random() * width;
        this.width = Math.random() * 200 + 50;
        this.angle = (Math.random() - 0.5) * 0.3;
        this.speed = Math.random() * 0.0005 + 0.0002;
        this.phase = Math.random() * Math.PI * 2;
        this.hue = Math.random() > 0.5 ? 200 : 180;
    }
    draw(ctx, time) {
        let alpha = (Math.sin(time * this.speed + this.phase) + 1) * 0.04 + 0.01;
        ctx.fillStyle = `hsla(${this.hue}, 100%, 80%, ${alpha})`;
        ctx.beginPath();
        let topX = this.x + Math.sin(time * 0.0003 + this.phase) * 100;
        ctx.moveTo(topX, 0);
        ctx.lineTo(topX + this.width, 0);
        let botX = topX + this.angle * height;
        ctx.lineTo(botX + this.width * 2, height);
        ctx.lineTo(botX - this.width, height);
        ctx.fill();
    }
}

class Particle {
    constructor() {
        this.x = Math.random() * width;
        this.y = Math.random() * height;
        this.z = Math.random() * 0.8 + 0.2; 
        this.vx = (Math.random() - 0.5) * 0.5 * this.z;
        this.vy = (Math.random() - 0.5) * 0.5 * this.z - 0.2 * this.z; 
        this.size = (Math.random() * 2 + 0.5) * this.z;
        this.alpha = (Math.random() * 0.5 + 0.1) * this.z;
    }
    update(time) {
        this.x += this.vx + Math.sin(time * 0.001 + this.z * 10) * 0.2;
        this.y += this.vy;
        if(this.x < 0) this.x = width;
        if(this.x > width) this.x = 0;
        if(this.y < 0) this.y = height;
        if(this.y > height) this.y = 0;
    }
    draw(ctx) {
        ctx.fillStyle = `rgba(255, 255, 255, ${this.alpha})`;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI*2);
        ctx.fill();
    }
}

class Seaweed {
    constructor() {
        this.x = Math.random() * width;
        this.height = Math.random() * 300 + 150;
        this.segments = 20;
        this.phase = Math.random() * Math.PI * 2;
        this.isBackground = Math.random() > 0.6;
        this.thickness = Math.random() * 8 + 4;
        this.swaySpeed = Math.random() * 0.001 + 0.0005;
        
        if (this.isBackground) {
            this.color = `hsl(${130 + Math.random()*30}, ${40+Math.random()*20}%, ${10+Math.random()*15}%)`;
            this.thickness *= 0.6;
            this.height *= 0.7;
        } else {
            this.color = `hsl(${130 + Math.random()*30}, ${60+Math.random()*30}%, ${20+Math.random()*20}%)`;
        }
    }
    draw(ctx, time) {
        ctx.beginPath();
        ctx.moveTo(this.x, height);
        let currentX = this.x;
        let currentY = height;
        let segHeight = this.height / this.segments;
        
        for(let i=1; i<=this.segments; i++) {
            let swayFactor = Math.pow(i / this.segments, 1.5);
            let sway = Math.sin(time * this.swaySpeed + this.phase + i * 0.15) * (swayFactor * 50);
            currentX = this.x + sway;
            currentY = height - i * segHeight;
            ctx.lineTo(currentX, currentY);
        }
        
        ctx.strokeStyle = this.color;
        ctx.lineWidth = this.thickness;
        ctx.lineCap = 'round';
        ctx.lineJoin = 'round';
        ctx.stroke();
    }
}

class Bubble {
    constructor() {
        this.reset(true);
    }
    reset(randomY = false) {
        this.x = Math.random() * width;
        this.y = randomY ? Math.random() * height : height + 10;
        this.size = Math.random() * 3 + 1;
        this.speed = Math.random() * 1.5 + 0.5;
        this.phase = Math.random() * Math.PI * 2;
        this.wobbleSpeed = Math.random() * 0.02 + 0.01;
    }
    update(time) {
        this.y -= this.speed;
        if(this.y < -10) this.reset();
    }
    draw(ctx, time) {
        let x = this.x + Math.sin(time * this.wobbleSpeed + this.phase) * (this.size * 2);
        ctx.beginPath();
        ctx.arc(x, this.y, this.size, 0, Math.PI*2);
        ctx.fillStyle = 'rgba(255,255,255,0.2)';
        ctx.fill();
        ctx.strokeStyle = 'rgba(255,255,255,0.5)';
        ctx.lineWidth = 1;
        ctx.stroke();
        
        ctx.beginPath();
        ctx.arc(x - this.size*0.3, this.y - this.size*0.3, this.size*0.2, 0, Math.PI*2);
        ctx.fillStyle = 'rgba(255,255,255,0.8)';
        ctx.fill();
    }
}

class Fish {
    constructor() {
        this.x = Math.random() * width;
        this.y = Math.random() * height;
        this.vx = (Math.random() - 0.5) * 2;
        this.vy = (Math.random() - 0.5) * 1;
        this.size = Math.random() * 15 + 15;
        this.color = `hsl(${Math.random()*360}, 70%, 50%)`;
        this.phase = Math.random() * Math.PI * 2;
        this.speed = Math.random() * 0.002 + 0.001;
        this.wanderSpeed = Math.random() * 0.001 + 0.0005;
        this.shapeType = Math.floor(Math.random() * 3);
    }
    update(time) {
        let angleNoise = Math.sin(time * this.wanderSpeed + this.phase) * 0.05;
        let currentAngle = Math.atan2(this.vy, this.vx);
        currentAngle += angleNoise;
        let speed = Math.sqrt(this.vx*this.vx + this.vy*this.vy);
        if (speed < 1.5) speed = 1.5;
        if (speed > 3) speed = 3;
        
        this.vx = Math.cos(currentAngle) * speed;
        this.vy = Math.sin(currentAngle) * speed;
        
        this.x += this.vx;
        this.y += this.vy;

        let margin = this.size * 3;
        if (this.x < -margin) this.x = width + margin;
        if (this.x > width + margin) this.x = -margin;
        if (this.y < -margin) this.y = height + margin;
        if (this.y > height + margin) this.y = -margin;
    }
    draw(ctx, time) {
        ctx.save();
        ctx.translate(this.x, this.y);
        let angle = Math.atan2(this.vy, this.vx);
        ctx.rotate(angle);
        if (angle > Math.PI / 2 || angle < -Math.PI / 2) {
            ctx.scale(1, -1);
        }
        
        ctx.fillStyle = 'rgba(0,0,0,0.15)';
        ctx.beginPath();
        ctx.moveTo(this.size + 4, 4);
        ctx.quadraticCurveTo(this.size/2, -this.size/2 + 4, -this.size + 4, 4);
        ctx.quadraticCurveTo(this.size/2, this.size/2 + 4, this.size + 4, 4);
        ctx.fill();

        ctx.fillStyle = this.color;
        ctx.beginPath();
        if (this.shapeType === 0) {
            ctx.moveTo(this.size, 0);
            ctx.quadraticCurveTo(this.size/2, -this.size*0.6, -this.size, 0);
            ctx.quadraticCurveTo(this.size/2, this.size*0.6, this.size, 0);
        } else if (this.shapeType === 1) {
            ctx.moveTo(this.size * 1.2, 0);
            ctx.quadraticCurveTo(this.size/2, -this.size*0.35, -this.size, 0);
            ctx.quadraticCurveTo(this.size/2, this.size*0.35, this.size * 1.2, 0);
        } else {
            ctx.ellipse(0, 0, this.size, this.size * 0.8, 0, 0, Math.PI * 2);
        }
        ctx.fill();

        let tailWiggle = Math.sin(time * 0.015 + this.phase) * (this.size * 0.4);
        ctx.beginPath();
        if (this.shapeType !== 2) {
            let tailStart = this.shapeType === 0 ? -this.size + 2 : -this.size;
            ctx.moveTo(tailStart, 0);
            ctx.lineTo(tailStart - this.size*0.6, -this.size*0.5 + tailWiggle);
            ctx.lineTo(tailStart - this.size*0.6, this.size*0.5 + tailWiggle);
        } else {
            ctx.moveTo(-this.size * 0.8, 0);
            ctx.lineTo(-this.size * 1.6, -this.size*0.6 + tailWiggle);
            ctx.lineTo(-this.size * 1.6, this.size*0.6 + tailWiggle);
        }
        ctx.fill();
        
        if (this.shapeType === 1) {
            ctx.fillStyle = 'rgba(255,255,255,0.3)';
            ctx.beginPath();
            ctx.moveTo(this.size*0.2, -this.size*0.25);
            ctx.lineTo(this.size*0.4, -this.size*0.25);
            ctx.lineTo(this.size*0.3, this.size*0.25);
            ctx.lineTo(this.size*0.1, this.size*0.25);
            ctx.fill();
            
            ctx.beginPath();
            ctx.moveTo(-this.size*0.2, -this.size*0.2);
            ctx.lineTo(0, -this.size*0.2);
            ctx.lineTo(-this.size*0.1, this.size*0.2);
            ctx.lineTo(-this.size*0.3, this.size*0.2);
            ctx.fill();
        }

        ctx.fillStyle = 'white';
        ctx.beginPath();
        ctx.arc(this.size*0.6, -this.size*0.15, this.size*0.15, 0, Math.PI*2);
        ctx.fill();
        ctx.fillStyle = 'black';
        ctx.beginPath();
        ctx.arc(this.size*0.65, -this.size*0.15, this.size*0.08, 0, Math.PI*2);
        ctx.fill();

        ctx.restore();
    }
}

class School {
    constructor(numFish) {
        this.fish = [];
        this.color = `hsl(${Math.random()*360}, 80%, 60%)`;
        this.wanderPhaseX = Math.random() * Math.PI * 2;
        this.wanderPhaseY = Math.random() * Math.PI * 2;
        let startX = Math.random() * width;
        let startY = Math.random() * height;
        for(let i=0; i<numFish; i++) {
            this.fish.push({
                x: startX + (Math.random() - 0.5) * 100,
                y: startY + (Math.random() - 0.5) * 100,
                vx: (Math.random() - 0.5) * 2,
                vy: (Math.random() - 0.5) * 2,
                phase: Math.random() * Math.PI * 2
            });
        }
    }
    update(time) {
        let cx = 0, cy = 0, cvx = 0, cvy = 0;
        for(let f of this.fish) {
            cx += f.x; cy += f.y;
            cvx += f.vx; cvy += f.vy;
        }
        cx /= this.fish.length;
        cy /= this.fish.length;
        cvx /= this.fish.length;
        cvy /= this.fish.length;

        let margin = Math.min(150, width * 0.2);
        this.targetX = Math.max(margin, Math.min(width - margin, cx + Math.sin(time * 0.0005 + this.wanderPhaseX) * (width * 0.4)));
        this.targetY = Math.max(margin, Math.min(height - margin, cy + Math.cos(time * 0.0007 + this.wanderPhaseY) * (height * 0.4)));

        for(let i=0; i<this.fish.length; i++) {
            let f = this.fish[i];
            
            let sepX = 0, sepY = 0;
            for(let j=0; j<this.fish.length; j++) {
                if(i===j) continue;
                let other = this.fish[j];
                let dx = f.x - other.x;
                let dy = f.y - other.y;
                let distSq = dx*dx + dy*dy;
                if(distSq < 400 && distSq > 0) { 
                    let dist = Math.sqrt(distSq);
                    sepX += dx / dist; 
                    sepY += dy / dist;
                } else if (distSq === 0) {
                    sepX += (Math.random() - 0.5) * 2;
                    sepY += (Math.random() - 0.5) * 2;
                }
            }

            let tx = this.targetX - f.x;
            let ty = this.targetY - f.y;
            let tDist = Math.sqrt(tx*tx + ty*ty);
            if(tDist > 0) { tx /= tDist; ty /= tDist; }

            let ax = cvx - f.vx;
            let ay = cvy - f.vy;

            f.vx += tx * 0.01 + ax * 0.05 + sepX * 0.05;
            f.vy += ty * 0.01 + ay * 0.05 + sepY * 0.05;

            let turn = 0.05;
            if (f.x < margin) f.vx += turn;
            if (f.x > width - margin) f.vx -= turn;
            if (f.y < margin) f.vy += turn;
            if (f.y > height - margin) f.vy -= turn;

            let speed = Math.sqrt(f.vx*f.vx + f.vy*f.vy);
            let maxSpeed = 2.5;
            let minSpeed = 1.0;
            if(speed > maxSpeed) {
                f.vx = (f.vx/speed) * maxSpeed;
                f.vy = (f.vy/speed) * maxSpeed;
            } else if(speed < minSpeed) {
                f.vx = (f.vx/speed) * minSpeed;
                f.vy = (f.vy/speed) * minSpeed;
            }
            
            f.x += f.vx;
            f.y += f.vy;
        }
    }
    draw(ctx, time) {
        ctx.fillStyle = this.color;
        for(let f of this.fish) {
            ctx.save();
            ctx.translate(f.x, f.y);
            let angle = Math.atan2(f.vy, f.vx);
            ctx.rotate(angle);
            if (angle > Math.PI / 2 || angle < -Math.PI / 2) {
                ctx.scale(1, -1);
            }
            
            let size = 6;
            ctx.beginPath();
            ctx.moveTo(size, 0);
            ctx.quadraticCurveTo(size/2, -size/2, -size, 0);
            ctx.quadraticCurveTo(size/2, size/2, size, 0);
            ctx.fill();

            let tailWiggle = Math.sin(time * 0.03 + f.phase) * (size * 0.5);
            ctx.beginPath();
            ctx.moveTo(-size, 0);
            ctx.lineTo(-size - size, -size*0.5 + tailWiggle);
            ctx.lineTo(-size - size, size*0.5 + tailWiggle);
            ctx.fill();
            
            ctx.restore();
        }
    }
}

class Jellyfish {
    constructor() {
        this.x = Math.random() * width;
        this.y = Math.random() * height;
        this.size = Math.random() * 20 + 20;
        this.phase = Math.random() * Math.PI * 2;
        this.speedY = -(Math.random() * 0.3 + 0.1);
        this.driftPhase = Math.random() * Math.PI * 2;
        this.hue = Math.random() * 40 + 240; 
    }
    update(time) {
        let pulse = Math.sin(time * 0.002 + this.phase);
        this.y += this.speedY * (1 + pulse * 0.8); 
        
        let drift = Math.sin(time * 0.0005 + this.driftPhase) * 0.5;
        this.x += drift;

        if (this.y < -this.size * 4) {
            this.y = height + this.size * 4;
            this.x = Math.random() * width;
        }
    }
    draw(ctx, time) {
        let pulse = Math.sin(time * 0.002 + this.phase);
        let bellWidth = this.size * (1 + pulse * 0.15);
        let bellHeight = this.size * (1 - pulse * 0.15);

        ctx.save();
        ctx.translate(this.x, this.y);
        
        ctx.strokeStyle = `hsla(${this.hue}, 80%, 80%, 0.6)`;
        ctx.lineWidth = 1.5;
        for(let i=-3; i<=3; i++) {
            ctx.beginPath();
            let startX = i * bellWidth * 0.25;
            ctx.moveTo(startX, bellHeight * 0.2);
            let tX = startX;
            let tY = bellHeight * 0.2;
            for(let j=1; j<=8; j++) {
                tY += bellHeight * 0.4;
                tX += Math.sin(time * 0.003 + this.phase + j * 0.5 + i) * 3;
                ctx.lineTo(tX, tY);
            }
            ctx.stroke();
        }

        ctx.shadowBlur = 15;
        ctx.shadowColor = `hsla(${this.hue}, 100%, 70%, 0.8)`;
        ctx.fillStyle = `hsla(${this.hue}, 80%, 70%, 0.7)`;
        ctx.beginPath();
        ctx.ellipse(0, 0, bellWidth, bellHeight, 0, Math.PI, Math.PI * 2);
        ctx.quadraticCurveTo(bellWidth * 0.5, bellHeight * 0.4, 0, bellHeight * 0.2);
        ctx.quadraticCurveTo(-bellWidth * 0.5, bellHeight * 0.4, -bellWidth, 0);
        ctx.fill();
        
        ctx.shadowBlur = 10;
        ctx.shadowColor = `hsla(${this.hue}, 100%, 90%, 1)`;
        ctx.fillStyle = `hsla(${this.hue}, 100%, 90%, 0.8)`;
        ctx.beginPath();
        ctx.ellipse(0, 0, bellWidth*0.6, bellHeight*0.6, 0, Math.PI, Math.PI * 2);
        ctx.quadraticCurveTo(bellWidth * 0.3, bellHeight * 0.2, 0, bellHeight * 0.1);
        ctx.quadraticCurveTo(-bellWidth * 0.3, bellHeight * 0.2, -bellWidth*0.6, 0);
        ctx.fill();

        ctx.restore();
    }
}

const lightRays = Array.from({length: 8}, () => new LightRay());
const particles = Array.from({length: 200}, () => new Particle());
const seaweeds = Array.from({length: 25}, () => new Seaweed());
const bubbles = Array.from({length: 60}, () => new Bubble());
const fishes = Array.from({length: 35}, () => new Fish());
const schools = Array.from({length: 4}, () => new School(20));
const jellyfishes = Array.from({length: 8}, () => new Jellyfish());

function animate(currentTime) {
    if (!lastTime) lastTime = currentTime;
    let dt = currentTime - lastTime;
    lastTime = currentTime;
    time += dt;

    let bgGradient = ctx.createLinearGradient(0, 0, 0, height);
    let shift = Math.sin(time * 0.0002) * 10;
    bgGradient.addColorStop(0, `hsl(${210 + shift}, 100%, 20%)`);
    bgGradient.addColorStop(1, `hsl(${230 + shift}, 100%, 5%)`);
    ctx.fillStyle = bgGradient;
    ctx.fillRect(0, 0, width, height);

    for(let ray of lightRays) ray.draw(ctx, time);
    for(let sw of seaweeds) { if(sw.isBackground) sw.draw(ctx, time); }
    for(let p of particles) { p.update(time); p.draw(ctx); }
    for(let s of schools) { s.update(time); s.draw(ctx, time); }
    for(let j of jellyfishes) { j.update(time); j.draw(ctx, time); }
    for(let f of fishes) { f.update(time); f.draw(ctx, time); }
    for(let b of bubbles) { b.update(time); b.draw(ctx, time); }
    for(let sw of seaweeds) { if(!sw.isBackground) sw.draw(ctx, time); }

    let vignette = ctx.createRadialGradient(width/2, height/2, height*0.4, width/2, height/2, height);
    vignette.addColorStop(0, 'rgba(0,0,0,0)');
    vignette.addColorStop(1, 'rgba(0,0,0,0.6)');
    ctx.fillStyle = vignette;
    ctx.fillRect(0, 0, width, height);

    requestAnimationFrame(animate);
}

requestAnimationFrame(animate);
