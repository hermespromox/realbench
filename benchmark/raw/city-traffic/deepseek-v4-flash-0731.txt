 Here is a complete HTML document that creates an animated city scene with traffic lights, moving cars, and pedestrians.
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>City Traffic Animation</title>
    <style>
        * {
            margin: 0;
            padding: 0;
            box-sizing: border-box;
        }
        body {
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
            background: #0d0d1a;
            font-family: Arial, sans-serif;
            overflow: hidden;
        }
        canvas {
            border-radius: 12px;
            box-shadow: 0 0 40px rgba(0, 0, 0, 0.6);
        }
    </style>
</head>
<body>
    <canvas id="city"></canvas>
    <script>
        const canvas = document.getElementById('city');
        const ctx = canvas.getContext('2d');
        const W = canvas.width = 960;
        const H = canvas.height = 720;

        // ── Layout ──
        const margin = 40;
        const roadW = 64;
        const cols = 3,
            rows = 3;
        const blockW = (W - 2 * margin - (cols - 1) * roadW) / cols;
        const blockH = (H - 2 * margin - (rows - 1) * roadW) / rows;

        function getIX(i) {
            if (i === 0) return margin;
            if (i === cols) return margin + cols * (blockW + roadW) - roadW;
            return margin + i * (blockW + roadW) - roadW / 2;
        }

        function getIY(j) {
            if (j === 0) return margin;
            if (j === rows) return margin + rows * (blockH + roadW) - roadW;
            return margin + j * (blockH + roadW) - roadW / 2;
        }

        function getBX(c) { return margin + c * (blockW + roadW); }

        function getBY(r) { return margin + r * (blockH + roadW); }

        // ── Intersections ──
        const intersections = [];
        for (let j = 0; j <= rows; j++) {
            for (let i = 0; i <= cols; i++) {
                intersections.push({ x: getIX(i), y: getIY(j), i, j });
            }
        }

        // ── Traffic light cycle ──
        const LIGHT_PHASES = [3.5, 1.2, 3.5, 1.2];
        const LIGHT_TOTAL = LIGHT_PHASES.reduce((a, b) => a + b, 0);
        let lightTime = 0;

        function getPhase() {
            const t = lightTime % LIGHT_TOTAL;
            let acc = 0;
            for (let i = 0; i < 4; i++) {
                acc += LIGHT_PHASES[i];
                if (t < acc) return i;
            }
            return 0;
        }
        const isNSGreen = p => p === 0;
        const isNSYellow = p => p === 1;
        const isEWGreen = p => p === 2;
        const isEWYellow = p => p === 3;

        // ── Buildings ──
        class Building {
            constructor(x, y, w, h) {
                this.x = x;
                this.y = y;
                this.w = w;
                this.h = h;
                this.color = `hsl(${200 + Math.random() * 40}, ${25 + Math.random() * 20}%, ${30 + Math.random() * 25}%)`;
                this.windows = [];
                const wc = Math.max(1, Math.floor(w / 14));
                const hc = Math.max(1, Math.floor(h / 14));
                for (let r = 0; r < hc; r++) {
                    for (let c = 0; c < wc; c++) {
                        this.windows.push({
                            x: this.x + 5 + c * 13,
                            y: this.y + 5 + r * 13,
                            w: 6,
                            h: 6,
                            lit: Math.random() > 0.25
                        });
                    }
                }
            }
            update(dt) {
                for (const win of this.windows) {
                    if (Math.random() < 0.0008 * dt * 60) {
                        win.lit = !win.lit;
                    }
                }
            }
            draw(ctx, time) {
                // Shadow
                ctx.fillStyle = 'rgba(0,0,0,0.25)';
                ctx.fillRect(this.x + 3, this.y + 3, this.w, this.h);
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.w, this.h);
                // Roof lines
                ctx.strokeStyle = 'rgba(0,0,0,0.2)';
                ctx.lineWidth = 1;
                ctx.strokeRect(this.x, this.y, this.w, this.h);
                // Windows
                for (const win of this.windows) {
                    if (win.lit) {
                        const flicker = 0.7 + 0.3 * Math.sin(time * 3 + win.x * 0.01 + win.y * 0.01);
                        ctx.fillStyle = `rgba(255, 220, 100, ${flicker})`;
                    } else {
                        ctx.fillStyle = 'rgba(15, 20, 30, 0.5)';
                    }
                    ctx.fillRect(win.x, win.y, win.w, win.h);
                }
            }
        }

        // ── Create buildings ──
        const buildings = [];
        for (let r = 0; r < rows; r++) {
            for (let c = 0; c < cols; c++) {
                const bx = getBX(c);
                const by = getBY(r);
                const bw = blockW / 2 - 6;
                const bh = blockH / 2 - 6;
                buildings.push(new Building(bx + 3, by + 3, bw, bh));
                buildings.push(new Building(bx + blockW / 2 + 3, by + 3, bw, bh));
                buildings.push(new Building(bx + 3, by + blockH / 2 + 3, bw, bh));
                buildings.push(new Building(bx + blockW / 2 + 3, by + blockH / 2 + 3, bw, bh));
            }
        }

        // ── Trees ──
        const trees = [];
        for (let r = 0; r < rows; r++) {
            for (let c = 0; c < cols; c++) {
                const bx = getBX(c);
                const by = getBY(r);
                // Add trees in the corners of each block
                const positions = [
                    { x: bx + 6, y: by + 6 },
                    { x: bx + blockW - 6, y: by + 6 },
                    { x: bx + 6, y: by + blockH - 6 },
                    { x: bx + blockW - 6, y: by + blockH - 6 }
                ];
                for (const pos of positions) {
                    trees.push({
                        x: pos.x + (Math.random() - 0.5) * 10,
                        y: pos.y + (Math.random() - 0.5) * 10,
                        size: 4 + Math.random() * 3
                    });
                }
            }
        }

        // ── Cars ──
        class Car {
            constructor() {
                this.reset();
            }
            reset() {
                const horiz = Math.random() > 0.5;
                const idx = Math.floor(Math.random() * (horiz ? (rows + 1) : (cols + 1)));
                if (horiz) {
                    const y = getIY(idx);
                    const dir = Math.random() > 0.5 ? 1 : -1;
                    this.x = dir > 0 ? getIX(0) - 20 : getIX(cols) + 20;
                    this.y = y + (dir > 0 ? roadW / 4 : -roadW / 4);
                    this.dirX = dir;
                    this.dirY = 0;
                } else {
                    const x = getIX(idx);
                    const dir = Math.random() > 0.5 ? 1 : -1;
                    this.x = x + (dir > 0 ? roadW / 4 : -roadW / 4);
                    this.y = dir > 0 ? getIY(0) - 20 : getIY(rows) + 20;
                    this.dirX = 0;
                    this.dirY = dir;
                }
                this.speed = 35 + Math.random() * 35;
                this.color = `hsl(${Math.random() * 360}, 65%, 65%)`;
                this.stopped = false;
                this.active = true;
                this.size = 10;
            }
            update(dt) {
                if (!this.active) return;
                this.stopped = false;

                // Check red light ahead
                for (const inter of intersections) {
                    const dx = inter.x - this.x;
                    const dy = inter.y - this.y;
                    const dist = Math.sqrt(dx * dx + dy * dy);
                    const dot = dx * this.dirX + dy * this.dirY;
                    if (dot > 0 && dist < 50) {
                        const phase = getPhase();
                        const goingNS = this.dirY !== 0;
                        const canGo = goingNS ? isNSGreen(phase) : isEWGreen(phase);
                        if (!canGo && dist > 28) {
                            this.stopped = true;
                            const stopDist = 28;
                            const ratio = stopDist / dist;
                            this.x += dx * ratio - dx;
                            this.y += dy * ratio - dy;
                            break;
                        }
                    }
                }

                if (this.stopped) return;

                this.x += this.dirX * this.speed * dt;
                this.y += this.dirY * this.speed * dt;

                // At intersection?
                for (const inter of intersections) {
                    const dx = Math.abs(this.x - inter.x);
                    const dy = Math.abs(this.y - inter.y);
                    if (dx < 12 && dy < 12) {
                        this.handleIntersection(inter);
                        break;
                    }
                }

                // Off bounds?
                if (this.x < margin - 80 || this.x > W - margin + 80 ||
                    this.y < margin - 80 || this.y > H - margin + 80) {
                    this.active = false;
                }
            }
            handleIntersection(inter) {
                const r = Math.random();
                let ndx = this.dirX,
                    ndy = this.dirY;
                if (r < 0.55) {
                    // straight
                } else if (r < 0.75) {
                    // right turn
                    const tx = -this.dirY,
                        ty = this.dirX;
                    const ni = inter.i + tx,
                        nj = inter.j + ty;
                    if (ni >= 0 && ni <= cols && nj >= 0 && nj <= rows) {
                        ndx = tx;
                        ndy = ty;
                    }
                } else {
                    // left turn
                    const tx = this.dirY,
                        ty = -this.dirX;
                    const ni = inter.i + tx,
                        nj = inter.j + ty;
                    if (ni >= 0 && ni <= cols && nj >= 0 && nj <= rows) {
                        ndx = tx;
                        ndy = ty;
                    }
                }
                this.dirX = ndx;
                this.dirY = ndy;

                // Snap to lane
                if (this.dirX === 1) {
                    this.x = inter.x + 6;
                    this.y = inter.y + roadW / 4;
                } else if (this.dirX === -1) {
                    this.x = inter.x - 6;
                    this.y = inter.y - roadW / 4;
                } else if (this.dirY === 1) {
                    this.x = inter.x + roadW / 4;
                    this.y = inter.y + 6;
                } else if (this.dirY === -1) {
                    this.x = inter.x - roadW / 4;
                    this.y = inter.y - 6;
                }
            }
            draw(ctx) {
                if (!this.active) return;
                ctx.save();
                ctx.translate(this.x, this.y);
                // Shadow
                ctx.fillStyle = 'rgba(0,0,0,0.3)';
                ctx.fillRect(-5, -3, 10, 6);
                // Car body
                ctx.fillStyle = this.color;
                ctx.fillRect(-4, -2.5, 8, 5);
                // Headlights
                ctx.fillStyle = '#ffffcc';
                if (this.dirX === 1) ctx.fillRect(3, -1, 2, 2);
                if (this.dirX === -1) ctx.fillRect(-5, -1, 2, 2);
                if (this.dirY === 1) ctx.fillRect(-1, 3, 2, 2);
                if (this.dirY === -1) ctx.fillRect(-1, -5, 2, 2);
                ctx.restore();
            }
        }

        // ── Pedestrians ──
        class Pedestrian {
            constructor() {
                const inter = intersections[Math.floor(Math.random() * intersections.length)];
                const dirs = [
                    { x: 1, y: 0 },
                    { x: -1, y: 0 },
                    { x: 0, y: 1 },
                    { x: 0, y: -1 }
                ];
                const dir = dirs[Math.floor(Math.random() * dirs.length)];
                this.x = inter.x;
                this.y = inter.y;
                this.dx = dir.x;
                this.dy = dir.y;
                this.speed = 15 + Math.random() * 10;
                this.active = true;
                this.color = `hsl(${Math.random() * 60 + 20}, 70%, 55%)`;
            }
            update(dt) {
                this.x += this.dx * this.speed * dt;
                this.y += this.dy * this.speed * dt;
                if (this.x < margin - 60 || this.x > W - margin + 60 ||
                    this.y < margin - 60 || this.y > H - margin + 60) {
                    this.active = false;
                }
            }
            draw(ctx) {
                if (!this.active) return;
                ctx.fillStyle = 'rgba(0,0,0,0.2)';
                ctx.beginPath();
                ctx.arc(this.x + 1, this.y + 1, 4, 0, Math.PI * 2);
                ctx.fill();
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, 3.5, 0, Math.PI * 2);
                ctx.fill();
                ctx.fillStyle = '#fff';
                ctx.beginPath();
                ctx.arc(this.x - 1, this.y - 1, 1.5, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        // ── Initialize ──
        const cars = [];
        const peds = [];

        function spawnCar() {
            const car = new Car();
            cars.push(car);
        }

        function spawnPed() {
            const ped = new Pedestrian();
            peds.push(ped);
        }

        // Spawn initial cars
        for (let i = 0; i < 6; i++) spawnCar();
        for (let i = 0; i < 3; i++) spawnPed();

        // ── Traffic density ──
        function trafficDensity() {
            return 0.5 + 0.5 * Math.sin(Date.now() / 8000);
        }

        // ── Draw helpers ──
        function drawRoads(time) {
            // Road surface
            ctx.fillStyle = '#2a2a35';
            // Horizontal roads
            for (let j = 0; j <= rows; j++) {
                const y = getIY(j);
                ctx.fillRect(margin - roadW / 2, y - roadW / 2, W - 2 * margin + roadW, roadW);
            }
            // Vertical roads
            for (let i = 0; i <= cols; i++) {
                const x = getIX(i);
                ctx.fillRect(x - roadW / 2, margin - roadW / 2, roadW, H - 2 * margin + roadW);
            }

            // Lane markings
            ctx.strokeStyle = 'rgba(255,255,255,0.3)';
            ctx.lineWidth = 2;
            ctx.setLineDash([10, 8]);
            // Horizontal
            for (let j = 0; j <= rows; j++) {
                const y = getIY(j);
                ctx.beginPath();
                ctx.moveTo(margin, y);
                ctx.lineTo(W - margin, y);
                ctx.stroke();
            }
            // Vertical
            for (let i = 0; i <= cols; i++) {
                const x = getIX(i);
                ctx.beginPath();
                ctx.moveTo(x, margin);
                ctx.lineTo(x, H - margin);
                ctx.stroke();
            }
            ctx.setLineDash([]);

            // Intersection boxes (pavement)
            for (const inter of intersections) {
                ctx.fillStyle = '#3a3a45';
                ctx.fillRect(inter.x - roadW / 2, inter.y - roadW / 2, roadW, roadW);
                // Crosswalk lines
                ctx.strokeStyle = 'rgba(255,255,255,0.15)';
                ctx.lineWidth = 2;
                for (let k = -3; k <= 3; k++) {
                    ctx.beginPath();
                    ctx.moveTo(inter.x - roadW / 2 + 5, inter.y + k * 5);
                    ctx.lineTo(inter.x + roadW / 2 - 5, inter.y + k * 5);
                    ctx.stroke();
                    ctx.beginPath();
                    ctx.moveTo(inter.x + k * 5, inter.y - roadW / 2 + 5);
                    ctx.lineTo(inter.x + k * 5, inter.y + roadW / 2 - 5);
                    ctx.stroke();
                }
            }
        }

        function drawTrafficLights(time) {
            const phase = getPhase();
            for (const inter of intersections) {
                const x = inter.x;
                const y = inter.y;
                // Draw traffic light poles
                ctx.fillStyle = '#555';
                // Four lights at each intersection
                const positions = [
                    { x: x - roadW / 2 - 4, y: y - roadW / 2 - 4, ns: true },
                    { x: x + roadW / 2 + 4, y: y + roadW / 2 + 4, ns: true },
                    { x: x + roadW / 2 + 4, y: y - roadW / 2 - 4, ns: false },
                    { x: x - roadW / 2 - 4, y: y + roadW / 2 + 4, ns: false }
                ];
                for (const pos of positions) {
                    ctx.fillStyle = '#222';
                    ctx.fillRect(pos.x - 3, pos.y - 6, 6, 12);
                    // Red
                    const isNS = pos.ns;
                    const showRed = isNS ? (phase === 2 || phase === 3) : (phase === 0 || phase === 1);
                    const showYellow = isNS ? (phase === 1) : (phase === 3);
                    const showGreen = isNS ? (phase === 0) : (phase === 2);
                    // Red
                    ctx.fillStyle = showRed ? 'rgba(255,50,50,0.9)' : 'rgba(60,20,20,0.6)';
                    ctx.beginPath();
                    ctx.arc(pos.x, pos.y - 3, 3, 0, Math.PI * 2);
                    ctx.fill();
                    // Yellow
                    ctx.fillStyle = showYellow ? 'rgba(255,220,50,0.9)' : 'rgba(60,50,20,0.6)';
                    ctx.beginPath();
                    ctx.arc(pos.x, pos.y, 3, 0, Math.PI * 2);
                    ctx.fill();
                    // Green
                    ctx.fillStyle = showGreen ? 'rgba(50,255,50,0.9)' : 'rgba(20,60,20,0.6)';
                    ctx.beginPath();
                    ctx.arc(pos.x, pos.y + 3, 3, 0, Math.PI * 2);
                    ctx.fill();
                }
            }
        }

        function drawSidewalks() {
            // Sidewalk strips around blocks
            for (let r = 0; r < rows; r++) {
                for (let c = 0; c < cols; c++) {
                    const bx = getBX(c);
                    const by = getBY(r);
                    ctx.fillStyle = '#4a4a55';
                    // Top
                    ctx.fillRect(bx - 6, by - 6, blockW + 12, 6);
                    // Bottom
                    ctx.fillRect(bx - 6, by + blockH, blockW + 12, 6);
                    // Left
                    ctx.fillRect(bx - 6, by, 6, blockH);
                    // Right
                    ctx.fillRect(bx + blockW, by, 6, blockH);
                }
            }
        }

        function drawTrees() {
            for (const tree of trees) {
                ctx.fillStyle = 'rgba(0,0,0,0.2)';
                ctx.beginPath();
                ctx.arc(tree.x + 1, tree.y + 1, tree.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.fillStyle = `hsl(${100 + Math.random() * 30}, 50%, ${35 + Math.random() * 20}%)`;
                ctx.beginPath();
                ctx.arc(tree.x, tree.y, tree.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        // ── Main loop ──
        let lastTime = 0;
        let spawnCounter = 0;

        function animate(time) {
            const dt = Math.min((time - lastTime) / 1000, 0.05);
            lastTime = time;

            // Update traffic light timer
            lightTime += dt;

            // Spawn cars based on density
            const density = trafficDensity();
            spawnCounter += density * dt * 4;
            if (spawnCounter >= 1) {
                spawnCar();
                spawnCounter -= 1;
            }

            // Spawn pedestrians occasionally
            if (Math.random() < dt * 0.5) {
                spawnPed();
            }

            // Update buildings
            for (const b of buildings) b.update(dt);

            // Update cars
            for (const car of cars) {
                car.update(dt);
            }

            // Update pedestrians
            for (const ped of peds) {
                ped.update(dt);
            }

            // Remove inactive
            for (let i = cars.length - 1; i >= 0; i--) {
                if (!cars[i].active) cars.splice(i, 1);
            }
            for (let i = peds.length - 1; i >= 0; i--) {
                if (!peds[i].active) peds.splice(i, 1);
            }

            // ── Draw ──
            ctx.fillStyle = '#1a1a2a';
            ctx.fillRect(0, 0, W, H);

            // Grass / park areas
            ctx.fillStyle = '#1e2a1e';
            for (let r = 0; r < rows; r++) {
                for (let c = 0; c < cols; c++) {
                    const bx = getBX(c);
                    const by = getBY(r);
                    ctx.fillRect(bx, by, blockW, blockH);
                }
            }

            drawSidewalks();
            drawRoads(time);

            // Draw buildings
            for (const b of buildings) b.draw(ctx, time);

            // Draw trees
            drawTrees();

            // Draw traffic lights
            drawTrafficLights(time);

            // Draw pedestrians
            for (const ped of peds) ped.draw(ctx);

            // Draw cars
            for (const car of cars) car.draw(ctx);

            // Overlay info
            ctx.fillStyle = 'rgba(0,0,0,0.5)';
            ctx.fillRect(8, 8, 180, 30);
            ctx.fillStyle = '#fff';
            ctx.font = '12px monospace';
            ctx.fillText(`Cars: ${cars.length}  Peds: ${peds.length}`, 14, 26);

            requestAnimationFrame(animate);
        }

        requestAnimationFrame(animate);
    </script>
</body>
</html>
```
### City Traffic Simulation

This page brings a miniature city to life with a focus on realistic traffic flow and visual feedback.

- **Traffic Light System:** A timed cycle controls red, yellow, and green lights at every intersection. Cars automatically stop at red lights and proceed on green, creating a safe and logical traffic flow.
- **Autonomous Vehicles:** Cars drive along the roads, making random but valid turns at intersections. Their speed and color vary, and the traffic density changes over time, simulating rush hour and quiet periods.
- **Living Environment:** Buildings have windows that randomly light up, and pedestrians cross at crosswalks. Trees and sidewalks add detail, while an on-screen counter shows the current number of cars and pedestrians, giving you real-time feedback on the simulation's activity.

---

**Optimization Tip:** You can adjust the `LIGHT_PHASES` array (line ~65) to change the traffic light timing, or modify the `trafficDensity()` function (line ~255) to alter how the number of cars fluctuates over time.
