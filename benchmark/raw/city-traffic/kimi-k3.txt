function glowSprite(size,r,g,b,a0){
  const c=document.createElement('canvas');c.width=c.height=size;
  const g=c.getContext('2d');
  const grd=g.createRadialGradient(size/2,size/2,0,size/2,size/2,size/2);

  grd.addColorStop(1,`rgba(${r},${g},${b},0)`);

  return c;
}

function lightAt(inter,t){
  const lt=((t+inter.off)%CYCLE+CYCLE)%CYCLE;
  if(lt<4.6)return{ns:'g',ew:'r'};
  if(lt<5.6)return{ns:'y',ew:'r'};
  if(lt<6.0)return{ns:'r',ew:'r'};
  if(lt<10.6)return{ns:'r',ew:'g'};
  if(lt<11.6)return{ns:'r',ew:'y'};
  return{ns:'r',ew:'r'};
}

/* ============================== cars ============================== */
function dirToAngle(d){return[0,Math.PI/2,Math.PI,-Math.PI/2][d];}
function maxCars(){return clamp(Math.round(inters.length*3.0),26,130);}

function spawnCar(anywhere){
  const vertical=Math.random()<0.5;
  let x,y,dir,road;
  if(vertical){
    const i=randi(0,xs.length-1),fromTop=Math.random()<0.5;
    dir=fromTop?1:3;
    x=xs[i]+(fromTop?-LANE:LANE);
    y=fromTop?-30:H+30;
    road={axis:'v',idx:i};
    if(anywhere)y=rand(20,H-20);
  }else{
    const j=randi(0,ys.length-1),fromLeft=Math.random()<0.5;
    dir=fromLeft?0:2;
    y=ys[j]+(fromLeft?LANE:-LANE);
    x=fromLeft?-30:W+30;
    road={axis:'h',idx:j};
    if(anywhere)x=rand(20,W-20);
  }
  for(const o of cars)if(Math.abs(o.x-x)<50&&Math.abs(o.y-y)<50)return false;
  const bus=Math.random()<0.1;
  cars.push({x,y,dir,road,speed:0,
    maxSpeed:bus?rand(38,46):rand(48,72),
    len:bus?26:rand(13.5,16),wid:bus?11.5:rand(9.5,11),
    col:bus?pick(['#d8b13c','#c8ccd6']):pick(CARCOL),
    bus,turn:null,plan:null,exiting:false,brake:false,heading:dirToAngle(dir)});
  if(anywhere)cars[cars.length-1].speed=cars[cars.length-1].maxSpeed*rand(0.5,1);
  return true;
}

function makePlan(c,inter){
  let type;
  if(c.exiting)type='straight';
  else{const r=Math.random();type=r<0.55?'straight':(r<0.80?'right':'left');}
  if(type==='straight')return{type,inter};
  const out=(c.dir+(type==='right'?1:3))%4;
  const vx=VX[c.dir],vy=VY[c.dir];
  const rnx=RNX[c.dir],rny=RNY[c.dir];
  const r=type==='right'?11:17;
  const px=inter.x+(rnx+RNX[out])*LANE,py=inter.y+(rny+RNY[out])*LANE;
  const t1x=px-vx*r,t1y=py-vy*r;
  const s=type==='right'?1:-1;
  const cx=t1x+rnx*r*s,cy=t1y+rny*r*s;
  const t2x=px+VX[out]*r,t2y=py+VY[out]*r;
  let th0=Math.atan2(t1y-cy,t1x-cx),th1=Math.atan2(t2y-cy,t2x-cx);
  while((th1-th0)*s<=0)th1+=TAU*s;
  const startDist=(inter.x-t1x)*vx+(inter.y-t1y)*vy;
  const road=(out===0||out===2)?{axis:'h',idx:inter.j}:{axis:'v',idx:inter.i};
  return{type,inter,startDist,arc:{cx,cy,r,th:th0,th1,s,out,x2:t2x,y2:t2y,road,tp:type}};
}

function updateCars(dt){
  for(let idx=cars.length-1;idx>=0;idx--){
    const c=cars[idx];
    if(c.turn){
      const tr=c.turn;
      const hx=Math.cos(c.heading),hy=Math.sin(c.heading);
      let blocked=false;
      for(const o of cars){
        if(o===c)continue;
        const dx=o.x-c.x,dy=o.y-c.y,ah=dx*hx+dy*hy;
        if(ah>0&&ah<(c.len+o.len)*0.5+8&&Math.abs(dx*-hy+dy*hx)<12){blocked=true;break;}
      }
      if(blocked){c.speed=0;c.brake=true;}
      else{
        c.brake=false;
        c.speed=Math.min(c.maxSpeed,c.speed+55*dt);
        tr.th+=tr.s*(Math.max(c.speed,16)/tr.r)*dt;
        const done=tr.s>0?tr.th>=tr.th1:tr.th<=tr.th1;
        if(done){c.x=tr.x2;c.y=tr.y2;c.dir=tr.out;c.road=tr.road;c.turn=null;c.heading=dirToAngle(c.dir);}
        else{c.x=tr.cx+tr.r*Math.cos(tr.th);c.y=tr.cy+tr.r*Math.sin(tr.th);c.heading=tr.th+tr.s*Math.PI/2;}
      }
      continue;
    }
    const vx=VX[c.dir],vy=VY[c.dir];
    const rnx=RNX[c.dir],rny=RNY[c.dir];
    let inter=null,dist=Infinity;
    if(c.road.axis==='h'){
      const Y=ys[c.road.idx];
      if(vx>0){for(const X of xs){const d=X-c.x;if(d>2&&d<dist){dist=d;inter=interXY.get(X+','+Y);}}}
      else{for(const X of xs){const d=c.x-X;if(d>2&&d<dist){dist=d;inter=interXY.get(X+','+Y);}}}
    }else{
      const X=xs[c.road.idx];
      if(vy>0){for(const Y of ys){const d=Y-c.y;if(d>2&&d<dist){dist=d;inter=interXY.get(X+','+Y);}}}
      else{for(const Y of ys){const d=c.y-Y;if(d>2&&d<dist){dist=d;inter=interXY.get(X+','+Y);}}}
    }
    if(inter&&c.plan&&c.plan.inter!==inter)c.plan=null;
    if(inter&&!c.plan&&dist<80)c.plan=makePlan(c,inter);
    let hz=Infinity;
    if(inter){
      const stopD=dist-(ROAD_W/2+13)-c.len/2;
      if(stopD>-3){
        const st=lightAt(inter,tNow);
        const sig=(c.dir===0||c.dir===2)?st.ew:st.ns;
        if(sig==='r'||(sig==='y'&&stopD>16))hz=Math.min(hz,Math.max(stopD,0));
        if(dist-c.len/2>ROAD_W/2){
          for(const o of cars){
            if(o===c)continue;
            if(Math.abs(o.x-inter.x)<ROAD_W/2+9&&Math.abs(o.y-inter.y)<ROAD_W/2+9){
              hz=Math.min(hz,Math.max(dist-ROAD_W/2-c.len/2-3,0));break;
            }
          }
          if(c.plan&&c.plan.type==='left'){
            const od=(c.dir+2)%4;
            for(const o of cars){
              if(o===c||o.turn||o.dir!==od)continue;
              if(o.road.axis===c.road.axis&&o.road.idx===c.road.idx){
                const od2=c.road.axis==='h'?Math.abs(o.x-inter.x):Math.abs(o.y-inter.y);
                if(od2<85){hz=Math.min(hz,Math.max(stopD,0));break;}
              }
            }
          }
        }
      }
    }
    for(const o of cars){
      if(o===c)continue;
      const dx=o.x-c.x,dy=o.y-c.y,ahead=dx*vx+dy*vy;
      if(ahead<=0||ahead>82)continue;
      if(Math.abs(dx*rnx+dy*rny)<13)hz=Math.min(hz,Math.max(ahead-(c.len+o.len)*0.35,0));
    }
    c.brake=hz<45;
    let ts=c.maxSpeed;
    if(hz<52)ts=c.maxSpeed*clamp((hz-7)/45,0,1);
    if(c.speed<ts)c.speed=Math.min(ts,c.speed+62*dt);
    else c.speed=Math.max(ts,c.speed-175*dt);
    c.x+=vx*c.speed*dt;c.y+=vy*c.speed*dt;
    c.heading=dirToAngle(c.dir);
    if(c.plan&&c.plan.type!=='straight'&&inter===c.plan.inter&&dist<=c.plan.startDist){c.turn=c.plan.arc;c.plan=null;}
    if(c.x<-70||c.x>W+70||c.y<-70||c.y>H+70)cars.splice(idx,1);
  }
  const target=Math.round(lerp(maxCars()*0.3,maxCars(),densitySm));
  spawnT-=dt;
  if(cars.length<target&&spawnT<=0)spawnT=spawnCar(false)?rand(0.12,0.5)/(0.3+densitySm):0.25;
  if(cars.length>target+8){const cand=cars.filter(c=>!c.exiting);if(cand.length)pick(cand).exiting=true;}
}

/* ============================== pedestrians ============================== */
function cornerPos(i,j,c){
  return{x:xs[i]+((c===1||c===2)?COFF:-COFF),y:ys[j]+((c>=2)?COFF:-COFF)};
}
function spawnPed(){
  if(!inters.length)return;
  const it=pick(inters),c=randi(0,3),cp=cornerPos(it.i,it.j,c);
  peds.push({i:it.i,j:it.j,corner:c,x:cp.x+rand(-2,2),y:cp.y+rand(-2,2),
    tx:cp.x,ty:cp.y,ti:it.i,tj:it.j,tc:c,state:'choose',
    speed:rand(22,38),col:pick(PEDCOL),ph:rand(0,TAU),crossing:false,needAxis:null});
}
function pedOptions(p){
  const i=p.i,j=p.j,c=p.corner,opts=[];
  if(c===0){opts.push({t:{i,j,c:1},cross:'ns'},{t:{i,j,c:3},cross:'ew'});
    if(j>0)opts.push({t:{i,j:j-1,c:3}});if(i>0)opts.push({t:{i:i-1,j,c:1}});}
  else if(c===1){opts.push({t:{i,j,c:0},cross:'ns'},{t:{i,j,c:2},cross:'ew'});
    if(j>0)opts.push({t:{i,j:j-1,c:2}});if(i<xs.length-1)opts.push({t:{i:i+1,j,c:0}});}
  else if(c===2){opts.push({t:{i,j,c:3},cross:'ns'},{t:{i,j,c:1},cross:'ew'});
    if(j<ys.length-1)opts.push({t:{i,j:j+1,c:1}});if(i<xs.length-1)opts.push({t:{i:i+1,j,c:3}});}
  else{opts.push({t:{i,j,c:2},cross:'ns'},{t:{i,j,c:0},cross:'ew'});
    if(j<ys.length-1)opts.push({t:{i,j:j+1,c:0}});if(i>0)opts.push({t:{i:i-1,j,c:2}});}
  return opts;
}
function updatePeds(dt){
  const targetN=clamp(Math.round(lerp(inters.length*0.55,inters.length*1.15,densitySm)),8,72);
  let guard=0;
  while(peds.length<targetN&&guard++<6)spawnPed();
  while(peds.length>targetN&&peds.length>0)peds.splice((Math.random()*peds.length)|0,1);
  for(const p of peds){
    p.ph+=dt*(p.state==='walk'?10:2);
    if(p.state==='walk'){
      const dx=p.tx-p.x,dy=p.ty-p.y,d=Math.hypot(dx,dy);
      const sp=p.speed*(p.crossing?1.25:1);
      if(d<2){p.x=p.tx;p.y=p.ty;p.i=p.ti;p.j=p.tj;p.corner=p.tc;p.state='choose';p.crossing=false;}
      else{p.x+=dx/d*sp*dt;p.y+=dy/d*sp*dt;}
    }else if(p.state==='wait'){
      const inter=interIJ.get(p.i+','+p.j);
      const st=lightAt(inter,tNow);
      const ok=p.needAxis==='ns'?st.ns==='r':st.ew==='r';
      if(ok){p.state='walk';p.crossing=true;}
    }else{
      const o=pick(pedOptions(p));
      const cp=cornerPos(o.t.i,o.t.j,o.t.c);
      p.tx=cp.x+rand(-2.5,2.5);p.ty=cp.y+rand(-2.5,2.5);
      p.ti=o.t.i;p.tj=o.t.j;p.tc=o.t.c;
      if(o.cross){
        p.needAxis=o.cross;
        const inter=interIJ.get(p.i+','+p.j);
        const st=lightAt(inter,tNow);
        const ok=o.cross==='ns'?st.ns==='r':st.ew==='r';
        p.state=ok?'walk':'wait';p.crossing=ok;
      }else{p.state='walk';p.crossing=false;}
    }
  }
}

/* ============================== update ============================== */
function densityAt(h){
  const g=(c,w)=>Math.exp(-((h-c)*(h-c))/(2*w*w));
  return clamp(0.14+0.75*g(17.6,1.15)+0.38*g(20.6,1.5)+0.22*g(14.8,1.2),0,1);
}
function initTraffic(){
  cars=[];peds=[];
  const n0=Math.round(maxCars()*0.55);
  for(let k=0;k<n0;k++)spawnCar(true);
  const np=clamp(Math.round(inters.length*0.8),8,60);
  for(let k=0;k<np;k++)spawnPed();
}
function update(dt){
  tNow+=dt;
  simMin+=dt*2.4;
  hourNow=START_HOUR+simMin/60;
  while(hourNow>=26)hourNow-=12;
  density=densityAt(hourNow);
  densitySm+=(density-densitySm)*Math.min(1,dt*0.55);
  updateCars(dt);
  updatePeds(dt);
  for(const w of winDyn){
    if(tNow>=w.next){w.on=Math.random()<0.52;w.next=tNow+rand(2.5,16);}
    w.br+=((w.on?1:0)-w.br)*Math.min(1,dt*3.5);
  }
}
