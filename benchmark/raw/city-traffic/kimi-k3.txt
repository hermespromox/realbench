function shade(hex,f){
  const n=parseInt(hex.slice(1),16);
  let r=(n>>16)&255,g=(n>>8)&255,b=n&255;
  if(f<0){r*=1+f;g*=1+f;b*=1+f;}else{r+=(255-r)*f;g+=(255-g)*f;b+=(255-b)*f;}
  return 'rgb('+(r|0)+','+(g|0)+','+(b|0)+')';
}
function rr(g,x,y,w,h,r){
  g.beginPath();g.moveTo(x+r,y);
  g.arcTo(x+w,y,x+w,y+h,r);g.arcTo(x+w,y+h,x,y+h,r);
  g.arcTo(x,y+h,x,y,r);g.arcTo(x,y,x+w,y,r);
  g.closePath();
}

  for(let i=0;i<nv;i++){
    grid[i]=[];
    for(let j=0;j<nh;j++){
      const it={i,j,x:vRoads[i],y:hRoads[j],off:((i*1.5+j*1.5)%CYCLE+CYCLE)%CYCLE};
      it.sigPts=[
        {x:it.x+RW/2+4.5,y:it.y-RW/2-4.5,ax:'ns'},

function buildBlocks(){
  buildings=[];wins=[];beacons=[];blockRects=[];
  const xi=[];let p=0;
  for(const x of vRoads){xi.push([p,x-RW/2]);p=x+RW/2;}xi.push([p,W]);
  const yi=[];p=0;
  for(const y of hRoads){yi.push([p,y-RW/2]);p=y+RW/2;}yi.push([p,H]);
  const roofCols=['#2e3547','#35314b','#2c3a44','#3b352f','#273241','#3a3652','#313b36','#403a33'];
  for(const [x0,x1] of xi)for(const [y0,y1] of yi){
    blockRects.push({x0,y0,x1,y1});
    const bx0=x0+SW,by0=y0+SW,bx1=x1-SW,by1=y1-SW;
    const bw=bx1-bx0,bh=by1-by0;
    if(bw<40||bh<40)continue;
    const g2=6,cells=[];
    const sX=bw>110&&Math.random()<0.7, sY=bh>90&&Math.random()<0.7;
    if(sX&&sY){
      const mx=bx0+bw*rand(0.42,0.58),my=by0+bh*rand(0.42,0.58);
      cells.push([bx0,by0,mx-g2/2,my-g2/2],[mx+g2/2,by0,bx1,my-g2/2],[bx0,my+g2/2,mx-g2/2,by1],[mx+g2/2,my+g2/2,bx1,by1]);
    }else if(sX){
      const mx=bx0+bw*rand(0.42,0.58);
      cells.push([bx0,by0,mx-g2/2,by1],[mx+g2/2,by0,bx1,by1]);
    }else if(sY){
      const my=by0+bh*rand(0.42,0.58);
      cells.push([bx0,by0,bx1,my-g2/2],[bx0,my+g2/2,bx1,by1]);
    }else cells.push([bx0,by0,bx1,by1]);
    for(const [cx0,cy0,cx1,cy1] of cells){
      const cw2=cx1-cx0,ch2=cy1-cy0;
      if(cw2<26||ch2<26)continue;
      const col=roofCols[(Math.random()*roofCols.length)|0];
      const tall=Math.random()<0.25;
      const b={x:cx0,y:cy0,w:cw2,h:ch2,col,tall,
        ac:Math.random()<0.7?{x:cx0+rand(6,Math.max(7,cw2-16)),y:cy0+rand(6,Math.max(7,ch2-16)),w:rand(6,12),h:rand(5,9)}:null,
        tank:!tall&&Math.random()<0.28?{x:cx0+rand(8,cw2-8),y:cy0+rand(8,ch2-8)}:null,
        garden:!tall&&Math.random()<0.16,
        heli:tall&&Math.random()<0.35,
        beacon:tall&&Math.random()<0.6?{x:cx0+cw2/2,y:cy0+ch2/2,ph:rand(6.28)}:null};
      buildings.push(b);
      if(b.beacon)beacons.push(b.beacon);
      for(let wx=cx0+4;wx<cx1-4;wx+=7) wins.push({x:wx,y:cy1+2.5,on:Math.random()<0.5,c:(Math.random()*3)|0});
      for(let wy=cy0+4;wy<cy1-4;wy+=7) wins.push({x:cx1+2,y:wy,on:Math.random()<0.45,c:(Math.random()*3)|0});
    }
  }
  // parked cars
  parked=[];
  const pcols=['#5d6672','#7a4a42','#425a7a','#6b6b6b','#4a5d50','#8a8f99'];
  for(let i=0;i<vRoads.length;i++){
    const x=vRoads[i];
    for(const side of [-1,1]){
      let y=rand(30,90);
      while(y<H-30){
        let ok=true;
        for(const hy of hRoads)if(Math.abs(y-hy)<RW/2+30){ok=false;break;}
        if(ok&&Math.random()<0.6)parked.push({x:x+side*(RW/2-4.5),y,ang:Math.random()<0.5?Math.PI/2:-Math.PI/2,col:pcols[(Math.random()*pcols.length)|0]});
        y+=rand(48,125);
      }
    }
  }
  for(let j=0;j<hRoads.length;j++){
    const y=hRoads[j];
    for(const side of [-1,1]){
      let x=rand(30,90);
      while(x<W-30){
        let ok=true;
        for(const vx of vRoads)if(Math.abs(x-vx)<RW/2+30){ok=false;break;}

        x+=rand(48,125);
      }
    }
  }
}

/* ---------------- signals ---------------- */
function signalAt(it,t){
  const ph=(((t+it.off)%CYCLE)+CYCLE)%CYCLE;
  let ns='r',ew='r';
  if(ph<NS_G)ns='g';
  else if(ph<NS_Y)ns='y';
  else if(ph<EW_G0){}
  else if(ph<EW_G1)ew='g';
  else if(ph<EW_Y)ew='y';
  return {ns,ew,ph};
}

/* ---------------- traffic ---------------- */
function trafficLevel(t){
  return clamp(0.58+0.34*Math.sin(t*0.14)+0.10*Math.sin(t*0.043+2.0),0.07,1);
}
function pedLevel(t){
  return clamp(0.5+0.38*Math.sin(t*0.11+1.2)+0.08*Math.sin(t*0.31),0.06,1);
}
function laneKey(c){return c.axis+':'+c.idx+':'+c.dir;}
function carXY(c){
  if(c.axis==='v')return{x:vRoads[c.idx]+(-c.dir)*LO,y:c.pos,ang:c.dir<0?-Math.PI/2:Math.PI/2};
  return{x:c.pos,y:hRoads[c.idx]+c.dir*LO,ang:c.dir>0?0:Math.PI};
}
function makeCar(axis,idx,dir,pos){
  let type='car',color;
  const r=Math.random();
  const hasPolice=cars.some(c=>c.type==='police');
  if(r<0.06){type='bus';color=busCols[(Math.random()*busCols.length)|0];}
  else if(r<0.14){type='taxi';color='#d9a83c';}
  else if(r<0.18&&!hasPolice){type='police';color='#e8ecf2';}
  else color=carCols[(Math.random()*carCols.length)|0];
  return{axis,idx,dir,pos,
    len:type==='bus'?26:16,
    speed:0,
    v0:type==='bus'?rand(42,58):type==='police'?rand(80,95):rand(55,85),
    type,color,spr:carSprite(color,type),
    turn:null,curInter:null,decision:'S',turnTo:null,brake:false,
    _x:0,_y:0,_a:0};
}
function entryClear(L){
  const entry=L.dir>0?-30:(L.axis==='v'?H+30:W+30);
  for(const c of cars){
    if(c.turn)continue;
    if(c.axis===L.axis&&c.idx===L.idx&&c.dir===L.dir){
      if((c.pos-entry)*L.dir<60)return false;
    }
  }
  return true;
}
function trySpawnCars(dt){
  const lvl=trafficLevel(simTime);
  const maxC=Math.floor(MAX_CARS*lvl);
  if(cars.length>=maxC)return;
  const counts=new Map();
  for(const c of cars){
    if(c.turn)continue;
    const k=laneKey(c);counts.set(k,(counts.get(k)||0)+1);
  }
  for(const L of lanes){
    if(cars.length>=maxC)break;
    const k=L.axis+':'+L.idx+':'+L.dir;
    const cap=Math.min(L.len/60,8)*lvl+0.4;
    if((counts.get(k)||0)>=cap)continue;
    if(Math.random()<2.2*lvl*dt&&entryClear(L)){
      const pos=L.dir>0?-30:(L.axis==='v'?H+30:W+30);
      const nc=makeCar(L.axis,L.idx,L.dir,pos);
      nc.speed=nc.v0*0.7;
      cars.push(nc);
      counts.set(k,(counts.get(k)||0)+1);
    }
  }
}
function nextInter(c){
  if(c.axis==='v'){
    let bestD=Infinity,bj=-1;
    for(let j=0;j<hRoads.length;j++){
      const d=(hRoads[j]-c.pos)*c.dir;
      if(d>-(RW/2+c.len+2)&&d<bestD){bestD=d;bj=j;}
    }
    if(bj<0)return null;
    return{inter:grid[c.idx][bj],coord:hRoads[bj]};
  }
  let bestD=Infinity,bi=-1;
  for(let i=0;i<vRoads.length;i++){
    const d=(vRoads[i]-c.pos)*c.dir;
    if(d>-(RW/2+c.len+2)&&d<bestD){bestD=d;bi=i;}
  }
  if(bi<0)return null;
  return{inter:grid[bi][c.idx],coord:vRoads[bi]};
}
function decideTurn(c,inter){
  const r=Math.random();
  const t=r<0.56?'S':(r<0.78?'L':'R');
  if(t==='S')return 'S';
  const newAxis=c.axis==='v'?'h':'v';
  const newIdx=c.axis==='v'?inter.j:inter.i;
  const newDir=c.axis==='v'?(t==='R'?-c.dir:c.dir):(t==='R'?c.dir:-c.dir);
  const along=c.axis==='v'?inter.x:inter.y;
  const exitPos=along+newDir*(RW/2+8);
  c.turnTo={axis:newAxis,idx:newIdx,dir:newDir,pos:exitPos};
  return t;
}
function exitClear(c){
  const T=c.turnTo;
  for(const o of cars){
    if(o===c||o.turn)continue;
    if(o.axis===T.axis&&o.idx===T.idx&&o.dir===T.dir){
      if(Math.abs(o.pos-T.pos)<26)return false;
    }
  }
  return true;
}
function oncomingBusy(c,inter){
  const coord=c.axis==='v'?inter.y:inter.x;
  for(const o of cars){
    if(o===c||o.turn)continue;
    if(o.axis===c.axis&&o.idx===c.idx&&o.dir===-c.dir){
      if(o.decision==='L'&&o.speed<3)continue;
      const d=(coord-o.pos)*o.dir;
      if(d>-(RW/2+26)&&d<RW/2+34&&(o.speed>3||d<RW/2+14))return true;
    }
  }
  return false;
}
function startTurn(c){
  const T=c.turnTo;
  const P0=carXY(c);
  let P1,P2;
  if(c.axis==='v'){
    P2={x:T.pos,y:hRoads[T.idx]+T.dir*LO};
    P1={x:vRoads[c.idx]+(-c.dir)*LO,y:hRoads[T.idx]+T.dir*LO};
  }else{
    P2={x:vRoads[T.idx]+(-T.dir)*LO,y:T.pos};
    P1={x:vRoads[T.idx]+(-T.dir)*LO,y:hRoads[c.idx]+c.dir*LO};
  }
  const approx=Math.hypot(P1.x-P0.x,P1.y-P0.y)+Math.hypot(P2.x-P1.x,P2.y-P1.y);
  const spd=clamp(c.speed*1.1,24,42);
  c.speed=spd;
  c.turn={t:0,dur:Math.max(0.35,approx/spd),p0:{x:P0.x,y:P0.y},p1:P1,p2:P2,to:T};
}
function turnPose(c){
  const T=c.turn,u=clamp(T.t,0,1),mt=1-u;
  const x=mt*mt*T.p0.x+2*mt*u*T.p1.x+u*u*T.p2.x;
  const y=mt*mt*T.p0.y+2*mt*u*T.p1.y+u*u*T.p2.y;
  const dx=2*mt*(T.p1.x-T.p0.x)+2*u*(T.p2.x-T.p1.x);
  const dy=2*mt*(T.p1.y-T.p0.y)+2*u*(T.p2.y-T.p1.y);
  return{x,y,ang:Math.atan2(dy,dx)};
}
function updateCar(c,ahead,dt){
  let target=c.v0;
  if(ahead){
    const gap=(ahead.pos-c.pos)*c.dir-(ahead.len+c.len)/2;
    const des=5+c.speed*0.35;
    if(gap<des){
      target=Math.min(target,ahead.speed*Math.max(0,gap/des)*1.15);
      if(gap<2.5)target=0;
    }
  }
  const ni=nextInter(c);
  if(ni){
    if(!c.curInter||c.curInter.inter!==ni.inter){
      c.curInter=ni;c.decision=decideTurn(c,ni.inter);
    }
  }else c.curInter=null;
  if(c.curInter){
    const inter=c.curInter.inter,coord=c.curInter.coord;
    const sig=signalAt(inter,simTime);
    const my=c.axis==='v'?sig.ns:sig.ew;
    const dBox=(coord-c.pos)*c.dir;
    const dStop=dBox-(RW/2+13);
    const committed=dBox<RW/2+13&&dBox>-(RW/2+c.len+2);
    if(!committed&&my!=='g'&&dStop>-1){
      let stopIt=true;
      if(my==='y'&&dStop<16&&c.speed>34)stopIt=false;
      if(stopIt){
        const bd=c.speed*c.speed/(2*BRAKE)+10;
        if(dStop<bd)target=Math.min(target,Math.max(0,(dStop-c.len/2-2)*2.5));
      }
    }
    if(c.decision&&c.decision!=='S'&&!c.turn&&c.turnTo){
      const canGo=my==='g'||committed;
      if(canGo&&dBox<=RW/2-3&&dBox>RW/2-26){
        if(c.decision==='L'&&oncomingBusy(c,inter)){
          target=0;
        }else if(exitClear(c)){
          startTurn(c);
        }else{
          c.decision='S';
        }
      }
    }
  }
  if(target>c.speed)c.speed=Math.min(target,c.speed+ACCEL*dt);
  else c.speed=Math.max(target,c.speed-BRAKE*dt);
  c.brake=(target<c.speed-1)||(c.speed<2.5);
  c.pos+=c.dir*c.speed*dt;
  if(ahead){
    const minGap=(ahead.len+c.len)/2+1.5;
    if((ahead.pos-c.pos)*c.dir<minGap){
      c.pos=ahead.pos-c.dir*minGap;
      c.speed=Math.min(c.speed,ahead.speed);
    }
  }
}
function updateCars(dt){
  const byLane=new Map();
  for(const c of cars){
    if(c.turn)continue;
    const k=laneKey(c);
    let a=byLane.get(k);
    if(!a){a=[];byLane.set(k,a);}
    a.push(c);
  }
  for(const a of byLane.values()){
    a.sort((x,y)=>x.pos-y.pos);
    if(a[0].dir>0){for(let k=a.length-1;k>=0;k--)updateCar(a[k],a[k+1]||null,dt);}
    else{for(let k=0;k<a.length;k++)updateCar(a[k],a[k-1]||null,dt);}
  }
  for(const c of cars){
    if(!c.turn)continue;
    c.turn.t+=dt/c.turn.dur;
    if(c.turn.t>=1){
      const T=c.turn.to;
      c.axis=T.axis;c.idx=T.idx;c.dir=T.dir;c.pos=T.pos;
      c.speed=Math.max(22,c.speed);
      c.turn=null;c.curInter=null;c.decision='S';c.turnTo=null;
    }
  }
  cars=cars.filter(c=>{
    if(c.turn)return true;
    if(c.axis==='v')return c.pos>-50&&c.pos<H+50;
    return c.pos>-50&&c.pos<W+50;
  });
}

/* ---------------- pedestrians ---------------- */
const pedCols=['#c7a27a','#7a9ec7','#c77a8e','#8ec77a','#c7c07a','#a08ec7','#d4d4d4','#e08a4a'];
function spawnPed(){
  if(peds.length>8+34*pedLevel(simTime))return;
  const it=inters[(Math.random()*inters.length)|0];
  const cross=['N','S','E','W'][(Math.random()*4)|0];
  peds.push({it,cross,side:Math.random()<0.5?0:1,prog:0,state:'wait',
    sp:rand(16,30),col:pedCols[(Math.random()*pedCols.length)|0],
    ph:rand(6.28),off:rand(-2,2)});
}
function pedPath(p){
  const it=p.it;
  if(p.cross==='N'||p.cross==='S'){
    const y=it.y+(p.cross==='N'?-(RW/2+6):(RW/2+6));
    return{axisH:true,a:it.x-(RW/2+9),b:it.x+(RW/2+9),fixed:y};
  }
  const x=it.x+(p.cross==='W'?-(RW/2+6):(RW/2+6));
  return{axisH:false,a:it.y-(RW/2+9),b:it.y+(RW/2+9),fixed:x};
}
function canWalk(p){
  const sig=signalAt(p.it,simTime);
  if(p.cross==='N'||p.cross==='S')
    return sig.ew==='g'&&sig.ph<EW_G0+(EW_G1-EW_G0)*0.72;
  return sig.ns==='g'&&sig.ph<NS_G*0.72;
}
function updatePeds(dt){
  if(Math.random()<2.0*pedLevel(simTime)*dt)spawnPed();
  for(let i=peds.length-1;i>=0;i--){
    const p=peds[i];
    if(p.state==='wait'){
      if(canWalk(p))p.state='cross';
    }else{
      const path=pedPath(p);
      p.prog+=p.sp*dt/Math.abs(path.b-path.a);
      if(p.prog>=1){
        if(Math.random()<0.14){p.prog=0;p.side^=1;p.state='wait';}
        else peds.splice(i,1);
      }
    }
  }
}
function pedPos(p){
  const path=pedPath(p);
  const t=p.side===0?p.prog:1-p.prog;
  let along=lerp(path.a,path.b,t);
  const perp=p.off+(p.state==='cross'?Math.sin(simTime*9+p.ph)*0.9:Math.sin(simTime*2.2+p.ph)*0.5);
  if(p.state==='wait')along=p.side===0?path.a-1.5:path.b+1.5;
  if(path.axisH)return{x:along,y:path.fixed+perp};
  return{x:path.fixed+perp,y:along};
}

/* ---------------- clouds ---------------- */
function initClouds(){
  clouds=[];
  for(let k=0;k<4;k++)clouds.push({x:rand(W),y:rand(H),r:rand(200,380),vx:rand(5,13),vy:rand(-3,3)});
}
function updateClouds(dt){
  for(const cl of clouds){
    cl.x+=cl.vx*dt;cl.y+=cl.vy*dt;
    if(cl.x-cl.r>W+60)cl.x=-cl.r-60;
    if(cl.y-cl.r>H+60)cl.y=-cl.r-60;
    if(cl.y+cl.r<-60)cl.y=H+cl.r+60;
  }
}

/* ---------------- render ---------------- */
const winCols=['#ffcf7d','#ffe2a6','#a8d4ff'];
function drawWindows(dt){
  const n=wins.length,tog=0.10*dt;
  for(let i=0;i<n;i++)if(Math.random()<tog)wins[i].on=!wins[i].on;
  ctx.globalAlpha=0.78+0.14*Math.sin(simTime*0.7);
  for(let ci=0;ci<3;ci++){
    ctx.fillStyle=winCols[ci];
    ctx.beginPath();
    for(let i=0;i<n;i++){
      const w=wins[i];
      if(w.on&&w.c===ci)ctx.rect(w.x-1.1,w.y-0.85,2.2,1.7);
    }
    ctx.fill();
  }
  ctx.globalAlpha=1;
  ctx.fillStyle='rgba(255,70,60,0.5)';
  for(const b of beacons){ctx.beginPath();ctx.arc(b.x,b.y,1.3,0,7);ctx.fill();}
}
function drawCars(){
  for(const c of cars){
    const pose=c.turn?turnPose(c):carXY(c);
    c._x=pose.x;c._y=pose.y;c._a=pose.ang;
    const spr=c.spr;
    ctx.save();
    ctx.translate(pose.x,pose.y);
    ctx.rotate(pose.ang);
    ctx.drawImage(spr,-spr.sprW/2,-spr.sprH/2,spr.sprW,spr.sprH);
    ctx.restore();
  }
}
function drawPeds(){
  for(const p of peds){
    const pos=pedPos(p);
    ctx.fillStyle='rgba(0,0,0,0.45)';
    ctx.beginPath();ctx.ellipse(pos.x+0.7,pos.y+1,2.4,1.7,0,0,7);ctx.fill();
    ctx.fillStyle=p.col;
    ctx.beginPath();ctx.arc(pos.x,pos.y,2.3,0,7);ctx.fill();
    ctx.fillStyle='rgba(255,255,255,0.35)';
    ctx.beginPath();ctx.arc(pos.x-0.6,pos.y-0.7,0.9,0,7);ctx.fill();
  }
}
function drawSignalCores(){
  for(const it of inters){
    const s=signalAt(it,simTime);
    for(const pt of it.sigPts){
      const st=pt.ax==='ns'?s.ns:s.ew;
      ctx.fillStyle=sigCol[st];
      ctx.beginPath();ctx.arc(pt.x,pt.y,1.7,0,7);ctx.fill();
    }
  }
}
function drawGlowPass(){
  ctx.globalCompositeOperation='lighter';
  for(const it of inters){
    const s=signalAt(it,simTime);
    for(const pt of it.sigPts){
      const st=pt.ax==='ns'?s.ns:s.ew;
      ctx.globalAlpha=0.5;
      ctx.drawImage(sigGlow[st],pt.x-8,pt.y-8,16,16);
    }
  }
  for(const c of cars){
    if(c.brake){
      const dx=Math.cos(c._a),dy=Math.sin(c._a);
      const rx=c._x-dx*c.len/2,ry=c._y-dy*c.len/2;
      ctx.globalAlpha=0.55;
      ctx.drawImage(glowBrake,rx-7,ry-7,14,14);
    }
    if(c.type==='police'){
      const dx=Math.cos(c._a),dy=Math.sin(c._a);
      const px=-dy,py=dx,on=Math.sin(simTime*11)>0;
      ctx.globalAlpha=0.85;
      ctx.drawImage(on?glowRed:glowBlue,c._x+px*2.2-8,c._y+py*2.2-8,16,16);
      ctx.drawImage(on?glowBlue:glowRed,c._x-px*2.2-8,c._y-py*2.2-8,16,16);
    }
  }
  for(const b of beacons){
    const a=Math.sin(simTime*2.2+b.ph);
    if(a>0.3){
      ctx.globalAlpha=(a-0.3)*0.9;
      ctx.drawImage(glowRed,b.x-10,b.y-10,20,20);
    }
  }
  ctx.globalCompositeOperation='source-over';
  ctx.globalAlpha=1;
}
function drawClouds(){
  for(const cl of clouds){
    ctx.globalAlpha=0.15;
    ctx.drawImage(cloudSprite,cl.x-cl.r,cl.y-cl.r,cl.r*2,cl.r*2);
  }
  ctx.globalAlpha=1;
}
function render(dt){
  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle='#0b0d12';
  ctx.fillRect(0,0,canvas.width,canvas.height);
  const z=1.045+0.035*Math.sin(simTime*0.09);
  const zx=W/2*(1-z)+14*Math.sin(simTime*0.05);
  const zy=H/2*(1-z)+10*Math.cos(simTime*0.043);
  ctx.setTransform(DPR*z,0,0,DPR*z,DPR*zx,DPR*zy);
  ctx.drawImage(staticC,0,0,W,H);
  drawWindows(dt);
  drawCars();
  drawPeds();
  drawSignalCores();
  drawGlowPass();
  drawClouds();
}

/* ---------------- init / loop ---------------- */
function initTraffic(){
  const lvl=trafficLevel(0);
  for(const L of lanes){
    const n=Math.floor(Math.min(L.len/70,6)*lvl*Math.random()+1);
    let pos=rand(40,120);
    for(let k=0;k<n;k++){
      const wp=L.dir>0?pos:L.len-pos;
      const nc=makeCar(L.axis,L.idx,L.dir,wp);
      nc.speed=nc.v0*0.8;
      cars.push(nc);
      pos+=rand(95,230);
      if(pos>L.len-60)break;
    }
  }
  for(let k=0;k<14;k++)spawnPed();
  for(const p of peds)if(Math.random()<0.4){p.state='cross';p.prog=rand(0.1,0.8);}
}
function resize(){
  DPR=Math.min(window.devicePixelRatio||1,1.5);
  W=window.innerWidth;H=window.innerHeight;
  canvas.width=Math.round(W*DPR);canvas.height=Math.round(H*DPR);
  canvas.style.width=W+'px';canvas.style.height=H+'px';
  buildLayout();
  buildBlocks();
  renderStatic();
  cars=[];peds=[];
  initTraffic();
  initClouds();
}
window.addEventListener('resize',resize);

initSprites();
resize();

let last=performance.now();
function frame(now){
  let dt=(now-last)/1000;
  last=now;
  dt=Math.min(dt,0.05);
  simTime+=dt;
  trySpawnCars(dt);
  updateCars(dt);
  updatePeds(dt);
  updateClouds(dt);
  render(dt);
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
</script>
