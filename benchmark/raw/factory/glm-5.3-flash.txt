function jointsOf(a){
  const dx=a.pos[0]-a.bx,dy=a.pos[1]-a.by;
  const yaw=Math.atan2(dy,dx),r=Math.hypot(dx,dy),bz=1.06,dz=a.pos[2]-bz;
  let D=Math.hypot(r,dz);const D0=D||1;
  D=Math.min(D,(a.L1+a.L2)*0.999);
  const ur=r/D0,uz=dz/D0;
  const wr=[a.bx+Math.cos(yaw)*ur*D,a.by+Math.sin(yaw)*ur*D,bz+uz*D];
  const base=Math.atan2(wr[2]-bz,Math.hypot(wr[0]-a.bx,wr[1]-a.by));
  const a1=base+Math.acos(clamp((D*D+a.L1*a.L1-a.L2*a.L2)/(2*D*a.L1),-1,1));

function drawCart(c,isTwo){

  const cos=Math.cos(c.yaw),sin=Math.sin(c.yaw);
  const loc=(lx,ly)=>[c.x+cos*lx-sin*ly,c.y+sin*lx+cos*ly];
  const ug=PP(c.x,c.y,0.02);glow(ug[0],ug[1],T*0.9,'#35e0ff',0.18*c.alpha);
  for(const s of[[-0.55,-0.5],[-0.55,0.5],[0.55,-0.5],[0.55,0.5]]){

  if(!isTwo){
    for(let i=0;i<3;i++){

      if(c.cargo[i])cyl(slot[0],slot[1],0.40,0.28,0.14,C.puck);
      else if(c.state==='loading'){
        const u=clamp((c.loadT-i*0.33)/0.42,0,1);
        if(u>0&&u<1)cyl(lerp(0.95,slot[0],easeIO(u)),lerp(7.0,slot[1],easeIO(u)),
          lerp(0.75,0.4,easeIO(u))+Math.sin(u*Math.PI)*0.4,0.28,0.14,C.puck);}
      else if(c.state==='unloading'){

          const tz=1.5;
          cyl(lerp(slot[0],MAG.x,easeIO(u)),lerp(slot[1],MAG.y,easeIO(u)),
            lerp(0.4,tz,easeIO(u))+Math.sin(u*Math.PI)*0.3,0.28,0.14,C.puck);}}}}
  else{for(const p of parts)if(p.state==='onCart')drawUnit(p.x,p.y,p.z,p);}
