  <canvas id="matrixCanvas"></canvas>

  <script>
    (function () {
      const canvas = document.getElementById('matrixCanvas');
      const ctx = canvas.getContext('2d', { alpha: false });

      // Authentic Matrix glyph set: Half-width Katakana, Latin, Digits, Math/Currency symbols
      const GLYPHS = 
        'ｦｱｳｴｵｶｷｹｺｻｼｽｾｿﾀﾂﾃﾅﾆﾇﾈﾊﾋﾎﾏﾐﾑﾒﾓﾔﾕﾗﾘﾜ' +
        '0123456789' +
        ':・.=*+-<>¦｜~Z!X#&%_';

      function getRandomChar() {
        return GLYPHS.charAt(Math.floor(Math.random() * GLYPHS.length));
      }

      let width, height;
      let fontSize, colWidth, numCols, numRows;
      let columns = [];

      class Drop {
        constructor(colIndex, isInitial = false) {
          this.colIndex = colIndex;
          this.init(isInitial);
        }

        init(isInitial = false) {
          // Speed variation: some drops plunge, others drift
          this.speed = 0.25 + Math.random() * 0.75;
          this.length = Math.floor(14 + Math.random() * 26);
          // Lead position in row units
          this.y = isInitial ? Math.random() * (numRows + this.length) - this.length : -Math.random() * 30 - 2;
          // Mutation tick rates
          this.mutateTimer = 0;
          this.mutateInterval = Math.floor(2 + Math.random() * 5);
          // Individual stream brightness multiplier for depth
          this.depthBrightness = 0.75 + Math.random() * 0.35;
        }

        update() {
          this.y += this.speed;
          this.mutateTimer++;
          // Reset when whole trail clears bottom
          if (this.y - this.length > numRows) {
            this.init(false);
          }
        }
      }

      class Column {
        constructor(index) {
          this.index = index;
          // Character cache for consistency as rain falls
          this.chars = new Array(numRows + 40);
          for (let i = 0; i < this.chars.length; i++) {
            this.chars[i] = getRandomChar();
          }
          // Multiple drops per column to ensure dense, continuous flow
          this.drops = [];
          this.drops.push(new Drop(this.index, true));
          if (Math.random() > 0.4) {
            const secondDrop = new Drop(this.index, true);
            secondDrop.y -= (this.drops[0].length + Math.random() * 20);
            this.drops.push(secondDrop);
          }
        }

        mutate() {
          // Random glyph change in the column to simulate fluctuating digital data
          for (let m = 0; m < 3; m++) {
            const randRow = Math.floor(Math.random() * this.chars.length);
            this.chars[randRow] = getRandomChar();
          }
        }

        update() {
          this.mutate();
          for (let i = 0; i < this.drops.length; i++) {
            this.drops[i].update();
          }
        }
      }

      function setupDimensions() {
        const dpr = window.devicePixelRatio || 1;
        width = window.innerWidth;
        height = window.innerHeight;

        canvas.width = Math.floor(width * dpr);
        canvas.height = Math.floor(height * dpr);

        ctx.resetTransform();
        ctx.scale(dpr, dpr);

        // Optimal font scale across desktops and mobile
        fontSize = width < 768 ? 13 : 16;
        colWidth = fontSize * 0.95;
        numCols = Math.ceil(width / colWidth) + 1;
        numRows = Math.ceil(height / fontSize) + 5;

        // Initialize dense column field
        columns = [];
        for (let x = 0; x < numCols; x++) {
          columns.push(new Column(x));
        }

        ctx.font = `bold ${fontSize}px "Courier New", Courier, monospace`;
        ctx.textBaseline = 'top';
        ctx.textAlign = 'center';
      }

      window.addEventListener('resize', () => {
        setupDimensions();
      });

      // Pre-baked trail color lookup for maximum speed and contrast
      const TRAIL_STEPS = 40;
      const trailColors = [];
      for (let i = 0; i < TRAIL_STEPS; i++) {
        const progress = i / TRAIL_STEPS; // 0 = head, 1 = tail
        // Matrix phosphor decay profile: bright neon green -> vibrant green -> deep forest green -> black
        let r, g, b;
        if (progress < 0.05) {
          // Lead glowing white-green
          r = Math.floor(220 + (1 - progress * 20) * 35);
          g = 255;
          b = Math.floor(220 + (1 - progress * 20) * 35);
        } else if (progress < 0.25) {
          // High intensity neon green
          const p = (progress - 0.05) / 0.20;
          r = Math.floor(70 * (1 - p));
          g = Math.floor(255 - p * 35);
          b = Math.floor(100 * (1 - p));
        } else if (progress < 0.65) {
          // Classic Matrix phosphor green
          const p = (progress - 0.25) / 0.40;
          r = 0;
          g = Math.floor(220 - p * 110);
          b = Math.floor(15 * (1 - p));
        } else {
          // Deep fading tail to pitch black
          const p = (progress - 0.65) / 0.35;
          r = 0;
          g = Math.max(0, Math.floor(110 * (1 - p)));
          b = 0;
        }
        trailColors.push(`rgb(${r},${g},${b})`);
      }

      setupDimensions();

      function animate() {
        // Pure deep black background clearance
        ctx.fillStyle = '#000000';
        ctx.fillRect(0, 0, width, height);

        ctx.font = `bold ${fontSize}px "Courier New", Courier, monospace`;

        // Render pass for code rain
        for (let c = 0; c < columns.length; c++) {
          const col = columns[c];
          col.update();

          const posX = c * colWidth + colWidth / 2;

          for (let d = 0; d < col.drops.length; d++) {
            const drop = col.drops[d];
            const headRow = Math.floor(drop.y);
            const tailRow = Math.floor(drop.y - drop.length);

            const startRow = Math.max(0, tailRow);
            const endRow = Math.min(numRows, headRow);

            if (startRow > numRows || endRow < 0) continue;

            for (let r = startRow; r <= endRow; r++) {
              const posY = r * fontSize;
              const charIdx = (r + 1000) % col.chars.length;
              let char = col.chars[charIdx];

              const distFromHead = headRow - r;

              if (distFromHead === 0) {
                // Leading Glyph: Radiant White with Intense Phosphor Glow
                ctx.save();
                ctx.shadowColor = '#00ff66';
                ctx.shadowBlur = 12;
                ctx.fillStyle = '#ffffff';
                ctx.fillText(char, posX, posY);
                // Double strike for blinding core brilliance
                ctx.fillText(char, posX, posY);
                ctx.restore();
              } else {
                // Occasional data-sparkle: random body characters flash bright white/green
                const isSparkle = (r === (headRow - 4) && Math.random() < 0.08) || (Math.random() < 0.003);

                if (isSparkle) {
                  ctx.fillStyle = '#b3ffc8';
                  ctx.shadowColor = '#00ff41';
                  ctx.shadowBlur = 6;
                  ctx.fillText(getRandomChar(), posX, posY);
                  ctx.shadowBlur = 0;
                } else {
                  const norm = Math.min(1, Math.max(0, distFromHead / drop.length));
                  const colorIdx = Math.floor(norm * (TRAIL_STEPS - 1));
                  ctx.fillStyle = trailColors[colorIdx];
                  ctx.fillText(char, posX, posY);
                }
              }
            }
          }
        }

        requestAnimationFrame(animate);
      }

      requestAnimationFrame(animate);
    })();
  </script>
