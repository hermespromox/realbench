<script>
(() => {
  const DURATION = 18000;
  const svg = document.getElementById('scene');
  const path1 = document.getElementById('ballOnePath');
  const path2 = document.getElementById('ballTwoPath');
  const ball1 = document.getElementById('ball1');
  const ball2 = document.getElementById('ball2');
  const dominoes = [...document.querySelectorAll('#dominoes > g')];
  const lever = document.getElementById('lever');
  const gate = document.getElementById('gate');
  const cord = document.getElementById('cord');
  const wheel = document.getElementById('wheel');
  const linkRod = document.getElementById('linkRod');
  const crankPin = document.getElementById('crankPin');
  const launcher = document.getElementById('launcherArm');
  const projectile = document.getElementById('projectile');
  const bell = document.getElementById('bell');
  const ringLines = document.getElementById('ringLines');
  const sparks = document.getElementById('sparks');
  const resetCard = document.getElementById('resetCard');
  const flash = document.getElementById('flash');
  const dots = [...document.querySelectorAll('.indicator')];

  const len1 = path1.getTotalLength();
  const len2 = path2.getTotalLength();
  let start = performance.now();

  const clamp = (v, a=0, b=1) => Math.max(a, Math.min(b, v));
  const smooth = v => {
    v = clamp(v);
    return v * v * (3 - 2 * v);
  };
  const easeOut = v => 1 - Math.pow(1 - clamp(v), 3);
  const segment = (t, a, b) => clamp((t - a) / (b - a));
  const pointOn = (path, length, u) => {
    const d = length * clamp(u);
    const p = path.getPointAtLength(d);
    const p2 = path.getPointAtLength(Math.min(length, d + 2));
    return {x:p.x, y:p.y, angle:Math.atan2(p2.y-p.y,p2.x-p.x)*180/Math.PI};
  };

  function frame(now) {
    const ms = (now - start) % DURATION;
    const t = ms / 1000;

    // Ball 1 rolls down the wooden ramp.
    let b1u = smooth(segment(t, 0.8, 3.65));
    const p1 = pointOn(path1, len1, b1u);
    let b1Bounce = 0;
    if (t > 3.55 && t < 4.15) {
      const q = (t - 3.55) / .6;
      b1Bounce = -Math.sin(q * Math.PI) * 10 * (1-q);
    }
    ball1.setAttribute('transform',
      `translate(${p1.x} ${p1.y + b1Bounce}) rotate(${b1u * 920})`);
    ball1.style.opacity = t < 4.45 ? 1 : Math.max(0, 1-(t-4.45)*4);

    // Sequential domino toppling.
    dominoes.forEach((d, i) => {
      const x = +d.dataset.x + .5;
      const u = easeOut(segment(t, 3.55 + i*.18, 4.08 + i*.18));
      const angle = 78 * u;
      d.setAttribute('transform', `rotate(${angle} ${x} 445)`);
    });

    // Lever depresses, lifting its far end and opening the release gate.
    const leverU = smooth(segment(t, 5.0, 5.85));
    const leverAngle = -18 * leverU;
    lever.setAttribute('transform', `rotate(${leverAngle} 835 440)`);
    gate.setAttribute('transform', `rotate(${-72 * leverU} 929 243)`);
    const rightAngle = leverAngle * Math.PI/180;
    const rx = 835 + 95*Math.cos(rightAngle);
    const ry = 440 + 95*Math.sin(rightAngle);
    cord.setAttribute('d', `M${rx.toFixed(1)} ${ry.toFixed(1)} Q900 350 930 250`);

    // Ball 2 waits in the release box, then runs into the wheel.
    let b2u = smooth(segment(t, 5.95, 8.7));
    const p2 = pointOn(path2, len2, b2u);
    ball2.setAttribute('transform',
      `translate(${p2.x} ${p2.y}) rotate(${b2u * 790})`);
    ball2.style.opacity = t < 9.15 ? 1 : Math.max(0, 1-(t-9.15)*5);

    // Wheel rotation and crank linkage.
    const wheelU = smooth(segment(t, 8.55, 11.25));
    const wheelAngle = 630 * wheelU;
    wheel.setAttribute('transform', `rotate(${wheelAngle} 1015 507)`);
    const a = (wheelAngle - 90) * Math.PI / 180;
    const cx = 1015 + Math.cos(a) * 58;
    const cy = 507 + Math.sin(a) * 58;
    crankPin.setAttribute('cx', cx);
    crankPin.setAttribute('cy', cy);
    linkRod.setAttribute('d', `M${cx.toFixed(1)} ${cy.toFixed(1)}L1135 641`);

    // Catapult arm snaps upward.
    let launchAngle = 0;
    if (t >= 10.35 && t < 11.05) {
      launchAngle = -8 * Math.sin(segment(t,10.35,11.05)*Math.PI*4);
    } else if (t >= 11.05) {
      launchAngle = 84 * easeOut(segment(t,11.05,11.28));
    }
    launcher.setAttribute('transform', `rotate(${launchAngle} 1135 645)`);

    // Projectile follows a clear parabolic arc toward the bell.
    const projU = smooth(segment(t, 11.18, 13.45));
    let px, py, rot;
    if (t < 11.18) {
      px = 1096; py = 579; rot = -28;
      projectile.style.opacity = 1;
    } else if (t <= 13.62) {
      px = 1174 + (1212-1174)*projU;
      py = 594 + (344-594)*projU - 145 * Math.sin(Math.PI*projU);
      const du = Math.min(1, projU+.012);
      const nx = 1174 + (1212-1174)*du;
      const ny = 594 + (344-594)*du - 145 * Math.sin(Math.PI*du);
      rot = Math.atan2(ny-py,nx-px)*180/Math.PI;
      projectile.style.opacity = t < 13.52 ? 1 : Math.max(0, 1-(t-13.52)*7);
    } else {
      px = 1212; py = 344; rot = 0;
      projectile.style.opacity = 0;
    }
    projectile.setAttribute('transform', `translate(${px} ${py}) rotate(${rot})`);

    // Bell rings with a damped oscillation.
    let bellAngle = 0;
    if (t >= 13.36) {
      const bt = t - 13.36;
      bellAngle = Math.sin(bt * 19) * 13 * Math.exp(-bt*.72);
    }
    bell.setAttribute('transform', `rotate(${bellAngle} 1268 270)`);
    const ringOpacity = t >= 13.36 && t < 16.55
      ? Math.max(0, .95 * Math.exp(-(t-13.36)*.34) * (.65 + .35*Math.abs(Math.sin((t-13.36)*10))))
      : 0;
    ringLines.style.opacity = ringOpacity;
    const sparkU = segment(t,13.34,13.85);
    sparks.style.opacity = t >= 13.34 && t < 13.85 ? 1-sparkU : 0;
    sparks.setAttribute('transform', `translate(${-22*sparkU} 0) scale(${1+sparkU*.7})`);

    // Progress lights.
    const thresholds = [0.8, 3.55, 5.0, 5.95, 8.55, 11.05, 13.36];
    dots.forEach((d,i) => {
      const active = t >= thresholds[i];
      d.setAttribute('fill', active ? '#ffd459' : '#36505c');
      d.setAttribute('stroke', active ? '#fff1a3' : '#192b35');
      d.setAttribute('stroke-width', active ? '3' : '2');
      d.style.filter = active ? 'url(#glow)' : 'none';
    });

    // Gentle end card and transition before the automatic restart.
    let resetOpacity = 0;
    if (t > 16.35) resetOpacity = smooth(segment(t,16.35,16.85));
    if (t > 17.5) resetOpacity *= 1-smooth(segment(t,17.5,17.9));
    resetCard.setAttribute('opacity', resetOpacity);
    let flashOpacity = 0;
    if (t > 17.45) flashOpacity = .18 * smooth(segment(t,17.45,17.92));
    if (t < .25) flashOpacity = .18 * (1-smooth(segment(t,0,.25)));
    flash.setAttribute('opacity', flashOpacity);

    requestAnimationFrame(frame);
  }

  ringLines.style.opacity = 0;
  sparks.style.opacity = 0;
  requestAnimationFrame(frame);
})();
