```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">

    <style>
        * {

<body>
    <canvas id="canvas"></canvas>
    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        let width, height;

        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resize);
        resize();

        const TILT = 22 * Math.PI / 180;
        let elapsed = 0;
        let lastTime = performance.now();

        // Stars
        const stars = [];
        for (let i = 0; i < 700; i++) {
            const depth = Math.random();
            stars.push({
                x: Math.random(),
                y: Math.random(),
                size: (0.2 + depth * 1.6) * (Math.random() * 0.6 + 0.7),
                brightness: 0.15 + depth * 0.85,
                twinkleSpeed: Math.random() * 2 + 0.3,
                twinklePhase: Math.random() * Math.PI * 2,
                driftX: (Math.random() - 0.5) * 0.0008 * (0.4 + depth),
                driftY: (Math.random() - 0.5) * 0.0005 * (0.4 + depth)
            });
        }

        // Planets
        const planets = [{
            name: 'Mercury',
            distance: 55,
            radius: 3,
            speed: 0.8,
            phase: 0,
            color: '#8c7e6d',
            moons: []
        }, {
            name: 'Venus',
            distance: 80,
            radius: 5.5,
            speed: 0.6,
            phase: 1.2,
            color: '#e6c88c',
            moons: []
        }, {
            name: 'Earth',
            distance: 108,
            radius: 6,
            speed: 0.45,
            phase: 2.5,
            color: '#4a90d9',
            moons: [
                { distance: 13, radius: 1.8, speed: 3, phase: 0.5 }
            ]
        }, {
            name: 'Mars',
            distance: 138,
            radius: 4,
            speed: 0.35,
            phase: 4.0,
            color: '#c1440e',
            moons: [
                { distance: 9, radius: 0.9, speed: 3.5, phase: 1 },
                { distance: 13, radius: 0.6, speed: 2.5, phase: 3 }
            ]
        }, {
            name: 'Jupiter',
            distance: 185,
            radius: 16,
            speed: 0.2,
            phase: 1.5,
            color: '#c8a882',
            moons: [
                { distance: 22, radius: 2.2, speed: 2.5, phase: 0 },
                { distance: 28, radius: 1.8, speed: 1.8, phase: 2 },
                { distance: 34, radius: 2.8, speed: 1.4, phase: 4 },
                { distance: 40, radius: 1.4, speed: 1, phase: 5.5 }
            ]
        }, {
            name: 'Saturn',
            distance: 240,
            radius: 13,
            speed: 0.15,
            phase: 3.0,
            color: '#e8c87a',
            ring: true,
            moons: [
                { distance: 19, radius: 1.8, speed: 2, phase: 1 },
                { distance: 25, radius: 1.3, speed: 1.5, phase: 4 },
                { distance: 31, radius: 0.9, speed: 1, phase: 2.5 }
            ]
        }, {
            name: 'Uranus',
            distance: 290,
            radius: 9,
            speed: 0.1,
            phase: 0.8,
            color: '#7ec8e3',
            moons: [
                { distance: 14, radius: 1.3, speed: 2.2, phase: 3 },
                { distance: 18, radius: 0.9, speed: 1.6, phase: 0.5 }
            ]
        }, {
            name: 'Neptune',
            distance: 340,
            radius: 8,
            speed: 0.07,
            phase: 2.0,
            color: '#4a6cd4',
            moons: [
                { distance: 13, radius: 1.3, speed: 2.5, phase: 2 },
                { distance: 17, radius: 0.8, speed: 1.5, phase: 5 }
            ]
        }];

        function getScale() {
            return Math.min(width, height) / 900;
        }

        function project(x, y, z, scale, cx, cy) {
            const cosT = Math.cos(TILT);
            const sinT = Math.sin(TILT);
            const y1 = y * cosT - z * sinT;
            const z1 = y * sinT + z * cosT;
            return {
                x: cx + x * scale,
                y: cy + y1 * scale,
                depth: z1 * scale
            };
        }

        function lightenColor(hex, amount) {
            const r = parseInt(hex.slice(1, 3), 16);
            const g = parseInt(hex.slice(3, 5), 16);
            const b = parseInt(hex.slice(5, 7), 16);
            return `rgb(${Math.round(r + (255 - r) * amount)},${Math.round(g + (255 - g) * amount)},${Math.round(b + (255 - b) * amount)})`;
        }

        function darkenColor(hex, amount) {
            const r = parseInt(hex.slice(1, 3), 16);
            const g = parseInt(hex.slice(3, 5), 16);
            const b = parseInt(hex.slice(5, 7), 16);
            return `rgb(${Math.round(r * (1 - amount))},${Math.round(g * (1 - amount))},${Math.round(b * (1 - amount))})`;
        }

        function drawNebulae(cx, cy) {
            const neb1 = ctx.createRadialGradient(width * 0.2, height * 0.25, 0, width * 0.2, height * 0.25, width * 0.5);
            neb1.addColorStop(0, 'rgba(25, 45, 80, 0.12)');
            neb1.addColorStop(1, 'rgba(25, 45, 80, 0)');
            ctx.fillStyle = neb1;
            ctx.fillRect(0, 0, width, height);

            const neb2 = ctx.createRadialGradient(width * 0.8, height * 0.7, 0, width * 0.8, height * 0.7, width * 0.35);
            neb2.addColorStop(0, 'rgba(70, 25, 55, 0.08)');
            neb2.addColorStop(1, 'rgba(70, 25, 55, 0)');
            ctx.fillStyle = neb2;
            ctx.fillRect(0, 0, width, height);

            const neb3 = ctx.createRadialGradient(width * 0.5, height * 0.1, 0, width * 0.5, height * 0.1, width * 0.3);
            neb3.addColorStop(0, 'rgba(15, 30, 60, 0.06)');
            neb3.addColorStop(1, 'rgba(15, 30, 60, 0)');
            ctx.fillStyle = neb3;
            ctx.fillRect(0, 0, width, height);
        }

        function drawStars() {
            for (const star of stars) {
                const x = ((star.x + elapsed * star.driftX) % 1 + 1) % 1;
                const y = ((star.y + elapsed * star.driftY) % 1 + 1) % 1;
                const twinkle = 0.5 + 0.5 * Math.sin(elapsed * star.twinkleSpeed + star.twinklePhase);
                const alpha = star.brightness * (0.35 + 0.65 * twinkle);
                ctx.globalAlpha = Math.max(0, Math.min(1, alpha));
                ctx.fillStyle = '#ffffff';
                ctx.beginPath();
                ctx.arc(x * width, y * height, star.size, 0, Math.PI * 2);
                ctx.fill();
            }
            ctx.globalAlpha = 1;
        }

        function drawOrbits(scale, cx, cy) {
            const squash = Math.sin(TILT);
            for (const planet of planets) {
                ctx.beginPath();
                ctx.ellipse(cx, cy, planet.distance * scale, planet.distance * scale * squash, 0, 0, Math.PI * 2);
                ctx.strokeStyle = 'rgba(255,255,255,0.05)';
                ctx.lineWidth = 1;
                ctx.stroke();
            }
        }

        function drawSunOuterGlow(x, y, scale) {
            const r = 30 * scale;
            const pulse = 1 + Math.sin(elapsed * 0.4) * 0.03;
            const glowR = r * 7 * pulse;
            const glow = ctx.createRadialGradient(x, y, 0, x, y, glowR);
            glow.addColorStop(0, 'rgba(255,200,60,0.5)');
            glow.addColorStop(0.2, 'rgba(255,160,30,0.2)');
            glow.addColorStop(0.5, 'rgba(255,120,20,0.06)');
            glow.addColorStop(1, 'rgba(255,100,0,0)');
            ctx.fillStyle = glow;
            ctx.fillRect(x - glowR, y - glowR, glowR * 2, glowR * 2);
        }

        function drawSunCore(x, y, scale) {
            const r = 30 * scale;
            const pulse = 1 + Math.sin(elapsed * 0.6) * 0.04;

            // Corona layers
            for (let i = 1; i <= 3; i++) {
                const cr = r * (1.4 + i * 0.5) * pulse;
                const alpha = 0.12 / i;
                ctx.beginPath();
                ctx.arc(x, y, cr, 0, Math.PI * 2);
                ctx.strokeStyle = `rgba(255,200,100,${alpha})`;
                ctx.lineWidth = 1.5;
                ctx.stroke();
            }

            // Core gradient
            const core = ctx.createRadialGradient(x, y, 0, x, y, r * pulse);
            core.addColorStop(0, '#ffffff');
            core.addColorStop(0.3, '#fff8e0');
            core.addColorStop(0.6, '#ffd44a');
            core.addColorStop(1, '#ff8800');
            ctx.fillStyle = core;
            ctx.beginPath();
            ctx.arc(x, y, r * pulse, 0, Math.PI * 2);
            ctx.fill();

            // Surface details
            for (let i = 0; i < 6; i++) {
                const angle = elapsed * 0.3 + i * 1.05;
                const dist = r * 0.45 * Math.sin(elapsed * 0.5 + i * 1.7);
                const sx = x + Math.cos(angle) * dist;
                const sy = y + Math.sin(angle) * dist;
                const sr = r * 0.15 * (0.4 + 0.6 * Math.sin(elapsed * 0.8 + i * 2));
                const spot = ctx.createRadialGradient(sx, sy, 0, sx, sy, sr);
                spot.addColorStop(0, 'rgba(255,255,255,0.5)');
                spot.addColorStop(1, 'rgba(255,255,255,0)');
                ctx.fillStyle = spot;
                ctx.beginPath();
                ctx.arc(sx, sy, sr, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        function drawPlanetBody(x, y, radius, color, cx, cy) {
            if (radius <= 0.3) return;
            const dx = cx - x;
            const dy = cy - y;
            const dist = Math.sqrt(dx * dx + dy * dy) || 1;
            const offsetX = dx / dist * radius * 0.35;
            const offsetY = dy / dist * radius * 0.35;

            const grad = ctx.createRadialGradient(
                x + offsetX, y + offsetY, radius * 0.1,
                x, y, radius
            );
            grad.addColorStop(0, lightenColor(color, 0.45));
            grad.addColorStop(0.5, color);
            grad.addColorStop(1, darkenColor(color, 0.65));

            ctx.fillStyle = grad;
            ctx.beginPath();
            ctx.arc(x, y, radius, 0, Math.PI * 2);
            ctx.fill();

            // Subtle atmosphere glow
            const glow = ctx.createRadialGradient(x, y, radius * 0.8, x, y, radius * 1.8);
            glow.addColorStop(0, 'rgba(255,255,255,0.04)');
            glow.addColorStop(1, 'rgba(255,255,255,0)');
            ctx.fillStyle = glow;
            ctx.beginPath();
            ctx.arc(x, y, radius * 1.8, 0, Math.PI * 2);
            ctx.fill();
        }

        function drawSaturnRingHalf(x, y, planetRadius, isBack) {
            const squash = Math.sin(TILT);
            const startAngle = isBack ? Math.PI : 0;
            const endAngle = isBack ? 2 * Math.PI : Math.PI;

            ctx.save();
            ctx.translate(x, y);

            const rings = [
                { r: 1.35, width: 0.28, color: 'rgba(200,170,130,0.55)' },
                { r: 1.65, width: 0.22, color: 'rgba(220,190,150,0.65)' },
                { r: 1.9, width: 0.18, color: 'rgba(190,160,120,0.45)' },
                { r: 2.1, width: 0.1, color: 'rgba(230,200,160,0.35)' }
            ];

            for (const ring of rings) {
                ctx.beginPath();
                ctx.ellipse(0, 0, planetRadius * ring.r, planetRadius * ring.r * squash, 0, startAngle, endAngle);
                ctx.strokeStyle = ring.color;
                ctx.lineWidth = planetRadius * ring.width;
                ctx.stroke();
            }

            ctx.restore();
        }

        function drawVignette() {
            const grad = ctx.createRadialGradient(
                width / 2, height / 2, Math.min(width, height) * 0.25,
                width / 2, height / 2, Math.max(width, height) * 0.75
            );
            grad.addColorStop(0, 'rgba(0,0,0,0)');
            grad.addColorStop(1, 'rgba(0,0,0,0.55)');
            ctx.fillStyle = grad;
            ctx.fillRect(0, 0, width, height);
        }

        function animate(time) {
            const dt = Math.min((time - lastTime) / 1000, 0.05);
            lastTime = time;
            elapsed += dt;

            const scale = getScale();
            const cx = width / 2 + Math.sin(elapsed * 0.05) * width * 0.004;
            const cy = height / 2 + Math.cos(elapsed * 0.04) * height * 0.004;

            // Clear
            ctx.fillStyle = '#020210';
            ctx.fillRect(0, 0, width, height);

            // Background layers
            drawNebulae(cx, cy);
            drawStars();
            drawOrbits(scale, cx, cy);
            drawSunOuterGlow(cx, cy, scale);

            // Build object list
            const objects = [];
            objects.push({ type: 'sun', x: cx, y: cy, depth: 0, scale });

            for (const planet of planets) {
                const angle = planet.phase + elapsed * planet.speed;
                const px = Math.cos(angle) * planet.distance;
                const pz = Math.sin(angle) * planet.distance;
                const proj = project(px, 0, pz, scale, cx, cy);

                objects.push({
                    type: 'planet',
                    planet,
                    x: proj.x,
                    y: proj.y,
                    depth: proj.depth,
                    scale
                });

                for (const moon of planet.moons) {
                    const mAngle = moon.phase + elapsed * moon.speed;
                    const mx = px + Math.cos(mAngle) * moon.distance;
                    const mz = pz + Math.sin(mAngle) * moon.distance;
                    const mProj = project(mx, 0, mz, scale, cx, cy);

                    objects.push({
                        type: 'moon',
                        moon,
                        x: mProj.x,
                        y: mProj.y,
                        depth: mProj.depth,
                        scale
                    });
                }
            }

            // Sort by depth descending (furthest first)
            objects.sort((a, b) => b.depth - a.depth);

            // Draw objects
            for (const obj of objects) {
                if (obj.type === 'sun') {
                    drawSunCore(obj.x, obj.y, obj.scale);
                } else if (obj.type === 'planet') {
                    const r = obj.planet.radius * obj.scale;
                    if (obj.planet.ring) {
                        drawSaturnRingHalf(obj.x, obj.y, r, true);
                        drawPlanetBody(obj.x, obj.y, r, obj.planet.color, cx, cy);
                        drawSaturnRingHalf(obj.x, obj.y, r, false);
                    } else {
                        drawPlanetBody(obj.x, obj.y, r, obj.planet.color, cx, cy);
                    }
                } else if (obj.type === 'moon') {
                    const r = obj.moon.radius * obj.scale;
                    drawPlanetBody(obj.x, obj.y, r, obj.moon.color, cx, cy);
                }
            }

            // Vignette
            drawVignette();

            requestAnimationFrame(animate);
        }

        requestAnimationFrame(animate);
    </script>
</body>
```
