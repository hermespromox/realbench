function coronaLayer(R, k1,k2,k3, colPrefix, alpha){
  ctx.beginPath();
  const N=80;
  for(let i=0;i<=N;i++){
    const a = i/N*TAU;
    const rr = R*(1 + 0.05*Math.sin(a*3 + now*0.7 + k1)
                    + 0.038*Math.sin(a*7 - now*1.15 + k2)
                    + 0.024*Math.sin(a*11 + now*1.8 + k3));
    const px = cx+Math.cos(a)*rr, py = cy+Math.sin(a)*rr;
    i===0 ? ctx.moveTo(px,py) : ctx.lineTo(px,py);
  }
  ctx.closePath();
  const g = ctx.createRadialGradient(cx,cy,R*0.5, cx,cy,R*1.12);
  g.addColorStop(0, colPrefix+alpha+')');
  g.addColorStop(1, colPrefix+'0)');
  ctx.fillStyle = g; ctx.fill();
}

function planetBody(x,y,rs,aSun,p,z){
  ctx.save();
  ctx.translate(x,y);
  const lx = Math.cos(aSun)*rs*0.42, ly = Math.sin(aSun)*rs*0.42;
  // sun-lit sphere: highlight offset toward the sun
  let g = ctx.createRadialGradient(lx,ly,rs*0.06, 0,0, rs*1.02);
  g.addColorStop(0,p.lit); g.addColorStop(0.55,p.base); g.addColorStop(1,p.dark);
  ctx.beginPath(); ctx.arc(0,0,rs,0,TAU); ctx.fillStyle=g; ctx.fill();
  // surface detail, clipped to the disc
  ctx.save();
  ctx.beginPath(); ctx.arc(0,0,rs,0,TAU); ctx.clip();
  drawDetail(p, rs);
  ctx.restore();
  // terminator: shadow centred on the anti-sun limb
  const sx = -Math.cos(aSun)*rs*0.95, sy = -Math.sin(aSun)*rs*0.95;
  g = ctx.createRadialGradient(sx,sy,rs*0.15, sx,sy, rs*2.1);
  g.addColorStop(0,'rgba(3,5,14,0.88)');
  g.addColorStop(0.5,'rgba(3,5,14,0.30)');
  g.addColorStop(1,'rgba(3,5,14,0)');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(0,0,rs,0,TAU); ctx.fill();
  // rim light on the lit limb
  ctx.beginPath(); ctx.arc(0,0,rs*0.97, aSun-1.15, aSun+1.15);
  ctx.strokeStyle = p.rim || 'rgba(255,240,214,0.32)';
  ctx.lineWidth = Math.max(0.7, rs*0.07); ctx.stroke();
  // distance haze on the far side of the system
  if(z<0){
    ctx.fillStyle = `rgba(4,6,16,${(-z*0.26).toFixed(3)})`;
    ctx.beginPath(); ctx.arc(0,0,rs,0,TAU); ctx.fill();
  }
  ctx.restore();
}

      break;
    case 'neptune':
      ctx.fillStyle='rgba(230,240,255,0.1)'; ctx.fillRect(-rs,-0.1*rs, 2*rs, 0.16*rs);
      ell(dr(1.6,0.2), -0.25*rs, 0.20*rs, 0.12*rs, 0.15, 'rgba(18,28,84,0.65)');
      ell(dr(2.6,0.3), -0.42*rs, 0.32*rs, 0.05*rs, 0.10, 'rgba(240,248,255,0.5)');
      break;
  }
}

function saturnRings(rs, back){
  const defs=[
    [1.38, 0.16, 'rgba(190,170,132,0.5)'],
    [1.62, 0.24, 'rgba(226,206,164,0.78)'],
    [1.98, 0.15, 'rgba(196,176,138,0.45)'],
    [2.22, 0.07, 'rgba(230,214,180,0.3)']
  ];
  ctx.globalAlpha = back ? 0.62 : 1;
  for(const [rr,wd,col] of defs){
    ctx.beginPath();
    ctx.ellipse(0,0, rr*rs, rr*rs*0.42, 0, back?Math.PI:0, back?TAU:Math.PI);
    ctx.strokeStyle=col; ctx.lineWidth=Math.max(1, wd*rs); ctx.stroke();
  }
  ctx.globalAlpha = 1;
}

function uranusRings(rs, back){
  ctx.beginPath();
  ctx.ellipse(0,0, rs*1.9, rs*0.55, -1.25, back?Math.PI:0, back?TAU:Math.PI);
  ctx.strokeStyle = 'rgba(190,225,230,1)';
  ctx.globalAlpha = back ? 0.22 : 0.38;
  ctx.lineWidth = Math.max(0.8, rs*0.1); ctx.stroke();
  ctx.globalAlpha = 1;
}

function drawMoonBody(mx,my,ms,aSun,m){
  if(ms < 1.3){
    ctx.globalAlpha=0.9; ctx.fillStyle=m.base;
    ctx.beginPath(); ctx.arc(mx,my,ms,0,TAU); ctx.fill();
    ctx.globalAlpha=1; return;
  }
  const lx=Math.cos(aSun)*ms*0.4, ly=Math.sin(aSun)*ms*0.4;
  const g=ctx.createRadialGradient(mx+lx,my+ly,ms*0.1, mx,my,ms*1.02);
  g.addColorStop(0,m.lit); g.addColorStop(0.6,m.base); g.addColorStop(1,'#0d1016');
  ctx.beginPath(); ctx.arc(mx,my,ms,0,TAU); ctx.fillStyle=g; ctx.fill();
}

function drawHalo(x,y,rs,rgb,alpha,mul){
  ctx.save();
  ctx.globalCompositeOperation='screen';
  const g=ctx.createRadialGradient(x,y,rs*0.85, x,y,rs*mul);
  g.addColorStop(0,`rgba(${rgb},${alpha})`);
  g.addColorStop(0.4,`rgba(${rgb},${alpha*0.32})`);
  g.addColorStop(1,`rgba(${rgb},0)`);
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(x,y,rs*mul,0,TAU); ctx.fill();
  ctx.restore();
}

function renderPlanetSystem(p){
  const th = p.ph + now*p.w;
  const cth = Math.cos(th), sth = Math.sin(th);
  const x = cx + cth*p.orbit*S;
  const y = cy + sth*p.orbit*S*TILT;
  const z = sth;                                  // depth: +1 near, -1 far
  const rs = p.r*S*(1 + z*0.10);                  // perspective size
  const aSun = Math.atan2(cy-y, cx-x);            // direction to the sun

  // soft trail glowing along the orbit just behind the planet
  ctx.lineWidth = Math.max(0.8, S*0.9);
  ctx.strokeStyle = p.lit;
  for(let k=0;k<7;k++){
    ctx.beginPath();
    ctx.ellipse(cx,cy, p.orbit*S, p.orbit*S*TILT, 0, th-(k+1)*0.085, th-k*0.085);
    ctx.globalAlpha = 0.15*(1-k/7); ctx.stroke();
  }
  ctx.globalAlpha = 1;

  // moons, split by which side of the planet they're on
  const backM=[], frontM=[];
  for(const m of p.moons||[]){
    const mth = m.ph + now*m.w;
    const d = m.d*rs;
    const mx = x + Math.cos(mth)*d;
    const my = y + Math.sin(mth)*d*TILT;
    const ms = Math.max(0.7, m.s*S*(1 + z*0.08));
    const f = ()=> drawMoonBody(mx, my, ms, aSun, m);
    (Math.sin(mth) < 0 ? backM : frontM).push(f);
  }
  backM.forEach(f=>f());
  if(p.type==='saturn'){ ctx.save(); ctx.translate(x,y); saturnRings(rs,true); ctx.restore(); }
  if(p.type==='uranus'){ ctx.save(); ctx.translate(x,y); uranusRings(rs,true); ctx.restore(); }
  planetBody(x,y,rs,aSun,p,z);
  if(p.type==='saturn'){ ctx.save(); ctx.translate(x,y); saturnRings(rs,false); ctx.restore(); }
  if(p.type==='uranus'){ ctx.save(); ctx.translate(x,y); uranusRings(rs,false); ctx.restore(); }
  if(p.type==='earth') drawHalo(x,y,rs,'110,170,255',0.30,1.9);
  if(p.type==='venus') drawHalo(x,y,rs,'255,215,150',0.16,1.6);
  frontM.forEach(f=>f());
}

/* ---------------- belt, orbits, comet ---------------- */

function drawBelt(){
  ctx.fillStyle = '#9a9182';
  for(let pass=0; pass<2; pass++){
    ctx.globalAlpha = pass ? 0.5 : 0.26;
    ctx.beginPath();
    for(const b of belt){
      const a = b.th + now*b.w;
      const s2 = Math.sin(a);
      const far = s2 < 0;
      if(pass===0 ? !far : far) continue;   // two depth-sorted batches
      const x = cx + Math.cos(a)*b.a*S + b.jx;
      const y = cy + s2*b.a*S*TILT + b.jy;
      const sz = Math.max(0.55, b.s*S);
      ctx.moveTo(x+sz, y); ctx.arc(x,y,sz,0,TAU);
    }
    ctx.fill();
  }
  ctx.globalAlpha = 1;
}

function drawOrbits(){
  ctx.lineWidth = 1;
  for(const p of PLANETS){
    const rx = p.orbit*S, ry = p.orbit*S*TILT;
    ctx.strokeStyle='rgba(148,168,205,0.05)';
    ctx.beginPath(); ctx.ellipse(cx,cy,rx,ry,0,0,TAU); ctx.stroke();
    ctx.strokeStyle='rgba(168,190,228,0.085)';          // near half slightly brighter
    ctx.beginPath(); ctx.ellipse(cx,cy,rx,ry,0,0,Math.PI); ctx.stroke();
  }
  // dashed trace of the comet's eccentric path
  ctx.setLineDash([3,9]);
  ctx.strokeStyle='rgba(150,190,235,0.07)';
  ctx.beginPath();
  const c=comet, co=Math.cos(c.om), so=Math.sin(c.om);
  for(let i=0;i<=96;i++){
    const u=i/96*TAU;
    const r=c.A*(1-c.e*c.e)/(1+c.e*Math.cos(u));
    const ox=r*Math.cos(u), oy=r*Math.sin(u);
    const X=cx+(ox*co-oy*so)*S, Y=cy+(ox*so+oy*co)*S*TILT;
    i ? ctx.lineTo(X,Y) : ctx.moveTo(X,Y);
  }
  ctx.stroke();
  ctx.setLineDash([]);
}

function updateComet(dt){
  const c = comet;
  let r = c.A*(1-c.e*c.e)/(1+c.e*Math.cos(c.u));
  c.u += dt * c.K/(r*r);                       // angular momentum: fast at perihelion
  r = c.A*(1-c.e*c.e)/(1+c.e*Math.cos(c.u));
  const ox = r*Math.cos(c.u), oy = r*Math.sin(c.u);
  const co=Math.cos(c.om), so=Math.sin(c.om);
  const xr = ox*co - oy*so, yr = ox*so + oy*co;
  const X = cx + xr*S, Y = cy + yr*S*TILT;
  cometState = {x:X, y:Y, z:clamp(yr/500,-1,1), r, pxr:c.px, pyr:c.py};
  c.px = X; c.py = Y;
}

function tail(x,y,dx,dy,L,w1,col){
  const px=-dy, py=dx;
  const tx=x+dx*L, ty=y+dy*L;
  const w0=Math.max(1, L*0.035);
  const g=ctx.createLinearGradient(x,y,tx,ty);
  g.addColorStop(0,col);
  g.addColorStop(1,col.replace(/,[\s\d.]+\)$/,',0)'));
  ctx.fillStyle=g;
  ctx.beginPath();
  ctx.moveTo(x+px*w0, y+py*w0);
  ctx.lineTo(tx+px*w1, ty+py*w1);
  ctx.lineTo(tx-px*w1, ty-py*w1);
  ctx.lineTo(x-px*w0, y-py*w0);
  ctx.closePath(); ctx.fill();
}

function drawComet(){
  const st = cometState;
  const bright = clamp(150/st.r, 0.2, 1);
  let vx=-1, vy=0;
  if(st.pxr != null){
    vx = st.x - st.pxr; vy = st.y - st.pyr;
    const vl = Math.hypot(vx,vy)||1; vx/=vl; vy/=vl;
  }
  const dxs = cx-st.x, dys = cy-st.y;
  const dl = Math.hypot(dxs,dys)||1;
  const ax = dxs/dl, ay = dys/dl;              // anti-sunward
  const L = clamp(30000/st.r, 16, 175)*S;      // tail grows near perihelion

  ctx.save();
  ctx.globalCompositeOperation='screen';
  let ddx = ax - vx*0.42, ddy = ay - vy*0.42;  // dust tail lags behind motion
  const dl2 = Math.hypot(ddx,ddy)||1; ddx/=dl2; ddy/=dl2;
  tail(st.x, st.y, ddx, ddy, L*0.82, Math.max(2, L*0.13), `rgba(255,228,190,${(0.3*bright).toFixed(3)})`);
  tail(st.x, st.y, ax, ay, L*1.18, Math.max(1.4, L*0.07), `rgba(150,200,255,${(0.4*bright).toFixed(3)})`);
  // coma
  const cr = (1.4 + 850/st.r)*S*0.5;
  const g = ctx.createRadialGradient(st.x,st.y,0, st.x,st.y,cr*3.2);
  g.addColorStop(0, `rgba(238,247,255,${(0.9*bright).toFixed(3)})`);
  g.addColorStop(0.22, `rgba(190,220,255,${(0.38*bright).toFixed(3)})`);
  g.addColorStop(1, 'rgba(160,200,255,0)');
  ctx.fillStyle=g; ctx.beginPath(); ctx.arc(st.x,st.y,cr*3.2,0,TAU); ctx.fill();
  ctx.restore();
}

/* ---------------- main loop ---------------- */

function fmt(s){
  const m = (s/60)|0, ss = s - m*60;
  return 'T +' + String(m).padStart(2,'0') + ':' + ss.toFixed(1).padStart(4,'0');
}

function frame(ts){
  const dt = Math.min(0.05, (ts-last)/1000);
  last = ts; now += dt;

  psx += (ptx-psx)*Math.min(1, dt*2.5);
  psy += (pty-psy)*Math.min(1, dt*2.5);
  cx = CX + psx*10;
  cy = CY + psy*7;

  // deep space
  ctx.fillStyle = '#04060f';
  ctx.fillRect(0,0,W,H);
  const ag = ctx.createRadialGradient(cx,cy,0, cx,cy, Math.max(W,H)*0.62);
  ag.addColorStop(0,'rgba(34,44,78,0.20)');
  ag.addColorStop(1,'rgba(34,44,78,0)');
  ctx.fillStyle = ag; ctx.fillRect(0,0,W,H);
  ctx.globalAlpha = 0.9;
  ctx.drawImage(neb, 0,0, W,H);
  ctx.globalAlpha = 1;

  drawStars();
  drawOrbits();
  drawBelt();
  updateComet(dt);

  // depth-sorted render list: everything near/far around the sun (z = 0)
  const items = [{z:0, f:drawSun}];
  for(const p of PLANETS){
    items.push({z: Math.sin(p.ph + now*p.w), f: ()=> renderPlanetSystem(p)});
  }
  items.push({z: cometState.z, f: drawComet});
  items.sort((a,b)=> a.z - b.z);
  for(const it of items) it.f();

  clockEl.textContent = fmt(now);
  requestAnimationFrame(frame);
}

resize();
buildMottle();
requestAnimationFrame(frame);
