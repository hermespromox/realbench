<script>
(() => {

  const canvas = document.getElementById("space");

      ]

      moons: [

      ]
    },
    {

      size: .295, colors: ["#e0ffff", "#68cbd3", "#163c4b"], type: "ice",

    },
    {
      name: "NEPTUNE", orbit: .875, period: 69, phase: 5.55, node: .018,
      size: .275, colors: ["#a9d8ff", "#276bd1", "#07183f"], type: "neptune",
      moons: [{ distance: 1.78, size: .12, period: 3.9, phase: 4.4, color: "#afbac8" }]
    }

  function rnd(seed) {
    const x = Math.sin(seed * 91.731 + 17.17) * 43758.5453;
    return x - Math.floor(x);
  }

  function resize() {
    DPR = Math.min(window.devicePixelRatio || 1, 2);
    W = window.innerWidth;
    H = window.innerHeight;
    CX = W / 2;
    CY = H / 2;
    unit = Math.min(W, H);

    canvas.width = Math.round(W * DPR);
    canvas.height = Math.round(H * DPR);
    canvas.style.width = W + "px";
    canvas.style.height = H + "px";
    ctx.setTransform(DPR, 0, 0, DPR, 0, 0);

    background.width = Math.max(1, Math.round(W));
    background.height = Math.max(1, Math.round(H));
    createBackground();
    createStars();
  }

  function createBackground() {
    bg.setTransform(1, 0, 0, 1, 0, 0);
    bg.fillStyle = "#02030a";
    bg.fillRect(0, 0, W, H);

    const g0 = bg.createRadialGradient(CX, CY, 0, CX, CY, Math.max(W, H) * .72);
    g0.addColorStop(0, "#091022");
    g0.addColorStop(.35, "#050a17");
    g0.addColorStop(1, "#010208");
    bg.fillStyle = g0;
    bg.fillRect(0, 0, W, H);

    const clouds = [
      [W * .18, H * .27, Math.max(W,H) * .47, "rgba(36,52,110,.10)"],
      [W * .82, H * .64, Math.max(W,H) * .42, "rgba(78,28,89,.075)"],
      [W * .58, H * .06, Math.max(W,H) * .31, "rgba(20,80,105,.055)"],
      [W * .32, H * .92, Math.max(W,H) * .30, "rgba(77,37,80,.05)"]
    ];
    clouds.forEach(c => {
      const g = bg.createRadialGradient(c[0], c[1], 0, c[0], c[1], c[2]);
      g.addColorStop(0, c[3]);
      g.addColorStop(.42, c[3].replace(/[\d.]+\)$/, "0.025)"));
      g.addColorStop(1, "rgba(0,0,0,0)");
      bg.fillStyle = g;
      bg.fillRect(0, 0, W, H);
    });

    for (let i = 0; i < 1600; i++) {
      const x = rnd(i * 3.17) * W;
      const y = rnd(i * 8.73 + 2) * H;
      const a = rnd(i * 5.21) * .045;
      bg.fillStyle = `rgba(145,170,220,${a})`;
      bg.fillRect(x, y, 1, 1);
    }
  }

  function createStars() {
    const count = Math.min(700, Math.floor((W * H) / 2600));
    stars = [];
    for (let i = 0; i < count; i++) {
      const layer = rnd(i + 5);
      stars.push({
        x: rnd(i * 2.73 + 1),
        y: rnd(i * 7.19 + 4),
        r: .25 + Math.pow(rnd(i * 1.31 + 8), 5) * 2.15,
        a: .18 + rnd(i * 3.77) * .72,
        phase: rnd(i * 9.13) * TAU,
        speed: .25 + rnd(i * 4.8) * 1.2,
        layer,
        hue: rnd(i * 6.23)
      });
    }

    dust = [];
    for (let i = 0; i < 90; i++) {
      dust.push({
        angle: rnd(i * 4.2) * TAU,
        radius: (.12 + rnd(i * 6.4) * .48) * Math.max(W, H),
        speed: .004 + rnd(i) * .007,
        alpha: .02 + rnd(i * 2) * .05
      });
    }
  }

  function orbitPoint(p, t) {
    const maxOrbit = unit * .493;
    const r = maxOrbit * p.orbit;
    const a = p.phase + t * TAU / p.period;
    const flatten = .285;
    const ca = Math.cos(a), sa = Math.sin(a);
    const cn = Math.cos(p.node), sn = Math.sin(p.node);
    return {
      x: CX + r * (ca * cn - sa * flatten * sn),
      y: CY + r * (ca * sn + sa * flatten * cn),
      z: sa,
      a,
      r
    };
  }

  function drawStars(t) {
    ctx.drawImage(background, 0, 0, W, H);

    const driftX = Math.sin(t * .035) * 8;
    const driftY = Math.cos(t * .027) * 5;

    for (const s of stars) {
      let x = s.x * W + driftX * (.25 + s.layer);
      let y = s.y * H + driftY * (.25 + s.layer);
      x = (x + W) % W;
      y = (y + H) % H;
      const twinkle = .66 + .34 * Math.sin(t * s.speed + s.phase);
      const alpha = s.a * twinkle;
      const radius = s.r * (.7 + s.layer * .45);

      let color;
      if (s.hue < .14) color = `rgba(255,215,184,${alpha})`;
      else if (s.hue > .86) color = `rgba(174,210,255,${alpha})`;
      else color = `rgba(232,241,255,${alpha})`;

      if (radius > 1.25) {
        ctx.save();
        ctx.globalCompositeOperation = "screen";
        const glow = ctx.createRadialGradient(x, y, 0, x, y, radius * 4);
        glow.addColorStop(0, color);
        glow.addColorStop(.2, color);
        glow.addColorStop(1, "rgba(100,160,255,0)");
        ctx.fillStyle = glow;
        ctx.beginPath();
        ctx.arc(x, y, radius * 4, 0, TAU);
        ctx.fill();
        ctx.restore();
      } else {
        ctx.fillStyle = color;
        ctx.beginPath();
        ctx.arc(x, y, radius, 0, TAU);
        ctx.fill();
      }
    }

    ctx.save();
    ctx.translate(CX, CY);
    ctx.rotate(t * .001);
    for (const d of dust) {
      const a = d.angle + t * d.speed;
      const x = Math.cos(a) * d.radius;
      const y = Math.sin(a) * d.radius * .37;
      ctx.fillStyle = `rgba(125,151,206,${d.alpha})`;
      ctx.fillRect(x, y, 1, 1);
    }
    ctx.restore();
  }

  function drawOrbits(t) {
    const maxOrbit = unit * .493;
    ctx.save();
    ctx.lineWidth = 1;

    planets.forEach((p, index) => {
      const r = maxOrbit * p.orbit;
      const flatten = .285;
      ctx.save();
      ctx.translate(CX, CY);
      ctx.rotate(p.node);

      ctx.setLineDash(index < 4 ? [1.5, 5] : [2, 7]);
      ctx.lineDashOffset = -t * (2 + index * .18);
      ctx.strokeStyle = `rgba(112,145,190,${.105 - index * .006})`;
      ctx.beginPath();
      ctx.ellipse(0, 0, r, r * flatten, 0, 0, TAU);
      ctx.stroke();

      ctx.setLineDash([]);
      ctx.strokeStyle = "rgba(180,205,235,.025)";
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.ellipse(0, 0, r, r * flatten, 0, Math.PI, TAU);
      ctx.stroke();
      ctx.restore();
    });
    ctx.restore();
  }

  function drawSun(t) {
    const R = Math.max(23, Math.min(55, unit * .057));

    ctx.save();
    ctx.globalCompositeOperation = "screen";

    const halo = ctx.createRadialGradient(CX, CY, R * .15, CX, CY, R * 4.7);
    halo.addColorStop(0, "rgba(255,244,171,.94)");
    halo.addColorStop(.12, "rgba(255,173,45,.45)");
    halo.addColorStop(.35, "rgba(255,91,17,.13)");
    halo.addColorStop(.72, "rgba(255,56,5,.035)");
    halo.addColorStop(1, "rgba(255,40,0,0)");
    ctx.fillStyle = halo;
    ctx.beginPath();
    ctx.arc(CX, CY, R * 4.7, 0, TAU);
    ctx.fill();

    ctx.translate(CX, CY);
    ctx.rotate(t * .025);
    for (let i = 0; i < 26; i++) {
      const a = i / 26 * TAU;
      const pulse = .8 + .2 * Math.sin(t * 1.7 + i * 2.9);
      const length = R * (1.28 + rnd(i) * .72) * pulse;
      ctx.strokeStyle = `rgba(255,129,24,${.035 + rnd(i + 4) * .05})`;
      ctx.lineWidth = .5 + rnd(i + 9) * 1.2;
      ctx.beginPath();
      ctx.moveTo(Math.cos(a) * R * 1.03, Math.sin(a) * R * 1.03);
      ctx.lineTo(Math.cos(a) * length, Math.sin(a) * length);
      ctx.stroke();
    }
    ctx.restore();

    ctx.save();
    ctx.translate(CX, CY);
    ctx.rotate(-t * .055);
    ctx.strokeStyle = "rgba(255,149,42,.28)";
    ctx.lineWidth = Math.max(1, R * .035);
    for (let i = 0; i < 5; i++) {
      const a = i * 1.31 + Math.sin(t * .22 + i) * .14;
      ctx.beginPath();
      ctx.arc(0, 0, R * (1.08 + i * .028), a, a + .35 + rnd(i) * .38);
      ctx.stroke();
    }
    ctx.restore();

    const body = ctx.createRadialGradient(
      CX - R * .28, CY - R * .3, R * .05,
      CX, CY, R * 1.05
    );
    body.addColorStop(0, "#fffce0");
    body.addColorStop(.18, "#fff19a");
    body.addColorStop(.52, "#ffba35");
    body.addColorStop(.83, "#f2680d");
    body.addColorStop(1, "#8a2105");
    ctx.fillStyle = body;
    ctx.beginPath();
    ctx.arc(CX, CY, R, 0, TAU);
    ctx.fill();

    ctx.save();
    ctx.beginPath();
    ctx.arc(CX, CY, R * .96, 0, TAU);
    ctx.clip();
    ctx.globalCompositeOperation = "screen";
    for (let i = 0; i < 24; i++) {
      const a = rnd(i * 7.8) * TAU + t * (.025 + rnd(i) * .035);
      const rr = Math.sqrt(rnd(i * 2.1 + 4)) * R * .83;
      const x = CX + Math.cos(a) * rr;
      const y = CY + Math.sin(a) * rr;
      const sr = R * (.025 + rnd(i * 3.2) * .055);
      ctx.fillStyle = `rgba(255,244,167,${.08 + rnd(i) * .13})`;
      ctx.beginPath();
      ctx.arc(x, y, sr, 0, TAU);
      ctx.fill();
    }
    ctx.restore();

    ctx.strokeStyle = "rgba(255,238,167,.68)";
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.arc(CX, CY, R + .7, 0, TAU);
    ctx.stroke();
  }

  function ringGeometry(x, y, size, pale) {
    return {
      rx: size * (pale ? 1.75 : 2.0),
      ry: size * (pale ? .39 : .46),
      rot: pale ? -.55 : -.18
    };
  }

  function drawRingBack(x, y, size, pale) {
    const q = ringGeometry(x, y, size, pale);
    ctx.save();
    ctx.translate(x, y);
    ctx.rotate(q.rot);
    ctx.lineWidth = Math.max(1.2, size * .16);
    ctx.strokeStyle = pale ? "rgba(150,220,230,.22)" : "rgba(218,188,127,.38)";
    ctx.beginPath();
    ctx.ellipse(0, 0, q.rx, q.ry, 0, 0, TAU);
    ctx.stroke();

    ctx.lineWidth = Math.max(.5, size * .035);
    ctx.strokeStyle = pale ? "rgba(214,245,248,.32)" : "rgba(255,224,166,.58)";
    [-.12, .06, .17].forEach(v => {
      ctx.beginPath();
      ctx.ellipse(0, 0, q.rx * (1 + v), q.ry * (1 + v), 0, 0, TAU);
      ctx.stroke();
    });
    ctx.restore();
  }

  function drawRingFront(x, y, size, pale) {
    const q = ringGeometry(x, y, size, pale);
    ctx.save();
    ctx.translate(x, y);
    ctx.rotate(q.rot);
    ctx.lineCap = "round";
    ctx.lineWidth = Math.max(1, size * .075);
    ctx.strokeStyle = pale ? "rgba(205,244,246,.43)" : "rgba(250,220,158,.66)";
    ctx.beginPath();
    ctx.ellipse(0, 0, q.rx, q.ry, 0, 0, Math.PI);
    ctx.stroke();
    ctx.restore();
  }

  function drawPlanetBody(p, x, y, size, t) {
    const dx = CX - x, dy = CY - y;
    const length = Math.hypot(dx, dy) || 1;
    const lx = dx / length, ly = dy / length;

    ctx.save();
    ctx.beginPath();
    ctx.arc(x, y, size, 0, TAU);
    ctx.clip();

    const sphere = ctx.createRadialGradient(
      x + lx * size * .38, y + ly * size * .38, size * .05,
      x, y, size * 1.12
    );
    sphere.addColorStop(0, p.colors[0]);
    sphere.addColorStop(.35, p.colors[1]);
    sphere.addColorStop(.78, p.colors[1]);
    sphere.addColorStop(1, p.colors[2]);
    ctx.fillStyle = sphere;
    ctx.fillRect(x - size, y - size, size * 2, size * 2);

    if (p.type === "jupiter" || p.type === "saturn" || p.type === "cloud") {
      const bands = p.type === "jupiter" ? 9 : 6;
      for (let i = 0; i < bands; i++) {
        const yy = y - size + (i + .55) * size * 2 / bands;
        const sway = Math.sin(t * .18 + i * 2.1) * size * .035;
        ctx.strokeStyle = p.type === "jupiter"
          ? `rgba(${i % 2 ? "98,42,25" : "255,236,191"},${i % 2 ? .25 : .2})`
          : `rgba(${i % 2 ? "112,75,39" : "255,241,185"},.16)`;
        ctx.lineWidth = size * (.055 + (i % 3) * .02);
        ctx.beginPath();
        ctx.moveTo(x - size, yy);
        ctx.bezierCurveTo(x - size * .3, yy + sway, x + size * .3, yy - sway, x + size, yy);
        ctx.stroke();
      }
      if (p.type === "jupiter") {
        ctx.fillStyle = "rgba(143,53,29,.52)";
        ctx.beginPath();
        ctx.ellipse(x + size * .34, y + size * .34, size * .23, size * .105, -.08, 0, TAU);
        ctx.fill();
      }
    }

    if (p.type === "earth") {
      ctx.fillStyle = "rgba(65,145,82,.78)";
      const blobs = [
        [-.25,-.24,.24,.15], [.12,.03,.17,.29], [.38,-.26,.18,.10], [-.43,.25,.12,.18]
      ];
      blobs.forEach((b, i) => {
        ctx.beginPath();
        ctx.ellipse(x + b[0]*size, y + b[1]*size, b[2]*size, b[3]*size, i*.45, 0, TAU);
        ctx.fill();
      });
      ctx.strokeStyle = "rgba(240,255,255,.45)";
      ctx.lineWidth = size * .055;
      ctx.beginPath();
      ctx.arc(x, y, size * .78, 3.75, 5.05);
      ctx.stroke();
    }

    if (p.type === "mars" || p.type === "rock") {
      for (let i = 0; i < 5; i++) {
        const a = rnd(i * 8.4 + p.orbit) * TAU;
        const rr = rnd(i * 3.9) * size * .65;
        ctx.fillStyle = `rgba(45,24,20,${.12 + rnd(i) * .12})`;
        ctx.beginPath();
        ctx.arc(x + Math.cos(a) * rr, y + Math.sin(a) * rr, size * (.05 + rnd(i+2) * .12), 0, TAU);
        ctx.fill();
      }
    }

    if (p.type === "neptune" || p.type === "ice") {
      ctx.strokeStyle = "rgba(220,250,255,.13)";
      ctx.lineWidth = size * .08;
      ctx.beginPath();
      ctx.ellipse(x, y - size * .2, size, size * .22, 0, 0, TAU);
      ctx.stroke();
    }

    const shade = ctx.createLinearGradient(
      x + lx * size, y + ly * size,
      x - lx * size, y - ly * size
    );
    shade.addColorStop(0, "rgba(0,0,0,0)");
    shade.addColorStop(.52, "rgba(0,0,8,.05)");
    shade.addColorStop(.78, "rgba(0,0,10,.48)");
    shade.addColorStop(1, "rgba(0,0,10,.86)");
    ctx.fillStyle = shade;
    ctx.fillRect(x - size, y - size, size * 2, size * 2);
    ctx.restore();

    ctx.strokeStyle = "rgba(220,240,255,.19)";
    ctx.lineWidth = .7;
    ctx.beginPath();
    ctx.arc(x, y, size, Math.atan2(ly, lx) - 1.25, Math.atan2(ly, lx) + 1.25);
    ctx.stroke();
  }

  function drawMoon(m, planetX, planetY, planetSize, t, frontOnly) {
    const a = m.phase + t * TAU / m.period;
    const distance = planetSize * m.distance;
    const z = Math.sin(a);
    if ((z >= 0) !== frontOnly) return;
    const x = planetX + Math.cos(a) * distance;
    const y = planetY + Math.sin(a) * distance * .32;
    const r = Math.max(1, planetSize * m.size * (.9 + z * .08));

    const g = ctx.createRadialGradient(x - r*.3, y-r*.35, .1, x, y, r);
    g.addColorStop(0, "#fff");
    g.addColorStop(.28, m.color);
    g.addColorStop(1, "#343541");
    ctx.fillStyle = g;
    ctx.beginPath();
    ctx.arc(x, y, r, 0, TAU);
    ctx.fill();
  }

  function drawPlanetSystem(entry, t) {
    const p = entry.p;
    const pos = entry.pos;
    const sunR = Math.max(23, Math.min(55, unit * .057));
    const depthScale = .90 + pos.z * .10;
    const size = Math.max(2.2, sunR * p.size * depthScale);

    if (p.moons) {
      ctx.save();
      ctx.strokeStyle = "rgba(160,190,220,.12)";
      ctx.lineWidth = .6;
      p.moons.forEach(m => {
        ctx.beginPath();
        ctx.ellipse(pos.x, pos.y, size*m.distance, size*m.distance*.32, 0, 0, TAU);
        ctx.stroke();
      });
      ctx.restore();
      p.moons.forEach(m => drawMoon(m, pos.x, pos.y, size, t, false));
    }

    if (p.rings) drawRingBack(pos.x, pos.y, size, p.paleRing);
    drawPlanetBody(p, pos.x, pos.y, size, t);
    if (p.rings) drawRingFront(pos.x, pos.y, size, p.paleRing);
    if (p.moons) p.moons.forEach(m => drawMoon(m, pos.x, pos.y, size, t, true));

    if (unit > 430) {
      const labelAlpha = .28 + Math.max(0, pos.z) * .13;
      ctx.save();
      ctx.textAlign = "center";
      ctx.font = `${Math.max(7, Math.min(10, unit*.012))}px Arial, sans-serif`;
      ctx.fillStyle = `rgba(190,213,238,${labelAlpha})`;
      ctx.fillText(p.name, pos.x, pos.y + size + 13);
      ctx.restore();
    }
  }

  function drawComets(t) {
    const cycle = (t % 18);
    if (cycle < 3.1) {
      const q = cycle / 3.1;
      const x = W * (1.08 - q * .58);
      const y = H * (-.08 + q * .48);
      const len = 95 + unit * .08;
      ctx.save();
      ctx.globalCompositeOperation = "screen";
      const g = ctx.createLinearGradient(x, y, x + len, y - len * .55);
      g.addColorStop(0, `rgba(235,249,255,${Math.sin(q*Math.PI)*.8})`);
      g.addColorStop(.18, `rgba(116,191,255,${Math.sin(q*Math.PI)*.28})`);
      g.addColorStop(1, "rgba(70,130,255,0)");
      ctx.strokeStyle = g;
      ctx.lineWidth = 1.2;
      ctx.beginPath();
      ctx.moveTo(x, y);
      ctx.lineTo(x + len, y - len * .55);
      ctx.stroke();
      ctx.fillStyle = `rgba(245,253,255,${Math.sin(q*Math.PI)})`;
      ctx.beginPath();
      ctx.arc(x, y, 1.6, 0, TAU);
      ctx.fill();
      ctx.restore();
    }
  }

  function drawInterface(t) {
    if (W < 520 || H < 400) return;
    ctx.save();
    const margin = 27;
    ctx.font = "10px Arial, sans-serif";
    ctx.letterSpacing = "2px";
    ctx.fillStyle = "rgba(185,211,240,.42)";
    ctx.fillText("HELIOCENTRIC SYSTEM", margin, 31);
    ctx.fillStyle = "rgba(108,151,198,.25)";
    ctx.fillRect(margin, 39, 105, 1);
    ctx.fillRect(margin, 39, 22 + Math.sin(t*.4)*8, 1);

    ctx.textAlign = "right";
    ctx.fillStyle = "rgba(185,211,240,.28)";
    ctx.fillText("8 PLANETS  •  ORBITAL PLANE", W - margin, 31);

    ctx.textAlign = "left";
    ctx.fillStyle = "rgba(160,191,225,.2)";
    ctx.fillText("SOL  00", margin, H - 24);
    ctx.textAlign = "right";
    ctx.fillText("CONTINUOUS EPHEMERIS", W - margin, H - 24);
    ctx.restore();
  }

  const start = performance.now();

  function frame(now) {
    const t = (now - start) / 1000;
    ctx.setTransform(DPR, 0, 0, DPR, 0, 0);
    ctx.clearRect(0, 0, W, H);

    drawStars(t);
    drawOrbits(t);
    drawComets(t);

    const systems = planets.map(p => ({ p, pos: orbitPoint(p, t) }));
    const rear = systems.filter(s => s.pos.z < 0).sort((a,b) => a.pos.z - b.pos.z);
    const front = systems.filter(s => s.pos.z >= 0).sort((a,b) => a.pos.z - b.pos.z);

    rear.forEach(s => drawPlanetSystem(s, t));
    drawSun(t);
    front.forEach(s => drawPlanetSystem(s, t));
    drawInterface(t);

    requestAnimationFrame(frame);
  }

  window.addEventListener("resize", resize, { passive: true });
  resize();
  requestAnimationFrame(frame);
})();
