function frame(now){
  const dt=Math.min(0.05,(now-last)/1000);last=now;
  const t=(now-start)/1000;
  // subtle camera drift
  const driftX=Math.sin(t*0.08)*12, driftY=Math.cos(t*0.06)*8;
  const oCX=CX+driftX, oCY=CY+driftY;

  ctx.clearRect(0,0,W,H);
  nebula(t);
  // stars drift
  stars.forEach(s=>{
    s.x-=s.z*s.sp*dt/W*60*0.0006;
    if(s.x<0){s.x+=1;s.y=Math.random();}
    const x=s.x*W, y=s.y*H;
    const tw=0.45+0.55*Math.abs(Math.sin(t*(0.6+s.z*1.5)+s.tw));
    let col='255,255,255';
    if(s.col>0.85)col='180,210,255';else if(s.col<0.12)col='255,220,180';

    ctx.beginPath();ctx.arc(x,y,sz,0,Math.PI*2);ctx.fill();

      ctx.strokeStyle='rgba('+col+','+(tw*0.35).toFixed(3)+')';ctx.lineWidth=1;
      ctx.beginPath();ctx.moveTo(x-sz*3,y);ctx.lineTo(x+sz*3,y);ctx.moveTo(x,y-sz*3);ctx.lineTo(x,y+sz*3);ctx.stroke();
    }
  });
  // dust foreground bokeh
  dust.forEach(d=>{
    d.x-=d.sp*dt*0.01;if(d.x<0)d.x+=1;
    const x=d.x*W,y=d.y*H+Math.sin(t*0.4+d.ph)*10;
    ctx.fillStyle='rgba(150,180,255,'+(0.05+0.04*Math.sin(t+d.ph)).toFixed(3)+')';
    ctx.beginPath();ctx.arc(x,y,d.s*2.2,0,Math.PI*2);ctx.fill();
  });

  const sunR=54*S;
  // orbits
  planets.forEach((p,i)=>{
    const rx=p.orb*S*(1+0.015*Math.sin(t*0.05+i));
    const ry=rx*TILT;
    drawOrbit(rx,ry, p.orb<220?0.16:0.10);
  });
  // kuiper ring faint
  ctx.save();ctx.strokeStyle='rgba(150,180,255,0.06)';ctx.lineWidth=18*S;
  ctx.beginPath();ctx.ellipse(CX,CY,512*S,512*S*TILT,0,0,Math.PI*2);ctx.stroke();ctx.restore();
  kuiper.forEach(k=>{
    k.a+=k.sp*dt;
    const x=CX+Math.cos(k.a)*k.rr*S, y=CY+Math.sin(k.a)*k.rr*S*TILT;
    ctx.fillStyle='rgba(180,200,255,'+k.al+')';ctx.fillRect(x,y,k.sz,k.sz);
  });

  // compute planet positions
  const pos=planets.map(p=>{
    const rx=p.orb*S, ry=rx*TILT;
    const a=p.phase + t*Math.PI*2/p.period;
    const x=CX+Math.cos(a)*rx, y=CY+Math.sin(a)*ry;
    const depth=(Math.sin(a)+1)/2;
    const sc=0.78+0.44*depth;
    return {p:p,a:a,rx:rx,ry:ry,x:x,y:y,depth:depth,sc:sc,R:p.r*S*sc};
  });

  // back asteroids
  belt.forEach(b=>{
    b.a+=b.sp*dt;
    const s=Math.sin(b.a);
    if(s>0.05) return;
    const x=CX+Math.cos(b.a)*b.rr*S, y=CY+Math.sin(b.a)*b.rr*S*TILT;
    ctx.fillStyle='rgba(200,180,150,'+b.al+')';
    ctx.save();ctx.translate(x,y);ctx.rotate(b.a+b.wob);ctx.fillRect(-b.sz/2,-b.sz/3,b.sz,b.sz*0.66);ctx.restore();
  });
  // back planets
  pos.forEach(o=>{ if(o.depth<0.5) drawFullPlanet(o,t); });
  // sun
  const realSunR=drawSun(t,sunR);
  // anamorphic flare
  const fg=ctx.createLinearGradient(CX-realSunR*5,0,CX+realSunR*5,0);
  fg.addColorStop(0,'rgba(100,160,255,0)');fg.addColorStop(0.5,'rgba(255,210,140,0.18)');fg.addColorStop(1,'rgba(100,160,255,0)');
  ctx.fillStyle=fg;ctx.fillRect(CX-realSunR*5,CY-1,W*0+realSunR*10,2);
  const fg2=ctx.createRadialGradient(CX,CY,0,CX,CY,realSunR*2.2);
  fg2.addColorStop(0,'rgba(255,255,255,0.22)');fg2.addColorStop(1,'rgba(255,255,255,0)');
  ctx.fillStyle=fg2;ctx.beginPath();ctx.arc(CX,CY,realSunR*2.2,0,Math.PI*2);ctx.fill();

  // front asteroids
  belt.forEach(b=>{
    const s=Math.sin(b.a);
    if(s<=0.05) return;
    const x=CX+Math.cos(b.a)*b.rr*S, y=CY+Math.sin(b.a)*b.rr*S*TILT;
    const sc=0.8+0.5*s;
    ctx.fillStyle='rgba(220,200,170,'+(b.al+0.15)+')';
    ctx.save();ctx.translate(x,y);ctx.rotate(b.a);ctx.fillRect(-b.sz*sc/2,-b.sz*sc/3,b.sz*sc,b.sz*sc*0.66);ctx.restore();
  });
  // front planets
  pos.forEach(o=>{ if(o.depth>=0.5) drawFullPlanet(o,t); });

  // comet every ~22s
  const ct=(t%24)/24;
  if(ct<0.35){
    const ca= -0.6 + ct*2.2;
    const crx=520*S, cry=170*S;
    const comX=CX+Math.cos(ca)*crx, comY=CY+Math.sin(ca)*cry -20;
    const tailLen=90*S*(1-ct*1.2);
    const ta=Math.atan2(CY-comY,CX-comX);
    const tg=ctx.createLinearGradient(comX,comY,comX-Math.cos(ta)*tailLen* -1*-1,comY-Math.sin(ta)*tailLen);
    // tail away from sun
    const away=Math.atan2(comY-CY,comX-CX);
    const tx=comX+Math.cos(away)*tailLen, ty=comY+Math.sin(away)*tailLen*0.6;
    const tg2=ctx.createLinearGradient(comX,comY,tx,ty);
    tg2.addColorStop(0,'rgba(180,230,255,0.7)');tg2.addColorStop(1,'rgba(180,230,255,0)');
    ctx.strokeStyle=tg2;ctx.lineWidth=3*S;ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(comX,comY);ctx.lineTo(tx,ty);ctx.stroke();
    ctx.strokeStyle='rgba(180,230,255,0.25)';ctx.lineWidth=7*S;
    ctx.beginPath();ctx.moveTo(comX,comY);ctx.lineTo(tx,ty);ctx.stroke();
    ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(comX,comY,2.2*S+1,0,Math.PI*2);ctx.fill();
    ctx.fillStyle='rgba(160,220,255,0.5)';ctx.beginPath();ctx.arc(comX,comY,5*S,0,Math.PI*2);ctx.fill();
  }

  // labels + depth shadow under planets
  ctx.textAlign='left';
  pos.forEach(o=>{
    // soft shadow glow beneath
    ctx.fillStyle='rgba(0,0,0,0.35)';
    ctx.beginPath();ctx.ellipse(o.x,o.y+o.R*1.4,o.R*1.1,o.R*0.28,0,0,Math.PI*2);ctx.fill();
  });

  CX=keepCX;CY=keepCY;
  // clock
  const ss=Math.floor(t);const hh=String(Math.floor(ss/3600)).padStart(2,'0');const mm=String(Math.floor(ss/60)%60).padStart(2,'0');const sec=String(ss%60).padStart(2,'0');
  if((now|0)%5===0) clockEl.textContent='T+ '+hh+':'+mm+':'+sec;
  requestAnimationFrame(frame);

  function drawFullPlanet(o,t){
    const p=o.p,x=o.x,y=o.y,R=o.R;
    const dx=CX-x, dy=CY-y; const dl=Math.hypot(dx,dy)||1; const lx=dx/dl, ly=dy/dl;
    // moon orbit path
    if(p.moons){
      ctx.save();ctx.strokeStyle='rgba(180,200,255,0.14)';ctx.lineWidth=1;
      p.moons.forEach(m=>{
        ctx.beginPath();ctx.ellipse(x,y,m.d*S*o.sc,m.d*S*o.sc*TILT,0,0,Math.PI*2);ctx.stroke();
      });
      ctx.restore();
    }
    // Saturn back ring
    if(p.ring){
      ctx.save();ctx.translate(x,y);ctx.rotate(-0.32);
      ctx.strokeStyle='rgba(220,200,150,0.45)';ctx.lineWidth=R*0.32;
      ctx.beginPath();ctx.ellipse(0,0,R*1.9,R*0.55,0,0,Math.PI*2);ctx.stroke();
      ctx.strokeStyle='rgba(255,240,200,0.35)';ctx.lineWidth=R*0.12;
      ctx.beginPath();ctx.ellipse(0,0,R*1.65,R*0.48,0,0,Math.PI*2);ctx.stroke();
      ctx.strokeStyle='rgba(120,90,50,0.5)';ctx.lineWidth=R*0.08;
      ctx.beginPath();ctx.ellipse(0,0,R*2.15,R*0.62,0,0,Math.PI*2);ctx.stroke();
      ctx.restore();
    }
    if(p.ring2){
      ctx.save();ctx.translate(x,y);ctx.rotate(0.5);
      ctx.strokeStyle='rgba(180,240,255,0.28)';ctx.lineWidth=R*0.12;
      ctx.beginPath();ctx.ellipse(0,0,R*1.6,R*0.5,0,0,Math.PI*2);ctx.stroke();
      ctx.restore();
    }
    // planet body
    const atmo=p.type==='earth'?'rgba(80,160,255,0.35)':p.type==='venus'?'rgba(255,220,150,0.22)':p.type==='uranus'?'rgba(150,240,255,0.22)':p.type==='neptune'?'rgba(100,140,255,0.25)':null;
    drawPlanetDetail(p,x,y,R,t+ p.phase,lx,ly);
    if(atmo){
      const ag=ctx.createRadialGradient(x,y,R*0.9,x,y,R*1.45);
      ag.addColorStop(0,'rgba(0,0,0,0)');ag.addColorStop(0.75,atmo);ag.addColorStop(1,'rgba(0,0,0,0)');
      ctx.fillStyle=ag;ctx.beginPath();ctx.arc(x,y,R*1.45,0,Math.PI*2);ctx.fill();
    }
    // Saturn front ring arc
    if(p.ring){
      ctx.save();ctx.translate(x,y);ctx.rotate(-0.32);
      ctx.strokeStyle='rgba(255,244,210,0.85)';ctx.lineWidth=R*0.18;
      ctx.beginPath();ctx.ellipse(0,0,R*1.9,R*0.55,0,0.15,Math.PI-0.15);ctx.stroke();
      ctx.strokeStyle='rgba(255,255,255,0.25)';ctx.lineWidth=R*0.32;
      ctx.beginPath();ctx.ellipse(0,0,R*1.9,R*0.55,0,0.25,Math.PI-0.25);ctx.stroke();
      ctx.restore();
    }
    // moons
    if(p.moons){
      p.moons.forEach(m=>{
        const ma=m.ph + t*Math.PI*2/m.per;
        const mx=x+Math.cos(ma)*m.d*S*o.sc;
        const my=y+Math.sin(ma)*m.d*S*o.sc*TILT;
        const mr=m.r*S*(0.8+0.4*((Math.sin(ma)+1)/2));
        const mdx=CX-mx, mdy=CY-my; const mdl=Math.hypot(mdx,mdy)||1;
        shadeSphere(mx,my,mr,mdx/mdl,mdy/mdl,'#d8dbe2','#3a3d48',null);
        ctx.fillStyle='rgba(0,0,0,0.35)';ctx.beginPath();ctx.arc(mx-mr*0.2,my-mr*0.1,mr*0.25,0,Math.PI*2);ctx.fill();
      });
    }
    // label
    const labA=0.35+0.45*o.depth;
    ctx.fillStyle='rgba(210,225,255,'+labA.toFixed(3)+')';
    ctx.font='600 '+(9*S+3)+'px Segoe UI, Arial';
    ctx.save();ctx.globalAlpha=labA;ctx.letterSpacing='2px';
    ctx.fillText(p.name,x+R+7,y+4);
    ctx.restore();
    // traveling highlight on orbit
    // small glow dot at planet pos on orbit already planet itself
  }
}
requestAnimationFrame(frame);
})();
