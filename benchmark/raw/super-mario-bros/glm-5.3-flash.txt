function hitBlock(ty,tx){
  if(tx<0||tx>=COLS) return false;
  const t=tiles[ty*COLS+tx];
  if(t===T_Q){
    tiles[ty*COLS+tx]=T_USED;
    bumps.push({tx:tx,ty:ty,t:0});
    pops.push({x:tx*16+3,y:ty*16-14,vy:-4.8,t:0});
    sparkle(tx*16+8,ty*16-4);
    return true;

  return t!==0;
}

/* ============ mario AI + physics ============ */
function updateMario(){
  while(trigIdx<TRIG.length){
    const t=TRIG[trigIdx];
    if(m.x>t.x+46){ trigIdx++; continue; }
    if(grounded && m.x>=t.x){ trigIdx++; jump(t.v); }
    break;
  }
  if(grounded){
    const fx=m.x+11;
    if(!solidP(fx+6,m.y+18)) jump(-7.3);
    else{
      const fr=(m.y+14)>>4;
      if(solidP(fx+8,fr*16+8) || (fr>0 && solidP(fx+8,(fr-1)*16+8))) jump(-7.3);
      else for(const go of goombas){
        if(!go.dead && Math.abs(go.y-m.y)<26 && go.x>fx && go.x-fx<24){ jump(-7.3); break; }
      }
    }
  }
  /* horizontal */
  m.vx=Math.min(1.8,m.vx+0.09);
  m.x+=m.vx;
  if(m.vx>0){
    const fx=m.x+11;
    if(solidP(fx,m.y+2)||solidP(fx,m.y+8)||solidP(fx,m.y+13)){ m.x=((fx>>4)<<4)-11; m.vx=0; }
  }
  /* vertical */
  grounded=false;
  m.vy+=(m.vy<0)?0.34:0.5; if(m.vy>7.5)m.vy=7.5;
  m.y+=m.vy;
  if(m.vy>=0){
    const fy=m.y+15;
    if(solidP(m.x+2,fy)||solidP(m.x+9,fy)){
      const ty=fy>>4;
      if(m.vy>3.6) dust();
      m.y=(ty<<4)-15; m.vy=0; grounded=true;
    }
  } else {
    const hy=m.y;
    if(solidP(m.x+2,hy)||solidP(m.x+9,hy)){
      const ty=hy>>4;
      m.y=(ty+1)<<4; m.vy=0;
      hitBlock(ty,(m.x+2)>>4) || hitBlock(ty,(m.x+9)>>4);
    }
  }
  /* grab flag pole mid-leap */
  if(m.x+11>=2898){ state='flag'; m.x=2893; m.vx=0; m.vy=0; return; }
  /* enemies: any contact = bounce on top */
  for(const go of goombas){
    if(go.dead) continue;
    if(m.x<go.x+14 && m.x+11>go.x && m.y<go.y+12 && m.y+15>go.y){
      go.dead=true; go.sqT=0;
      m.vy=-5.6;
      if(m.y+15<go.y+10) m.y=go.y-15;
      poof(go.x+7,go.y+4);
    }
  }
  /* coins */
  for(const c of coins){
    if(!c.got && m.x<c.x+12 && m.x+11>c.x-2 && m.y<c.y+16 && m.y+15>c.y-2){
      c.got=true; sparkle(c.x+5,c.y+7);
    }
  }
  if(m.vx>0.2) runPhase+=m.vx*0.16;
  /* failsafes keep the run alive no matter what */
  if(m.x>bestX+2){ bestX=m.x; stuckT=0; } else stuckT++;
  if(stuckT===110 && grounded) jump(-7.3);
  if(stuckT>420 || m.y>270){ state='done'; doneT=0; if(fadeMode===0) fadeMode=1; }
}

/* ============ world update ============ */
function updateWorld(){
  for(const go of goombas){
    if(go.dead){ go.sqT++; continue; }
    go.x+=go.dir*0.42;
    if(go.x<go.a){ go.x=go.a; go.dir=1; }
    if(go.x>go.b-14){ go.x=go.b-14; go.dir=-1; }
  }
  for(const b of bumps) b.t++;
  bumps=bumps.filter(b=>b.t<=12);
  for(const p of pops){ p.t++; p.vy+=0.42; p.y+=p.vy; }
  pops=pops.filter(p=>p.t<=28);
  for(const p of parts){ p.t++; p.x+=p.vx; p.y+=p.vy; p.vy+=0.1; }
  parts=parts.filter(p=>p.t<p.life);
  for(const s of sparks) s.t++;
  sparks=sparks.filter(s=>s.t<=16);
  if(state==='flag'||state==='flagwait') flagY=Math.min(168,flagY+2.4);
}

function step(){
  gt++;
  if(fadeMode===1){ fadeA+=0.02; if(fadeA>=1){ fadeA=1; reset(); fadeMode=2; } return; }
  if(fadeMode===2){ fadeA-=0.02; if(fadeA<=0){ fadeA=0; fadeMode=0; } }
  if(state==='run') updateMario();
  else if(state==='flag'){ m.y+=2.4; if(m.y>=177){ m.y=177; state='flagwait'; waitT=0; } }
  else if(state==='flagwait'){ if(++waitT>26){ state='walk'; m.vx=0.9; m.vy=-4.4; } }
  else if(state==='walk'){
    m.vy+=0.5; if(m.vy>7.5)m.vy=7.5;
    m.y+=m.vy; m.x+=m.vx;
    if(m.y+15>=208){ m.y=193; m.vy=0; m.vx=1.3; }
    runPhase+=m.vx*0.16;
    if(m.x>=DOOR_X){ state='done'; doneT=0; hidden=true; }
  }
  else if(state==='done'){ if(++doneT===80 && fadeMode===0) fadeMode=1; }
  updateWorld();
  const target=Math.max(0,Math.min(LW-VIEW_W,m.x+6-VIEW_W*0.35));
  camX+=(target-camX)*0.14;
  if(camX<0)camX=0;
}

  const pulse=(gt>>4)&1;
  const c0=(cx>>4)-1, c1=c0+Math.ceil(VIEW_W/16)+3;
  for(let ty=0;ty<ROWS;ty++){
    for(let tx=c0;tx<c1;tx++){
      if(tx<0||tx>=COLS) continue;
      const t=tiles[ty*COLS+tx];
      if(!t||t===T_PIPE) continue;
      let dy=0;
      for(const b of bumps) if(b.tx===tx&&b.ty===ty){ dy=-Math.round(Math.sin(Math.PI*b.t/12)*5); break; }
      const cvv = t===T_GROUND?groundCv : t===T_BRICK?brickCv : t===T_Q?(pulse?QA:QB) : t===T_USED?usedCv : solidCv;
      g.drawImage(cvv,tx*16-cx,ty*16+dy);
    }
  }
  /* pipes */
  for(const p of PIPES){
    const sx=p.tx*16-cx;
    if(sx<-40||sx>VIEW_W+40) continue;
    const top=(13-p.h)*16;
    g.fillStyle='#000'; g.fillRect(sx+2,top+14,28,p.h*16-14);
    g.fillStyle='#00a800'; g.fillRect(sx+3,top+15,26,p.h*16-16);
    g.fillStyle='#80d010'; g.fillRect(sx+5,top+15,5,p.h*16-16);
    g.fillStyle='#0a6e0a'; g.fillRect(sx+24,top+15,4,p.h*16-16);
    g.fillStyle='#000'; g.fillRect(sx,top,32,16);
    g.fillStyle='#00a800'; g.fillRect(sx+1,top+1,30,14);
    g.fillStyle='#80d010'; g.fillRect(sx+3,top+1,6,14);
    g.fillStyle='#0a6e0a'; g.fillRect(sx+26,top+1,4,14);
  }
  /* floating coins */
  coins.forEach((c,i)=>{
    if(c.got) return;
    const sx=c.x-cx;
    if(sx<-16||sx>VIEW_W+16) return;
    const bob=Math.round(Math.sin(gt*0.06+i*1.13)*2.5);
    drawCoinS(sx,c.y+bob,(Math.floor(gt/7)+i)%4);
  });
  /* coins popping out of blocks */
  for(const p of pops) drawCoinS(Math.round(p.x)-cx,Math.round(p.y),(p.t>>2)%4);
  /* goombas */
  for(const go of goombas){
    if(go.dead&&go.sqT>30) continue;
    const sx=Math.round(go.x)-1-cx;
    if(sx<-20||sx>VIEW_W+20) continue;
    const cvs=go.dead?GSQ:((Math.floor(gt/10)%2)?G1:G2);
    g.drawImage(cvs,sx,Math.round(go.dead?go.y-4:go.y-2));
  }
  /* mario */
  if(!hidden){
    let mc;
    if(state==='flag'||state==='flagwait') mc=MJ;
    else if(m.y+15<206) mc=MJ;
    else if(m.vx>0.2) mc=RUNF[Math.floor(runPhase)%4];
    else mc=MS;
    g.drawImage(mc,Math.round(m.x)-1-cx,Math.round(m.y)-1);
  }
  /* particles + sparkles */
  for(const p of parts){ g.fillStyle=p.c; g.fillRect(Math.round(p.x)-cx,Math.round(p.y),2,2); }
  for(const s of sparks){
    const L=s.t<6?2:4, x=Math.round(s.x)-cx, y=Math.round(s.y);
    g.fillStyle='#fff';
    g.fillRect(x,y-L,1,2*L+1);
    g.fillRect(x-L,y,2*L+1,1);
  }
  if(fadeA>0){ g.fillStyle='rgba(0,0,0,'+fadeA.toFixed(3)+')'; g.fillRect(0,0,VIEW_W,VIEW_H); }
}

  requestAnimationFrame(frame);
  if(!last) last=ts;
  acc+=Math.min(200,ts-last); last=ts;
  let n=0;
  while(acc>=1000/60 && n<4){ step(); acc-=1000/60; n++; }
  render();
}
