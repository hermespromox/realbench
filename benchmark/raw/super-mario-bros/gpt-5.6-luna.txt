<script>
const c=document.getElementById('c'),x=c.getContext('2d');

const W=320,H=180, WORLD=2450;
let t=0,cam=0,last=0,run=0;
const ground=[
 [0,390,150],[450,790,150],[835,1160,150],[1210,1530,150],
 [1580,1880,150],[1935,2260,150],[2310,2450,150]
];

 [115,132,48,7],[185,110,58,7],[285,125,55,7],[500,116,78,7],
 [650,91,70,7],[875,119,63,7],[1000,99,70,7],[1135,126,52,7],
 [1260,108,72,7],[1390,82,70,7],[1635,117,72,7],[1760,91,68,7],
 [2010,112,75,7],[2140,86,68,7],[2350,116,70,7]
];

 [305,132,18,18],[545,130,20,20],[1045,132,20,18],[1288,130,20,20],
 [1700,130,22,20],[2055,130,20,20],[2390,130,20,20]
];
const blocks=[
 [150,105,'q'],[167,105,'b'],[184,105,'q'],[312,82,'q'],[329,82,'b'],
 [515,89,'q'],[532,89,'b'],[549,89,'q'],[680,63,'q'],[697,63,'b'],
 [1015,72,'q'],[1032,72,'q'],[1320,81,'q'],[1337,81,'b'],[1410,55,'q'],
 [1660,90,'q'],[1775,64,'q'],[2030,85,'q'],[2155,59,'b'],[2172,59,'q']
];
const coins=[];
for(let i=0;i<blocks.length;i++) if(blocks[i][2]=='q') coins.push({x:blocks[i][0]+4,y:blocks[i][1]-15,got:false,phase:i});
[[75,123],[96,113],[205,91],[266,111],[476,103],[610,103],[730,73],
[865,101],[952,101],[1070,91],[1150,108],[1245,91],[1370,63],[1460,63],
[1600,101],[1810,70],[1970,115],[2075,96],[2220,74],[2330,100]].forEach((p,i)=>coins.push({x:p[0],y:p[1],got:false,phase:i+20}));
let enemies=[
 {x:235,y:136,v:.35,alive:true},{x:370,y:136,v:.3,alive:true},{x:575,y:136,v:.32,alive:true},
 {x:715,y:136,v:-.3,alive:true},{x:920,y:105,v:.25,alive:true,ledge:[875,938]},
 {x:1100,y:136,v:-.32,alive:true},{x:1450,y:136,v:.3,alive:true},{x:1680,y:103,v:.26,alive:true,ledge:[1635,1707]},
 {x:1840,y:136,v:-.3,alive:true},{x:1990,y:136,v:.3,alive:true},{x:2200,y:136,v:-.3,alive:true}
];
let p={x:24,y:120,vx:1.02,vy:0,w:10,h:14,on:false,frame:0,inv:0,face:1};
let bumps=[],dust=[];
function reset(){
 p={x:24,y:120,vx:1.02,vy:0,w:10,h:14,on:false,frame:0,inv:0,face:1};
 enemies.forEach(e=>e.alive=true); coins.forEach(q=>q.got=false); bumps=[];dust=[];cam=0;
}
function rectHit(a,b){return a.x<b.x+b.w&&a.x+a.w>b.x&&a.y<b.y+b.h&&a.y+a.h>b.y}
function surfaces(){
 return ground.map(q=>({x:q[0],y:q[2],w:q[1]-q[0],h:30})).concat(
 plats.map(q=>({x:q[0],y:q[1],w:q[2],h:q[3]}))
 );
}
function jump(){if(p.on){p.vy=-4.25;p.on=false;for(let i=0;i<3;i++)dust.push({x:p.x+4,y:p.y+13,vx:(Math.random()-.5)*.5,life:12})}}
function update(dt){
 t+=dt; run+=dt;
 if(p.inv>0)p.inv-=dt;
 p.vx=1.02;
 let danger=false;
 for(const q of pipes)if(q[0]>p.x-3&&q[0]-p.x<30&&p.y>115)danger=true;
 for(const s of surfaces())if(s.x>p.x-5&&s.x-p.x<28&&s.y>=130&&s.y<151)danger=true;
 if(Math.floor(p.x)%173<2 || danger && p.on || (p.on&&Math.sin(p.x*.07)>0.96))jump();
 p.x+=p.vx;
 p.vy+=.22;
 let oldy=p.y;p.y+=p.vy;p.on=false;
 for(const s of surfaces()){
   if(p.x+p.w>s.x&&p.x<s.x+s.w&&oldy+p.h<=s.y&&p.y+p.h>=s.y&&p.vy>=0){
     p.y=s.y-p.h;p.vy=0;p.on=true;
   }
 }
 for(const b of blocks){
   let z={x:b[0],y:b[1],w:16,h:8};
   if(p.x+p.w>z.x&&p.x<z.x+z.w&&p.y<z.y+z.h&&p.y+p.h>z.y&&p.vy<0){
     p.y=z.y+z.h;p.vy=.65;bumps.push({x:b[0],y:b[1],life:10});
     if(b[2]=='q') coins.forEach(q=>{if(Math.abs(q.x-(b[0]+4))<3&&Math.abs(q.y-(b[1]-15))<3&&!q.got){q.got=true}});
   }
 }
 for(const e of enemies){
   if(!e.alive)continue;
   e.x+=e.v;
   if(e.ledge){if(e.x<e.ledge[0]||e.x>e.ledge[1]-10)e.v*=-1}
   else if(e.x<0||e.x>WORLD)e.v*=-1;
   let z={x:e.x,y:e.y,w:10,h:10};
   if(rectHit(p,z)){
     if(p.vy>0&&p.y+p.h-3<e.y+5){e.alive=false;p.vy=-3.1;dust.push({x:e.x,y:e.y+8,vx:0,life:14})}
     else if(p.inv<=0){p.vy=-3.3;p.inv=80;p.x-=10}
   }
 }
 coins.forEach(q=>{if(!q.got&&Math.abs(p.x+5-q.x)<8&&Math.abs(p.y+6-q.y)<11)q.got=true});
 bumps=bumps.filter(b=>(b.y--,b.life-- >0));
 dust.forEach(d=>{d.x+=d.vx;d.y+=.16;d.life--});dust=dust.filter(d=>d.life>0);
 if(p.y>190){p.x+=45;p.y=100;p.vy=0}
 p.frame+=dt*12;
 cam+=(Math.max(0,p.x-105)-cam)*.09;
 if(cam>WORLD-W)cam=WORLD-W;
 if(p.x>WORLD-45){reset()}
}

 x.fillStyle='#5c94fc';x.fillRect(0,0,W,H);
 x.fillStyle='#76adff';
 for(let i=0;i<7;i++){let sx=((i*71-cam*.18)%380+380)%380-35,sy=22+(i%3)*18;
  x.fillRect(sx,sy,25,5);x.fillRect(sx+6,sy-4,14,9);x.fillRect(sx+13,sy-7,9,12);x.fillRect(sx+23,sy+2,9,3)}
 x.fillStyle='#87d05b';
 for(let i=-1;i<9;i++){let hx=i*62-(cam*.24%62);x.fillRect(hx,127,63,30);x.fillRect(hx+10,121,43,10);x.fillRect(hx+19,116,24,9)}
 x.fillStyle='#45aa4a';
 for(let i=-1;i<14;i++){let bx=i*32-(cam*.43%32);x.fillRect(bx,140,29,16);x.fillRect(bx+5,135,18,12);x.fillRect(bx+10,131,9,9)}
}
function block(b){
 let y=b[1], xx=b[0];
 x.fillStyle=b[2]=='q'?'#f6a91a':'#b95728';x.fillRect(xx,y,16,8);
 x.fillStyle=b[2]=='q'?'#ffd957':'#e6813c';x.fillRect(xx+1,y+1,14,1);x.fillRect(xx+1,y+6,14,1);

}
