function renderSprite(spriteArr) {
  const c = document.createElement('canvas');
  c.width = spriteArr[0].length;
  c.height = spriteArr.length;
  const cx = c.getContext('2d');
  for (let y = 0; y < spriteArr.length; y++) {
    for (let x = 0; x < spriteArr[y].length; x++) {
      const char = spriteArr[y][x];
      if (PALETTE[char]) {
        cx.fillStyle = PALETTE[char];
        cx.fillRect(x, y, 1, 1);
      }
    }
  }
  return c;
}

function isColliding(a, b) {
  return a.x < b.x + b.w && a.x + a.w > b.x && a.y < b.y + b.h && a.y + a.h > b.y;
}

function spawnCoin(x, y) {
  coins.push({ x: x, y: y, collected: false, frame: 0, vy: -4, life: 30 });
}

function update() {
  if (fadingOut) {
    fadeAlpha += 0.05;
    if (fadeAlpha >= 1) {
      resetLevel();
      fadeAlpha = 1;
      fadingOut = false;
    }
    return;
  }
  
  if (fadeAlpha > 0) {
    fadeAlpha -= 0.05;
    if (fadeAlpha < 0) fadeAlpha = 0;
  }

  globalFrame++;

  for (let obj of level) {
    if (obj.bumpTimer > 0) {
      obj.bumpTimer--;
      obj.drawY = obj.y - Math.sin(obj.bumpTimer / 10 * Math.PI) * 4;
    } else {
      obj.drawY = obj.y;
    }
  }

  let lookaheadX = mario.x + mario.w + 15;
  let groundAhead = false;
  let obstacleAhead = false;

  for (let obj of level) {
    if (lookaheadX >= obj.x && lookaheadX <= obj.x + obj.w) {
      if (obj.y >= mario.y + mario.h - 1 && obj.y <= mario.y + mario.h + 20) {
        groundAhead = true;
      }
    }
    if (obj.y < mario.y + mario.h - 1 && obj.y + obj.h > mario.y) {
      if (lookaheadX >= obj.x && lookaheadX <= obj.x + obj.w) {
        obstacleAhead = true;
      }
    }
  }

  if (mario.grounded && (!groundAhead || obstacleAhead)) {
    mario.vy = mario.jumpForce;
    mario.grounded = false;
  }

  for (let e of enemies) {
    if (!e.dead) {
      let dist = e.x - mario.x;
      if (dist > 65 && dist < 75 && mario.grounded) {
        mario.vy = mario.jumpForce;
        mario.grounded = false;
      }
    }
  }

  if (mario.grounded && Math.random() < 0.02) {
    for (let obj of level) {
      if ((obj.type === 'qBlock' && !obj.used) || obj.type === 'brick') {
        if (Math.abs(obj.x - mario.x) < 24 && obj.y < mario.y && obj.y > mario.y - 60) {
          mario.vy = mario.jumpForce;
          mario.grounded = false;
          break;
        }
      }
    }
  }

  mario.x += mario.vx;
  for (let obj of level) {
    if (isColliding(mario, obj)) {
      if (mario.vx > 0) mario.x = obj.x - mario.w;
      else if (mario.vx < 0) mario.x = obj.x + obj.w;
    }
  }

  mario.vy += mario.gravity;
  mario.y += mario.vy;
  mario.grounded = false;

  for (let obj of level) {
    if (isColliding(mario, obj)) {
      if (mario.vy > 0) {
        mario.y = obj.y - mario.h;
        mario.vy = 0;
        mario.grounded = true;
      } else if (mario.vy < 0) {
        mario.y = obj.y + obj.h;
        mario.vy = 0;
        if (obj.type === 'qBlock' && !obj.used) {
          obj.used = true;
          obj.bumpTimer = 10;
          spawnCoin(obj.x, obj.y - 16);
        } else if (obj.type === 'brick') {
          obj.bumpTimer = 10;
        }
      }
    }
  }

  for (let i = enemies.length - 1; i >= 0; i--) {
    let e = enemies[i];
    if (e.dead) {
      e.deadTimer--;
      if (e.deadTimer <= 0) enemies.splice(i, 1);
      continue;
    }
    
    e.x += e.vx;
    e.vy = (e.vy || 0) + mario.gravity;
    e.y += e.vy;
    e.grounded = false;
    
    for (let obj of level) {
      if (isColliding({x: e.x, y: e.y, w: 16, h: 16}, obj)) {
        if (e.vy > 0) {
          e.y = obj.y - 16;
          e.vy = 0;
          e.grounded = true;
        }
      }
    }
    
    let hitWall = false;
    for (let obj of level) {
      if (isColliding({x: e.x, y: e.y + 2, w: 16, h: 14}, obj)) {
         hitWall = true; break;
      }
    }
    if (hitWall) {
      e.vx *= -1;
      e.x += e.vx;
    }
    
    if (e.grounded) {
      let edgeAhead = true;
      let checkX = e.vx > 0 ? e.x + 16 : e.x - 1;
      for (let obj of level) {
        if (checkX >= obj.x && checkX <= obj.x + obj.w) {
          if (obj.y >= e.y + 14 && obj.y <= e.y + 26) {
            edgeAhead = false;
          }
        }
      }
      if (edgeAhead) e.vx *= -1;
    }
    
    if (isColliding(mario, {x: e.x, y: e.y, w: 16, h: 16})) {
      if (mario.vy > 0 && mario.y + mario.h - mario.vy <= e.y + 8) {
        e.dead = true;
        e.deadTimer = 30;
        mario.vy = -4;
      }
    }
  }

  for (let i = coins.length - 1; i >= 0; i--) {
    let c = coins[i];
    if (c.life !== undefined) {
      c.vy += mario.gravity;
      c.y += c.vy;
      c.life--;
      if (c.life <= 0) {
        coins.splice(i, 1);
        continue;
      }
    }
    if (!c.collected && Math.abs(c.x - mario.x) < 16 && Math.abs(c.y - mario.y) < 20) {
      c.collected = true;
      if (c.life !== undefined) coins.splice(i, 1);
    }
  }

  camX = mario.x - 100;
  if (camX < 0) camX = 0;
  if (camX > 4000 - 256) camX = 4000 - 256;

  if (!mario.grounded) {
    mario.state = 'jump';
  } else {
    if (mario.state !== 'run') {
      mario.state = 'run';
      mario.frame = 0;
    }
  }

  mario.animTimer++;
  if (mario.animTimer > 5) {
    mario.animTimer = 0;
    if (mario.state === 'run') {
      mario.frame = (mario.frame + 1) % 3;
    }
  }

  if (globalFrame % 10 === 0) {
    for (let c of coins) {
      c.frame = (c.frame + 1) % 4;
    }
  }

  if (mario.y > 250 || mario.x > 3900) {
    fadingOut = true;
  }
}

function draw() {
  ctx.fillStyle = '#5c94fc';
  ctx.fillRect(0, 0, 256, 240);

  for (let c of clouds) {
    let x = (c.x - camX * 0.5) % 1200;
    if (x < -50) x += 1200;
    if (x < 256) drawCloud(ctx, x, c.y);
  }
  for (let h of hills) {
    let x = (h.x - camX * 0.7) % 1500;
    if (x < -80) x += 1500;
    if (x < 256) drawHill(ctx, x, h.y);
  }
  for (let b of bushes) {
    let x = (b.x - camX) % 2000;
    if (x < -50) x += 2000;
    if (x < 256) drawBush(ctx, x, b.y);
  }

  for (let obj of level) {
    let x = obj.x - camX;
    if (x > -50 && x < 300) {
      if (obj.type === 'ground') {
        ctx.fillStyle = '#c84c0c';
        ctx.fillRect(x, obj.y, obj.w, obj.h);
        ctx.fillStyle = '#fc9838';
        ctx.fillRect(x, obj.y, obj.w, 2);
        ctx.fillRect(x, obj.y, 2, obj.h);
        ctx.fillStyle = '#000000';
        for(let i=4; i<obj.w; i+=16) {
          for(let j=4; j<obj.h; j+=16) {
            ctx.fillRect(x + i, obj.y + j, 2, 2);
            if (i+8 < obj.w && j+8 < obj.h) {
               ctx.fillRect(x + i + 8, obj.y + j + 8, 2, 2);
            }
          }
        }
      } else if (obj.type === 'brick') {
        ctx.drawImage(brickCanvas, x, obj.drawY);
      } else if (obj.type === 'qBlock') {
        if (obj.used) {
           ctx.fillStyle = '#c84c0c';
           ctx.fillRect(x, obj.drawY, 16, 16);
           ctx.fillStyle = '#fc9838';
           ctx.fillRect(x, obj.drawY, 15, 2);
           ctx.fillRect(x, obj.drawY, 2, 15);
        } else {
           ctx.drawImage(qBlockCanvas, x, obj.drawY);
        }
      } else if (obj.type === 'pipe') {
        ctx.drawImage(pipeTopCanvas, x, obj.y);
        for (let i = 1; i < obj.h / 16; i++) {
          ctx.drawImage(pipeBodyCanvas, x, obj.y + i * 16);
        }
      }
    }
  }

  for (let c of coins) {
    if (c.collected) continue;
    let x = c.x - camX;
    if (x > -20 && x < 280) {
      ctx.drawImage(coinFrames[c.frame], x, c.y);
    }
  }

  for (let e of enemies) {
    let x = e.x - camX;
    if (x > -20 && x < 280) {
      if (e.dead) {
        ctx.drawImage(goombaSquashedCanvas, x, e.y);
      } else {
        let frame = Math.floor(globalFrame / 15) % 2;
        ctx.drawImage(frame === 0 ? goombaWalk1Canvas : goombaWalk2Canvas, x, e.y);
      }
    }
  }

  let mx = mario.x - camX;
  let sprite = mario.state === 'jump' ? marioCanvases.jump : marioCanvases['run' + (mario.frame + 1)];
  ctx.drawImage(sprite, mx, mario.y);

  if (fadeAlpha > 0) {
    ctx.fillStyle = `rgba(0, 0, 0, ${fadeAlpha})`;
    ctx.fillRect(0, 0, 256, 240);
  }
}

let lastTime = 0;
let accumulator = 0;
const STEP = 1000 / 60;

function loop(time) {
  requestAnimationFrame(loop);
  if (!lastTime) lastTime = time;
  let dt = time - lastTime;
  lastTime = time;
  if (dt > 100) dt = 100;
  accumulator += dt;
  while (accumulator >= STEP) {
    update();
    accumulator -= STEP;
  }
  draw();
}

buildLevel();
requestAnimationFrame(loop);
