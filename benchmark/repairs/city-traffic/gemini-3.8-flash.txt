  function buildCity() {
    hRoads = [];
    vRoads = [];
    intersections = [];
    blocks = [];
    buildings = [];
    crosswalks = [];
    cars = [];
    pedestrians = [];

    // Horizontal roads (3 to 4 based on height)
    const numHRoads = Math.max(3, Math.min(5, Math.floor(height / 240)));
    const hSpacing = height / (numHRoads + 1);
    for (let i = 1; i <= numHRoads; i++) {
      const isAvenue = (i % 2 === 1);
      hRoads.push({
        y: Math.round(i * hSpacing),
        width: isAvenue ? ROAD_WIDTH_4L : ROAD_WIDTH_2L,
        lanes: isAvenue ? 4 : 2,
        isAvenue: isAvenue,
        id: `h_${i}`
      });
    }

    // Vertical roads (4 to 7 based on width)
    const numVRoads = Math.max(4, Math.min(8, Math.floor(width / 260)));
    const vSpacing = width / (numVRoads + 1);
    for (let j = 1; j <= numVRoads; j++) {
      const isAvenue = (j % 2 === 0);
      vRoads.push({
        x: Math.round(j * vSpacing),
        width: isAvenue ? ROAD_WIDTH_4L : ROAD_WIDTH_2L,
        lanes: isAvenue ? 4 : 2,
        isAvenue: isAvenue,
        id: `v_${j}`
      });
    }

    // Generate intersections with traffic signal controller
    let interIdx = 0;
    for (let r = 0; r < hRoads.length; r++) {
      for (let c = 0; c < vRoads.length; c++) {
        const hr = hRoads[r];
        const vr = vRoads[c];
        intersections.push({
          id: interIdx++,
          x: vr.x,
          y: hr.y,
          width: vr.width,
          height: hr.width,
          hRoad: hr,
          vRoad: vr,
          // Signal cycle: 0: NS Green, 1: NS Yellow, 2: EW Green, 3: EW Yellow
          phaseOffset: (r * 3.7 + c * 2.3) % 18,
          cycleDuration: 18,
          state: 'NS_GREEN'
        });
      }
    }

    // Generate city blocks between road corridors
    const xBounds = [0, ...vRoads.map(r => r.x), width];
    const yBounds = [0, ...hRoads.map(r => r.y), height];

    for (let i = 0; i < yBounds.length - 1; i++) {
      for (let j = 0; j < xBounds.length - 1; j++) {

        const left = xBounds[j] + (j === 0 ? 0 : (vRoads[j - 1].width / 2 + 10));
        const right = xBounds[j + 1] - (j === xBounds.length - 2 ? 0 : (vRoads[j].width / 2 + 10));

        const bw = right - left;
        const bh = bottom - top;

        if (bw > 30 && bh > 30) {

            x: left,
            y: top,
            w: bw,
            h: bh,

      }
    }

        h: 10,

      crosswalks.push({
        interId: inter.id,
        dir: 'S',
        x: inter.x - hw,

  function generateBuildingsForBlock(b) {
    const pad = 3;
    const availW = b.w - pad * 2;
    const availH = b.h - pad * 2;

    const splitH = availW > 160 && Math.random() > 0.3;
    const splitV = availH > 160 && Math.random() > 0.3;

    const parcels = [];
    if (!splitH && !splitV) {
      parcels.push({ x: b.x + pad, y: b.y + pad, w: availW, h: availH });
    } else if (splitH && !splitV) {
      const w1 = availW * (0.4 + Math.random() * 0.2);
      parcels.push({ x: b.x + pad, y: b.y + pad, w: w1 - 2, h: availH });
      parcels.push({ x: b.x + pad + w1, y: b.y + pad, w: availW - w1, h: availH });
    } else if (!splitH && splitV) {
      const h1 = availH * (0.4 + Math.random() * 0.2);
      parcels.push({ x: b.x + pad, y: b.y + pad, w: availW, h: h1 - 2 });
      parcels.push({ x: b.x + pad, y: b.y + pad + h1, w: availW, h: availH - h1 });
    } else {
      const w1 = availW * 0.5;
      const h1 = availH * 0.5;
      parcels.push({ x: b.x + pad, y: b.y + pad, w: w1 - 2, h: h1 - 2 });
      parcels.push({ x: b.x + pad + w1, y: b.y + pad, w: w1, h: h1 - 2 });
      parcels.push({ x: b.x + pad, y: b.y + pad + h1, w: w1 - 2, h: h1 });
      parcels.push({ x: b.x + pad + w1, y: b.y + pad + h1, w: w1, h: h1 });
    }

    parcels.forEach((p, idx) => {
      // Architectural properties
      const heightTier = Math.floor(Math.random() * 4); // 0 low, 3 high rise
      const depthOffset = (heightTier + 1) * 3;
      const roofColor = [
        '#181e28', '#1e2433', '#161d2b', '#20293a', '#141a24'
      ][Math.floor(Math.random() * 5)];

      // Rooftop features
      const hasHelipad = (heightTier === 3 && p.w > 50 && p.h > 50 && Math.random() > 0.6);
      const hasHVAC = Math.random() > 0.2;
      const hvacs = [];
      if (hasHVAC) {
        const count = 1 + Math.floor(Math.random() * 3);
        for (let k = 0; k < count; k++) {

            rx: 6 + Math.random() * (p.w - 20),
            ry: 6 + Math.random() * (p.h - 20),

            rh: 8 + Math.random() * 8,
            fanSpeed: 0.15 + Math.random() * 0.25,
            fanAngle: Math.random() * Math.PI

        ay: p.h * (0.3 + Math.random() * 0.4),
        blinkFreq: 1.5 + Math.random() * 1.5,
        color: Math.random() > 0.3 ? '#ff3b30' : '#00e5ff'
      } : null;

      // Window grid on roof or terrace
      const windows = [];
      const cols = Math.floor(p.w / 14);
      const rows = Math.floor(p.h / 14);
      for (let r = 1; r < rows; r++) {
        for (let c = 1; c < cols; c++) {
          if (Math.random() > 0.45) {
            windows.push({
              wx: c * (p.w / cols),
              wy: r * (p.h / rows),
              size: 2.5 + Math.random() * 2,
              baseBrightness: 0.4 + Math.random() * 0.6,
              flickerRate: 0.5 + Math.random() * 2,
              tint: Math.random() > 0.6 ? '#ffeaa7' : (Math.random() > 0.5 ? '#74b9ff' : '#ffffff')
            });
          }
        }
      }

      buildings.push({
        x: p.x,
        y: p.y,
        w: p.w,
        h: p.h,

        roofColor: roofColor,

  // Traffic signal state machine
  function updateTrafficLights(time) {
    intersections.forEach(inter => {
      const cyclePos = (time + inter.phaseOffset) % inter.cycleDuration;
      // Cycle breakdown (total 18s):
      // 0.0 - 7.5 : NS Green (EW Red)
      // 7.5 - 9.0 : NS Yellow (EW Red)
      // 9.0 - 16.5: EW Green (NS Red)
      // 16.5 - 18 : EW Yellow (NS Red)
      if (cyclePos < 7.0) {
        inter.state = 'NS_GREEN';
      } else if (cyclePos < 9.0) {
        inter.state = 'NS_YELLOW';
      } else if (cyclePos < 16.0) {
        inter.state = 'EW_GREEN';
      } else {
        inter.state = 'EW_YELLOW';
      }
    });
  }

  // Vehicles logic
  function spawnCar() {
    if (cars.length >= 80) return;

    // Pick random horizontal or vertical road
    const isHorizontal = Math.random() > 0.5;
    let road, dir, laneIdx;

    if (isHorizontal) {
      road = hRoads[Math.floor(Math.random() * hRoads.length)];
      dir = Math.random() > 0.5 ? 1 : -1; // 1: East, -1: West
      const lanesPerSide = road.lanes / 2;
      laneIdx = Math.floor(Math.random() * lanesPerSide);
      // Lane offset from centerline
      const laneOffset = (laneIdx + 0.5) * LANE_WIDTH * dir;
      const y = road.y + laneOffset;
      const x = dir === 1 ? -40 : width + 40;

      cars.push({
        axis: 'H',
        dir: dir,
        road: road,
        laneIdx: laneIdx,
        x: x,
        y: y,

        maxSpeed: (2.4 + Math.random() * 0.9) * dir,
        color: PALETTES.carColors[Math.floor(Math.random() * PALETTES.carColors.length)],
        brake: false,
        turnIntent: Math.random() > 0.7 ? (Math.random() > 0.5 ? 'LEFT' : 'RIGHT') : 'NONE',
        hasTurned: false
      });
    } else {
      road = vRoads[Math.floor(Math.random() * vRoads.length)];
      dir = Math.random() > 0.5 ? 1 : -1; // 1: South, -1: North
      const lanesPerSide = road.lanes / 2;
      laneIdx = Math.floor(Math.random() * lanesPerSide);
      const laneOffset = (laneIdx + 0.5) * LANE_WIDTH * -dir;
      const x = road.x + laneOffset;
      const y = dir === 1 ? -40 : height + 40;

      cars.push({
        axis: 'V',
        dir: dir,
        road: road,
        laneIdx: laneIdx,
        x: x,
        y: y,
        targetX: x,
        length: 22 + Math.random() * 8,
        width: 10,
        speed: (2.2 + Math.random() * 1.0) * dir,
        maxSpeed: (2.4 + Math.random() * 0.9) * dir,
        color: PALETTES.carColors[Math.floor(Math.random() * PALETTES.carColors.length)],
        brake: false,
        turnIntent: Math.random() > 0.7 ? (Math.random() > 0.5 ? 'LEFT' : 'RIGHT') : 'NONE',
        hasTurned: false
      });
    }
  }

  function updateCars() {
    for (let i = cars.length - 1; i >= 0; i--) {
      const c = cars[i];
      let shouldBrake = false;
      const sign = Math.sign(c.maxSpeed);

      // 1. Proximity to car ahead in same lane
      for (let j = 0; j < cars.length; j++) {
        if (i === j) continue;
        const other = cars[j];
        if (other.axis === c.axis && Math.sign(other.speed) === sign) {
          if (c.axis === 'H') {
            const laneDiff = Math.abs(c.y - other.y);
            const distAhead = (other.x - c.x) * sign;
            if (laneDiff < 8 && distAhead > 0 && distAhead < c.length + 24) {
              shouldBrake = true;
              break;
            }
          } else {
            const laneDiff = Math.abs(c.x - other.x);
            const distAhead = (other.y - c.y) * sign;
            if (laneDiff < 8 && distAhead > 0 && distAhead < c.length + 24) {
              shouldBrake = true;
              break;
            }
          }
        }
      }

      // 2. Traffic Light check
      if (!shouldBrake) {
        intersections.forEach(inter => {
          if (c.axis === 'H') {
            if (Math.abs(c.y - inter.y) < inter.height / 2) {
              const stopLineX = inter.x - (sign * (inter.width / 2 + 10));
              const distToStop = (stopLineX - c.x) * sign;
              const passedStop = (c.x - stopLineX) * sign > 0;
              const insideInter = Math.abs(c.x - inter.x) < inter.width / 2;

              if (!passedStop && distToStop > 0 && distToStop < 45) {
                // Must stop if state is NOT EW_GREEN
                if (inter.state === 'NS_GREEN' || inter.state === 'NS_YELLOW') {
                  shouldBrake = true;
                } else if (inter.state === 'EW_YELLOW' && distToStop > 25) {
                  shouldBrake = true;
                }
              }

              // Optional turning inside intersection
              if (!c.hasTurned && c.turnIntent !== 'NONE' && insideInter && !shouldBrake) {
                if (c.turnIntent === 'RIGHT') {
                  // Turn to vertical
                  c.axis = 'V';
                  c.dir = c.dir === 1 ? 1 : -1;
                  c.speed = c.dir * Math.abs(c.speed);
                  c.maxSpeed = c.dir * Math.abs(c.maxSpeed);
                  c.y = inter.y + (c.dir * 4);
                  c.hasTurned = true;
                }
              }
            }
          } else {
            // Vertical axis car
            if (Math.abs(c.x - inter.x) < inter.width / 2) {
              const stopLineY = inter.y - (sign * (inter.height / 2 + 10));
              const distToStop = (stopLineY - c.y) * sign;
              const passedStop = (c.y - stopLineY) * sign > 0;
              const insideInter = Math.abs(c.y - inter.y) < inter.height / 2;

              if (!passedStop && distToStop > 0 && distToStop < 45) {
                // Must stop if state is NOT NS_GREEN
                if (inter.state === 'EW_GREEN' || inter.state === 'EW_YELLOW') {
                  shouldBrake = true;
                } else if (inter.state === 'NS_YELLOW' && distToStop > 25) {
                  shouldBrake = true;
                }
              }
            }
          }
        });
      }

      // Pedestrian avoidance at crosswalks
      if (!shouldBrake) {
        for (let p = 0; p < pedestrians.length; p++) {
          const ped = pedestrians[p];
          if (c.axis === 'H') {
            const dx = (ped.x - c.x) * sign;
            const dy = Math.abs(ped.y - c.y);
            if (dx > 0 && dx < c.length + 18 && dy < 14) {
              shouldBrake = true;
              break;
            }
          } else {
            const dy = (ped.y - c.y) * sign;
            const dx = Math.abs(ped.x - c.x);
            if (dy > 0 && dy < c.length + 18 && dx < 14) {
              shouldBrake = true;
              break;
            }
          }
        }
      }

      c.brake = shouldBrake;

      // Physics acceleration / braking
      if (shouldBrake) {
        c.speed *= 0.82;
        if (Math.abs(c.speed) < 0.05) c.speed = 0;
      } else {
        if (Math.abs(c.speed) < Math.abs(c.maxSpeed)) {
          c.speed += sign * 0.08;
        }
      }

      // Move car
      if (c.axis === 'H') {
        c.x += c.speed;
      } else {
        c.y += c.speed;
      }

      // Despawn off bounds
      const margin = 80;
      if (c.x < -margin || c.x > width + margin || c.y < -margin || c.y > height + margin) {
        cars.splice(i, 1);
      }
    }
  }

  // Pedestrians logic
  function spawnPedestrian() {
    if (pedestrians.length >= 70) return;
    // Pick an intersection crosswalk to cross or walk sidewalk
    if (crosswalks.length === 0) return;
    const cw = crosswalks[Math.floor(Math.random() * crosswalks.length)];

    let px, py, targetX, targetY;
    if (cw.axis === 'H') {
      // Walks North/South across horizontal crosswalk
      px = cw.x + 4 + Math.random() * (cw.w - 8);
      py = cw.y - 6;
      targetX = px;
      targetY = cw.y + cw.h + 6;
    } else {
      // Walks East/West across vertical crosswalk
      px = cw.x - 6;
      py = cw.y + 4 + Math.random() * (cw.h - 8);
      targetX = cw.x + cw.w + 6;
      targetY = py;
    }

    pedestrians.push({
      x: px,
      y: py,
      startX: px,
      startY: py,
      targetX: targetX,
      targetY: targetY,
      crosswalk: cw,
      speed: 0.6 + Math.random() * 0.5,
      color: ['#00f0ff', '#ffeaa7', '#ff7675', '#a29bfe', '#55efc4', '#fd79a8'][Math.floor(Math.random() * 6)],
      stride: Math.random() * Math.PI * 2,
      waiting: false
    });
  }

  function updatePedestrians() {
    for (let i = pedestrians.length - 1; i >= 0; i--) {
      const p = pedestrians[i];
      const inter = intersections.find(it => it.id === p.crosswalk.interId);

      // Can pedestrian safely cross?
      // When inter is EW_GREEN, N-S crosswalks are safe because cars are stopped on V road, etc.
      let safeToCross = false;
      if (p.crosswalk.axis === 'H') {
        // Across vertical road: cars on V are stopped if state is EW_GREEN
        safeToCross = (inter && inter.state === 'EW_GREEN');
      } else {
        // Across horizontal road: cars on H are stopped if state is NS_GREEN
        safeToCross = (inter && inter.state === 'NS_GREEN');
      }

      const dx = p.targetX - p.x;
      const dy = p.targetY - p.y;
      const dist = Math.hypot(dx, dy);

      // If at start edge of crosswalk and not safe, wait
      const distFromStart = Math.hypot(p.x - p.startX, p.y - p.startY);
      if (distFromStart < 4 && !safeToCross) {
        p.waiting = true;
        p.stride += 0.02;
        continue;
      }
      p.waiting = false;

      if (dist < 2) {
        // Completed crossing: despawn and spawn new
        pedestrians.splice(i, 1);
        spawnPedestrian();
      } else {
        p.x += (dx / dist) * p.speed;
        p.y += (dy / dist) * p.speed;
        p.stride += 0.2;
      }
    }
  }

  // Update Train / Monorail
  function updateTrain() {
    if (!train) return;
    train.progress += train.speed;
    if (train.progress > 1.3) {
      train.progress = -0.3;
    }
  }

  // RENDER PASSES
  function drawGround() {
    ctx.fillStyle = PALETTES.pavement;
    ctx.fillRect(0, 0, width, height);
  }

  function drawParksAndBlocks() {
    blocks.forEach(b => {
      if (b.isPark) {
        // Park background
        ctx.fillStyle = PALETTES.parkGrass;
        ctx.fillRect(b.x, b.y, b.w, b.h);

        // Curvy strolling pathway
        ctx.strokeStyle = PALETTES.parkPath;
        ctx.lineWidth = 4;
        ctx.beginPath();
        ctx.moveTo(b.x + 10, b.y + b.h / 2);
        ctx.bezierCurveTo(
          b.x + b.w * 0.3, b.y + 10,
          b.x + b.w * 0.7, b.y + b.h - 10,
          b.x + b.w - 10, b.y + b.h / 2
        );
        ctx.stroke();

        // Park trees with multi-layer top-down canopy
        b.trees.forEach(t => {
          // Shadow
          ctx.fillStyle = 'rgba(0,0,0,0.4)';
          ctx.beginPath();
          ctx.arc(t.x + 3, t.y + 4, t.r, 0, Math.PI * 2);
          ctx.fill();

          // Canopy
          ctx.fillStyle = t.color;
          ctx.beginPath();
          ctx.arc(t.x, t.y, t.r, 0, Math.PI * 2);
          ctx.fill();

          // Highlight
          ctx.fillStyle = 'rgba(255,255,255,0.08)';
          ctx.beginPath();
          ctx.arc(t.x - 2, t.y - 2, t.r * 0.5, 0, Math.PI * 2);
          ctx.fill();
        });
      } else {
        // Urban Sidewalk platform
        ctx.fillStyle = PALETTES.sidewalk;
        ctx.fillRect(b.x, b.y, b.w, b.h);

        // Curb border
        ctx.strokeStyle = PALETTES.curbColor;
        ctx.lineWidth = 2;
        ctx.strokeRect(b.x, b.y, b.w, b.h);
      }
    });
  }

  function drawRoads() {
    // Horizontal asphalt ribbons
    hRoads.forEach(r => {
      ctx.fillStyle = '#11141a';
      ctx.fillRect(0, r.y - r.width / 2, width, r.width);

      // Lane dividers
      const halfW = r.width / 2;
      // Center dividing line (Double yellow)
      ctx.strokeStyle = '#e0a106';
      ctx.lineWidth = 1.5;
      ctx.beginPath();
      ctx.moveTo(0, r.y - 1.5);
      ctx.lineTo(width, r.y - 1.5);
      ctx.moveTo(0, r.y + 1.5);
      ctx.lineTo(width, r.y + 1.5);
      ctx.stroke();

      // Dashed lane lines
      if (r.lanes === 4) {
        ctx.strokeStyle = 'rgba(255,255,255,0.35)';
        ctx.lineWidth = 1.2;
        ctx.setLineDash([12, 16]);
        // North side lane divider
        ctx.beginPath();
        ctx.moveTo(0, r.y - halfW + LANE_WIDTH);
        ctx.lineTo(width, r.y - halfW + LANE_WIDTH);
        // South side lane divider
        ctx.moveTo(0, r.y + halfW - LANE_WIDTH);
        ctx.lineTo(width, r.y + halfW - LANE_WIDTH);
        ctx.stroke();
        ctx.setLineDash([]);
      }
    });

    // Vertical asphalt ribbons
    vRoads.forEach(r => {
      ctx.fillStyle = '#11141a';
      ctx.fillRect(r.x - r.width / 2, 0, r.width, height);

      // Center dividing line (Double yellow)
      ctx.strokeStyle = '#e0a106';
      ctx.lineWidth = 1.5;
      ctx.beginPath();
      ctx.moveTo(r.x - 1.5, 0);
      ctx.lineTo(r.x - 1.5, height);
      ctx.moveTo(r.x + 1.5, 0);
      ctx.lineTo(r.x + 1.5, height);
      ctx.stroke();

      // Dashed lane lines
      if (r.lanes === 4) {
        const halfW = r.width / 2;
        ctx.strokeStyle = 'rgba(255,255,255,0.35)';
        ctx.lineWidth = 1.2;
        ctx.setLineDash([12, 16]);
        ctx.beginPath();
        ctx.moveTo(r.x - halfW + LANE_WIDTH, 0);
        ctx.lineTo(r.x - halfW + LANE_WIDTH, height);
        ctx.moveTo(r.x + halfW - LANE_WIDTH, 0);
        ctx.lineTo(r.x + halfW - LANE_WIDTH, height);
        ctx.stroke();
        ctx.setLineDash([]);
      }
    });

    // Clear intersection interiors (smoother asphalt)
    intersections.forEach(inter => {
      ctx.fillStyle = '#11141a';
      ctx.fillRect(
        inter.x - inter.width / 2,
        inter.y - inter.height / 2,
        inter.width,
        inter.height
      );
    });

    // Zebra crosswalks & Stop lines
    crosswalks.forEach(cw => {
      ctx.fillStyle = 'rgba(255, 255, 255, 0.55)';
      if (cw.axis === 'H') {
        const stripeWidth = 5;
        const gap = 4;
        const totalStripes = Math.floor(cw.w / (stripeWidth + gap));
        for (let s = 0; s < totalStripes; s++) {
          ctx.fillRect(cw.x + s * (stripeWidth + gap), cw.y, stripeWidth, cw.h);
        }
      } else {
        const stripeHeight = 5;
        const gap = 4;
        const totalStripes = Math.floor(cw.h / (stripeHeight + gap));
        for (let s = 0; s < totalStripes; s++) {
          ctx.fillRect(cw.x, cw.y + s * (stripeHeight + gap), cw.w, stripeHeight);
        }
      }
    });

    // Traffic light signals (glowing indicator lamps on corner poles)
    intersections.forEach(inter => {
      const hw = inter.width / 2 + 5;
      const hh = inter.height / 2 + 5;

      const nsGreen = inter.state === 'NS_GREEN';
      const nsYellow = inter.state === 'NS_YELLOW';
      const ewGreen = inter.state === 'EW_GREEN';
      const ewYellow = inter.state === 'EW_YELLOW';

      const nsColor = nsGreen ? '#00ff88' : (nsYellow ? '#ffbb00' : '#ff3344');
      const ewColor = ewGreen ? '#00ff88' : (ewYellow ? '#ffbb00' : '#ff3344');

      // North & South signal beacons
      drawSignalLamp(inter.x - hw, inter.y - hh - 4, nsColor);
      drawSignalLamp(inter.x + hw, inter.y + hh + 4, nsColor);

      // East & West signal beacons
      drawSignalLamp(inter.x + hw + 4, inter.y - hh, ewColor);
      drawSignalLamp(inter.x - hw - 4, inter.y + hh, ewColor);
    });
  }

  function drawSignalLamp(x, y, color) {
    ctx.save();
    // Halo glow
    ctx.fillStyle = color;
    ctx.globalAlpha = 0.25;
    ctx.beginPath();
    ctx.arc(x, y, 7, 0, Math.PI * 2);
    ctx.fill();

    // Solid core bulb
    ctx.globalAlpha = 0.95;
    ctx.beginPath();
    ctx.arc(x, y, 2.5, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
  }

  function drawPedestrians() {
    pedestrians.forEach(p => {
      ctx.save();
      // Walking bob / limb indication
      const bob = Math.sin(p.stride) * 1.5;

      // Soft shadow
      ctx.fillStyle = 'rgba(0,0,0,0.5)';
      ctx.beginPath();
      ctx.arc(p.x + 1, p.y + 2, 2.5, 0, Math.PI * 2);
      ctx.fill();

      // Torso & head
      ctx.fillStyle = p.color;
      ctx.beginPath();
      ctx.arc(p.x, p.y, 2.2, 0, Math.PI * 2);
      ctx.fill();

      // Stride shoulder highlight
      ctx.fillStyle = '#ffffff';
      ctx.fillRect(p.x - 1, p.y - 1 + bob * 0.4, 2, 1);
      ctx.restore();
    });
  }

  function drawCars() {
    cars.forEach(c => {
      ctx.save();
      ctx.translate(c.x, c.y);

      let angle = 0;
      if (c.axis === 'H') {
        angle = c.dir === 1 ? 0 : Math.PI;
      } else {
        angle = c.dir === 1 ? Math.PI / 2 : -Math.PI / 2;
      }
      ctx.rotate(angle);

      const hw = c.length / 2;
      const hh = c.width / 2;

      // 1. Headlights beam cones (illuminating asphalt)
      ctx.save();
      const grad = ctx.createRadialGradient(hw + 4, 0, 2, hw + 45, 0, 50);
      grad.addColorStop(0, 'rgba(255, 255, 230, 0.45)');
      grad.addColorStop(0.5, 'rgba(255, 255, 200, 0.15)');
      grad.addColorStop(1, 'rgba(255, 255, 200, 0.0)');
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.moveTo(hw + 2, -hh + 1);
      ctx.lineTo(hw + 55, -hh - 18);
      ctx.lineTo(hw + 55, hh + 18);
      ctx.lineTo(hw + 2, hh - 1);
      ctx.closePath();
      ctx.fill();
      ctx.restore();

      // 2. Car shadow
      ctx.fillStyle = 'rgba(0, 0, 0, 0.45)';
      ctx.fillRect(-hw + 2, -hh + 3, c.length, c.width);

      // 3. Vehicle Chassis
      ctx.fillStyle = c.color;
      ctx.beginPath();
      ctx.roundRect(-hw, -hh, c.length, c.width, 3);
      ctx.fill();

      // 4. Roof & Windshield styling
      ctx.fillStyle = '#0a0d12';
      ctx.fillRect(-hw + 6, -hh + 1.5, c.length - 11, c.width - 3);

      // Windshield glass tint
      ctx.fillStyle = 'rgba(120, 200, 255, 0.35)';
      ctx.fillRect(hw - 9, -hh + 2, 3, c.width - 4);
      // Rear glass
      ctx.fillRect(-hw + 5, -hh + 2, 2.5, c.width - 4);

      // 5. Headlight bulbs
      ctx.fillStyle = '#ffffff';
      ctx.fillRect(hw - 1.5, -hh + 1, 1.5, 2.5);
      ctx.fillRect(hw - 1.5, hh - 3.5, 1.5, 2.5);

      // 6. Taillights & Brake lights
      if (c.brake) {
        // High intensity brake flare
        ctx.fillStyle = '#ff1100';
        ctx.shadowColor = '#ff0000';
        ctx.shadowBlur = 8;
        ctx.fillRect(-hw - 1.5, -hh + 0.5, 2.5, 3);
        ctx.fillRect(-hw - 1.5, hh - 3.5, 2.5, 3);
      } else {
        ctx.fillStyle = '#880000';
        ctx.fillRect(-hw - 1, -hh + 1, 1.5, 2.5);
        ctx.fillRect(-hw - 1, hh - 3.5, 1.5, 2.5);
      }

      ctx.restore();
    });
  }

  function drawBuildings() {
    buildings.forEach(b => {
      ctx.save();

      // Architectural isometric drop shadow
      ctx.fillStyle = 'rgba(0, 0, 0, 0.55)';
      ctx.fillRect(b.x + b.depth, b.y + b.depth, b.w, b.h);

      // Building Wall Extrusion effect
      ctx.fillStyle = '#0b0f17';
      ctx.fillRect(b.x, b.y, b.w, b.h);

      // Rooftop Slab
      ctx.fillStyle = b.roofColor;
      ctx.fillRect(b.x, b.y, b.w, b.h);

      // Rooftop border parapet
      ctx.strokeStyle = PALETTES.buildingBorder;
      ctx.lineWidth = 2;
      ctx.strokeRect(b.x + 1, b.y + 1, b.w - 2, b.h - 2);

      // Neon roofline accent
      if (b.neonEdge) {
        ctx.strokeStyle = b.neonEdge;
        ctx.lineWidth = 1.5;
        ctx.shadowColor = b.neonEdge;
        ctx.shadowBlur = 6;
        ctx.beginPath();
        ctx.moveTo(b.x + 1, b.y + 1);
        ctx.lineTo(b.x + b.w - 1, b.y + 1);
        ctx.stroke();
        ctx.shadowBlur = 0;
      }

      // Windows / Skylights
      b.windows.forEach(w => {
        const glow = Math.sin(simTime * w.flickerRate + w.wx) * 0.3 + w.baseBrightness;
        ctx.fillStyle = w.tint;
        ctx.globalAlpha = Math.max(0.1, Math.min(1.0, glow));
        ctx.fillRect(b.x + w.wx, b.y + w.wy, w.size, w.size);
      });
      ctx.globalAlpha = 1.0;

      // Helipad
      if (b.hasHelipad) {
        const cx = b.x + b.w / 2;
        const cy = b.y + b.h / 2;
        const rad = Math.min(b.w, b.h) * 0.32;

        ctx.strokeStyle = '#e2b714';
        ctx.lineWidth = 2.5;
        ctx.beginPath();
        ctx.arc(cx, cy, rad, 0, Math.PI * 2);
        ctx.stroke();

        ctx.fillStyle = '#e2b714';
        ctx.font = `bold ${Math.floor(rad * 1.1)}px monospace`;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText('H', cx, cy);
      }

      // HVAC units with rotating exhaust fans
      b.hvacs.forEach(h => {
        h.fanAngle += h.fanSpeed;
        ctx.fillStyle = '#11151e';
        ctx.fillRect(b.x + h.rx, b.y + h.ry, h.rw, h.rh);
        ctx.strokeStyle = '#384252';
        ctx.lineWidth = 1;
        ctx.strokeRect(b.x + h.rx, b.y + h.ry, h.rw, h.rh);

        // Circular fan grate
        const fcx = b.x + h.rx + h.rw / 2;
        const fcy = b.y + h.ry + h.rh / 2;
        const fr = Math.min(h.rw, h.rh) * 0.38;

        ctx.strokeStyle = '#202838';
        ctx.beginPath();
        ctx.arc(fcx, fcy, fr, 0, Math.PI * 2);
        ctx.stroke();

        // Spinning blades
        ctx.strokeStyle = '#4e5d78';
        ctx.beginPath();
        ctx.moveTo(fcx - Math.cos(h.fanAngle) * fr, fcy - Math.sin(h.fanAngle) * fr);
        ctx.lineTo(fcx + Math.cos(h.fanAngle) * fr, fcy + Math.sin(h.fanAngle) * fr);
        ctx.moveTo(fcx - Math.cos(h.fanAngle + Math.PI/2) * fr, fcy - Math.sin(h.fanAngle + Math.PI/2) * fr);
        ctx.lineTo(fcx + Math.cos(h.fanAngle + Math.PI/2) * fr, fcy + Math.sin(h.fanAngle + Math.PI/2) * fr);
        ctx.stroke();
      });

      // Aviation warning antenna beacon
      if (b.antenna) {
        const ax = b.x + b.antenna.ax;
        const ay = b.y + b.antenna.ay;
        const blink = Math.sin(simTime * b.antenna.blinkFreq) > 0.6;

        ctx.fillStyle = '#475569';
        ctx.beginPath();
        ctx.arc(ax, ay, 2, 0, Math.PI * 2);
        ctx.fill();

        if (blink) {
          ctx.fillStyle = b.antenna.color;
          ctx.shadowColor = b.antenna.color;
          ctx.shadowBlur = 8;
          ctx.beginPath();
          ctx.arc(ax, ay, 2.5, 0, Math.PI * 2);
          ctx.fill();
          ctx.shadowBlur = 0;
        }
      }

      ctx.restore();
    });
  }

  function drawMonorail() {
    if (!train) return;
    const { p1, p2 } = train;

    // Track pillars and double rails
    ctx.save();
    // Rail track lines
    ctx.strokeStyle = '#2b3445';
    ctx.lineWidth = 10;
    ctx.beginPath();
    ctx.moveTo(p1.x, p1.y);
    ctx.lineTo(p2.x, p2.y);
    ctx.stroke();

    ctx.strokeStyle = '#48566e';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(p1.x, p1.y - 3);
    ctx.lineTo(p2.x, p2.y - 3);
    ctx.moveTo(p1.x, p1.y + 3);
    ctx.lineTo(p2.x, p2.y + 3);
    ctx.stroke();

    // Moving train composition
    const tx = p1.x + (p2.x - p1.x) * train.progress;
    const ty = p1.y + (p2.y - p1.y) * train.progress;
    const angle = Math.atan2(p2.y - p1.y, p2.x - p1.x);

    ctx.translate(tx, ty);
    ctx.rotate(angle);

    // Train shadow
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(-train.length / 2 + 10, -train.width / 2 + 18, train.length, train.width);

    // Train cars (3 articulated segments)
    const segLen = train.length / 3 - 2;
    for (let s = -1; s <= 1; s++) {
      const sx = s * (segLen + 2) - segLen / 2;

      // Chassis
      ctx.fillStyle = '#e2e8f0';
      ctx.beginPath();
      ctx.roundRect(sx, -train.width / 2, segLen, train.width, 3);
      ctx.fill();

      // Blue accent stripe
      ctx.fillStyle = '#0284c7';
      ctx.fillRect(sx, -train.width / 2 + 1, segLen, 2.5);

      // Glowing interior windows
      ctx.fillStyle = '#38bdf8';
      ctx.shadowColor = '#38bdf8';
      ctx.shadowBlur = 4;
      const numWin = 4;
      for (let w = 1; w <= numWin; w++) {
        ctx.fillRect(sx + (w * segLen) / (numWin + 1) - 2, -2, 4, 4);
      }
      ctx.shadowBlur = 0;
    }

    // Front high-beam cone
    const frontX = train.length / 2;
    const beamGrad = ctx.createRadialGradient(frontX, 0, 2, frontX + 90, 0, 70);
    beamGrad.addColorStop(0, 'rgba(56, 189, 248, 0.6)');
    beamGrad.addColorStop(1, 'rgba(56, 189, 248, 0.0)');
    ctx.fillStyle = beamGrad;
    ctx.beginPath();
    ctx.moveTo(frontX, -5);
    ctx.lineTo(frontX + 110, -35);
    ctx.lineTo(frontX + 110, 35);
    ctx.lineTo(frontX, 5);
    ctx.closePath();
    ctx.fill();

    ctx.restore();
  }

  function drawVignetteAndAtmosphere() {
    // Subtle overall ambient night vignette
    const ambientGrad = ctx.createRadialGradient(
      width / 2, height / 2, Math.min(width, height) * 0.3,
      width / 2, height / 2, Math.max(width, height) * 0.8
    );
    ambientGrad.addColorStop(0, 'rgba(10, 15, 24, 0.0)');
    ambientGrad.addColorStop(1, 'rgba(5, 8, 14, 0.55)');

    ctx.fillStyle = ambientGrad;
    ctx.fillRect(0, 0, width, height);
  }

  // Master simulation step & render loop
  let lastFrameTime = performance.now();

  function animate(now) {
    const dt = Math.min((now - lastFrameTime) / 1000, 0.1);
    lastFrameTime = now;
    simTime += dt;

    // Density modulation over 50-second wave
    // Sine wave oscillates between low density (quiet night) and high density (rush hour)
    const wave = (Math.sin(simTime * 0.12) + 1) * 0.5; // 0.0 to 1.0
    densityCycle = wave;
    targetCarCount = Math.floor(25 + wave * 40);

    // Dynamic car spawning based on target density
    if (cars.length < targetCarCount && Math.random() < 0.15 + wave * 0.35) {
      spawnCar();
    }

    // Occasional pedestrian respawn
    if (Math.random() < 0.04) {
      spawnPedestrian();
    }

    // Updates
    updateTrafficLights(simTime);
    updateCars();
    updatePedestrians();
    updateTrain();

    // Render layers
    ctx.clearRect(0, 0, width, height);
    drawGround();
    drawParksAndBlocks();
    drawRoads();
    drawBuildings();
    drawPedestrians();
    drawCars();
    drawMonorail();
    drawVignetteAndAtmosphere();

    // UI Updates
    if (Math.floor(simTime * 4) % 2 === 0) {
      uiVehicles.textContent = cars.length;
      if (wave > 0.65) {
        uiPhase.textContent = 'RUSH HOUR SURGE';
        uiDensity.textContent = 'HIGH // CONGESTED';
        uiDensity.style.color = '#ff7b72';
      } else if (wave < 0.35) {
        uiPhase.textContent = 'LATE NIGHT QUIET';
        uiDensity.textContent = 'LOW // OPTIMAL';
        uiDensity.style.color = '#7ee787';
      } else {
        uiPhase.textContent = 'MIDTOWN FLOW';
        uiDensity.textContent = 'MODERATE';
        uiDensity.style.color = '#58a6ff';
      }
    }

    requestAnimationFrame(animate);
  }

  window.addEventListener('resize', resize);
  resize();
  requestAnimationFrame(animate);
})();
