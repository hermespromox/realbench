(() => {
  const canvas = document.getElementById('factory');

  let totalDispatched = 142;

  function resize() {
    dpr = window.devicePixelRatio || 1;
    width = window.innerWidth;
    height = window.innerHeight;

  // Isometric projection helper
  function iso(gx, gy, gz = 0) {
    return {
      x: originX + (gx - gy) * (TILE_W / 2),
      y: originY + (gx + gy) * (TILE_H / 2) - gz
    };
  }

  // Particle System (Sparks, Steam, Scans)
  const particles = [];
  function addSparks(x, y, z, count = 12, color = '#ffb300', speed = 2) {
    for (let i = 0; i < count; i++) {
      const angle = Math.random() * Math.PI * 2;
      const v = (0.5 + Math.random() * speed);
      particles.push({
        gx: x, gy: y, gz: z,
        vx: Math.cos(angle) * v * 0.04,
        vy: Math.sin(angle) * v * 0.04,
        vz: (0.8 + Math.random() * 2.0) * (speed * 0.6),
        gravity: 0.09,
        life: 1,
        decay: 0.02 + Math.random() * 0.03,
        color: color,
        type: 'spark'
      });
    }
  }

  function addSteam(x, y, z, count = 2) {
    for (let i = 0; i < count; i++) {
      particles.push({
        gx: x + (Math.random() - 0.5) * 0.4,
        gy: y + (Math.random() - 0.5) * 0.4,
        gz: z,
        vx: (Math.random() - 0.5) * 0.01,
        vy: (Math.random() - 0.5) * 0.01,
        vz: 0.4 + Math.random() * 0.5,
        radius: 2 + Math.random() * 3,
        growth: 0.15,
        life: 0.7,
        decay: 0.012,
        color: 'rgba(180, 220, 255,',
        type: 'steam'
      });
    }
  }

  function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
      const p = particles[i];
      p.life -= p.decay;
      if (p.life <= 0) {
        particles.splice(i, 1);
        continue;
      }
      p.gx += p.vx;
      p.gy += p.vy;
      p.gz += p.vz;
      if (p.type === 'spark') {
        p.vz -= p.gravity;
        if (p.gz < 0) {
          p.gz = 0;
          p.vz = -p.vz * 0.3;
        }
      } else if (p.type === 'steam') {
        p.radius += p.growth;
      }
    }
  }

  // Draw basic Isometric Box
  function drawIsoBox(gx, gy, gz, sx, sy, sz, cTop, cLeft, cRight) {
    const p0 = iso(gx, gy, gz + sz);
    const p1 = iso(gx + sx, gy, gz + sz);
    const p2 = iso(gx + sx, gy + sy, gz + sz);
    const p3 = iso(gx, gy + sy, gz + sz);

    const b0 = iso(gx, gy, gz);
    const b1 = iso(gx + sx, gy, gz);
    const b2 = iso(gx + sx, gy + sy, gz);
    const b3 = iso(gx, gy + sy, gz);

    // Left Face (X axis aligned, facing viewer left)
    ctx.fillStyle = cLeft;
    ctx.beginPath();
    ctx.moveTo(b0.x, b0.y);
    ctx.lineTo(b3.x, b3.y);
    ctx.lineTo(p3.x, p3.y);
    ctx.lineTo(p0.x, p0.y);
    ctx.closePath();
    ctx.fill();

    // Right Face (Y axis aligned, facing viewer right)
    ctx.fillStyle = cRight;
    ctx.beginPath();
    ctx.moveTo(b3.x, b3.y);
    ctx.lineTo(b2.x, b2.y);
    ctx.lineTo(p2.x, p2.y);
    ctx.lineTo(p3.x, p3.y);
    ctx.closePath();
    ctx.fill();

    // Top Face
    ctx.fillStyle = cTop;
    ctx.beginPath();
    ctx.moveTo(p0.x, p0.y);
    ctx.lineTo(p1.x, p1.y);
    ctx.lineTo(p2.x, p2.y);
    ctx.lineTo(p3.x, p3.y);
    ctx.closePath();
    ctx.fill();
  }

  // Conveyor Belts definition
  // Conveyor 1: Infeed from (2, 2) to (2, 8)
  // Conveyor 2: Assembly from (4, 8) to (10, 8)
  // Conveyor 3: Outfeed / Inspection from (10, 8) to (10, 15)
  // Pallet loading at (7, 15) -> Dispatch track to (0, 15)

  // Products workflow tracker
  // States:
  // 1: on Conveyor 1 (raw cube -> shaped)
  // 2: Arm 1 transfer
  // 3: on Conveyor 2 (stamped with core + energy cell)
  // 4: on Conveyor 3 (laser scanned & certified)
  // 5: Arm 2 transfer to pallet
  // 6: on Pallet waiting
  const products = [];
  let productCounter = 0;
  let spawnTimer = 0;

  // Stamping Machine state
  const stamper = {
    gx: 7, gy: 8, gz: 0,
    headOffset: 0,
    active: false,
    timer: 0
  };

  // Robot Arm 1 (transfers from (2,8) to (4,8))
  const robot1 = {
    gx: 3, gy: 7.2,
    angleBase: 0,
    armAngle1: 0,
    armAngle2: 0,
    carrying: null,
    target: null,
    phase: 0 // 0: resting, 1: picking, 2: lifting, 3: swinging, 4: dropping
  };

  // Robot Arm 2 (transfers from (10, 15) to pallet at (8, 15))
  const robot2 = {
    gx: 9, gy: 14.5,
    angleBase: 0,
    armAngle1: 0,
    armAngle2: 0,
    carrying: null,
    target: null,
    phase: 0
  };

  // Pallet Staging Area
  const pallet = {
    gx: 7.5, gy: 15, gz: 2,
    items: 0,
    maxItems: 4,
    shippingProgress: 0,
    isShipping: false
  };

  // Autonomous Guided Vehicles (AGVs)
  const agvs = [
    {
      id: 1,
      route: [
        { gx: 12, gy: 3 },
        { gx: 12, gy: 16 },
        { gx: 4, gy: 16 },
        { gx: 4, gy: 11 },
        { gx: 12, gy: 11 }
      ],
      currentIdx: 0,
      t: 0,
      gx: 12, gy: 3,
      angle: 0,
      cargo: 'raw_materials',
      color: '#00f0ff'
    },
    {
      id: 2,
      route: [
        { gx: 3, gy: 16 },
        { gx: 3, gy: 3 },
        { gx: 13, gy: 3 },
        { gx: 13, gy: 16 }
      ],
      currentIdx: 0,
      t: 0.5,
      gx: 3, gy: 9,
      angle: 0,
      cargo: 'battery_cells',
      color: '#ffaa00'
    }
  ];

  function spawnProduct() {
    productCounter++;
    products.push({
      id: productCounter,
      stage: 'conveyor1',
      gx: 2,
      gy: 2,
      gz: 12,
      progress: 0,
      processedLaser: false,
      stamped: false,
      scanned: false,
      assembled: false,
      glowColor: '#00f0ff',
      coreColor: '#3a4b60'
    });
  }

  // Pre-seed a few products along lines
  for (let i = 0; i < 4; i++) {
    const p = {
      id: ++productCounter,
      stage: 'conveyor1',
      gx: 2,
      gy: 2 + i * 1.5,
      gz: 12,
      progress: (i * 1.5) / 6,
      processedLaser: i >= 2,
      stamped: false,
      scanned: false,
      assembled: false,
      glowColor: i >= 2 ? '#00f0ff' : '#64748b',
      coreColor: i >= 2 ? '#00e5ff' : '#334155'
    };
    products.push(p);
  }

  // Add one already on conveyor 2 and conveyor 3
  products.push({
    id: ++productCounter,
    stage: 'conveyor2',
    gx: 5.5,
    gy: 8,
    gz: 12,
    progress: 0.25,
    processedLaser: true,
    stamped: false,
    scanned: false,
    assembled: true,
    glowColor: '#00f0ff',
    coreColor: '#00e5ff'
  });
  products.push({
    id: ++productCounter,
    stage: 'conveyor3',
    gx: 10,
    gy: 11,
    gz: 12,
    progress: 0.45,
    processedLaser: true,
    stamped: true,
    scanned: true,
    assembled: true,
    glowColor: '#00ff88',
    coreColor: '#00ffaa'
  });

  // Main Logic Updates
  let lastTime = performance.now();
  function update(dt) {
    // Product Spawner
    spawnTimer += dt;
    if (spawnTimer > 3.2) {
      spawnTimer = 0;
      spawnProduct();
    }

    // Conveyor 1 Updates (moves from gy=2 to gy=8)
    for (let p of products) {
      if (p.stage === 'conveyor1') {
        p.gy += dt * 0.95;
        p.gx = 2;
        p.gz = 12;

        // Laser cutter station at gy=5
        if (p.gy >= 4.8 && !p.processedLaser) {
          p.processedLaser = true;
          p.glowColor = '#00f0ff';
          p.coreColor = '#00e5ff';
          addSparks(2, 5, 14, 18, '#00f0ff', 2.5);
        }

        // Reached end of Conveyor 1 -> wait for Arm 1
        if (p.gy >= 8) {
          p.gy = 8;
          p.stage = 'waiting_arm1';
        }
      }

      // Conveyor 2 Updates (moves from gx=4 to gx=10 at gy=8)
      else if (p.stage === 'conveyor2') {
        p.gy = 8;
        p.gz = 12;

        // Check if passing under stamper at gx=7
        if (p.gx >= 6.8 && p.gx <= 7.2 && !p.stamped) {
          if (!stamper.active && stamper.timer <= 0) {
            stamper.active = true;
            stamper.targetProduct = p;
          }
        }

        if (!p.waitingForStamper) {
          p.gx += dt * 0.95;
        }

        if (p.gx >= 10) {
          p.gx = 10;
          p.stage = 'conveyor3';
        }
      }

      // Conveyor 3 Updates (moves from gy=8 to gy=15 at gx=10)
      else if (p.stage === 'conveyor3') {
        p.gx = 10;
        p.gz = 12;
        p.gy += dt * 0.95;

        // Laser Quality Inspection at gy=11.5
        if (p.gy >= 11.3 && !p.scanned) {
          p.scanned = true;
          p.glowColor = '#00ff88';
          p.coreColor = '#00ffaa';
          addSparks(10, 11.5, 14, 10, '#00ffaa', 1.8);
        }

        if (p.gy >= 15) {
          p.gy = 15;
          p.stage = 'waiting_arm2';
        }
      }
    }

    // Stamper Press Animation
    if (stamper.active) {
      stamper.timer += dt * 4.5;
      if (stamper.timer < 1.0) {
        // Slam down fast
        stamper.headOffset = Math.sin(stamper.timer * Math.PI * 0.5) * 14;
      } else if (stamper.timer < 1.2) {
        stamper.headOffset = 14;
        if (stamper.targetProduct && !stamper.targetProduct.stamped) {
          stamper.targetProduct.stamped = true;
          stamper.targetProduct.assembled = true;
          addSparks(7, 8, 14, 25, '#ffaa00', 3.2);
          addSteam(7, 8, 22, 6);
        }
      } else if (stamper.timer < 2.0) {
        // Retract up
        const t = (stamper.timer - 1.2) / 0.8;
        stamper.headOffset = 14 * (1 - t);
      } else {
        stamper.headOffset = 0;
        stamper.active = false;
        stamper.timer = 0;
        if (stamper.targetProduct) {
          stamper.targetProduct.waitingForStamper = false;
          stamper.targetProduct = null;
        }
      }
    }

    // Robot Arm 1 Controller
    // Picks from (2, 8) and places onto (4, 8)
    const arm1Candidate = products.find(p => p.stage === 'waiting_arm1');
    if (robot1.phase === 0 && arm1Candidate) {
      robot1.phase = 1; // Start pick
      robot1.target = arm1Candidate;
      robot1.target.stage = 'grabbed_arm1';
      robot1.progress = 0;
    }

    if (robot1.phase > 0) {
      robot1.progress += dt * 1.4;
      if (robot1.progress < 0.3) {
        // Hover and descend
        const t = robot1.progress / 0.3;
        robot1.gx = 2;
        robot1.gy = 8;
        robot1.gz = 12 + (1 - t) * 8;
      } else if (robot1.progress < 0.7) {
        // Lift and swing to (4, 8)
        const t = (robot1.progress - 0.3) / 0.4;
        const ease = 0.5 - 0.5 * Math.cos(t * Math.PI);
        robot1.gx = 2 + ease * 2;
        robot1.gy = 8;
        robot1.gz = 12 + Math.sin(t * Math.PI) * 10;
        if (robot1.target) {
          robot1.target.gx = robot1.gx;
          robot1.target.gy = robot1.gy;
          robot1.target.gz = robot1.gz;
        }
      } else if (robot1.progress < 1.0) {
        // Lower to conveyor 2
        robot1.gx = 4;
        robot1.gy = 8;
        const t = (robot1.progress - 0.7) / 0.3;
        robot1.gz = 12 + (1 - t) * 4;
        if (robot1.target) {
          robot1.target.gx = 4;
          robot1.target.gy = 8;
          robot1.target.gz = robot1.gz;
        }
      } else {
        // Release
        if (robot1.target) {
          robot1.target.gx = 4;
          robot1.target.gy = 8;
          robot1.target.gz = 12;
          robot1.target.stage = 'conveyor2';
          robot1.target = null;
        }
        robot1.phase = 0;
        robot1.progress = 0;
      }
    }

    // Robot Arm 2 Controller
    // Picks finished item from (10, 15) to Pallet at (7.5, 15)
    const arm2Candidate = products.find(p => p.stage === 'waiting_arm2');
    if (robot2.phase === 0 && arm2Candidate && !pallet.isShipping) {
      robot2.phase = 1;
      robot2.target = arm2Candidate;
      robot2.target.stage = 'grabbed_arm2';
      robot2.progress = 0;
    }

    if (robot2.phase > 0) {
      robot2.progress += dt * 1.5;
      if (robot2.progress < 0.3) {
        robot2.gx = 10;
        robot2.gy = 15;
      } else if (robot2.progress < 0.7) {
        const t = (robot2.progress - 0.3) / 0.4;
        const ease = 0.5 - 0.5 * Math.cos(t * Math.PI);
        robot2.gx = 10 - ease * 2.5;
        robot2.gy = 15;
        robot2.gz = 12 + Math.sin(t * Math.PI) * 8;
        if (robot2.target) {
          robot2.target.gx = robot2.gx;
          robot2.target.gy = robot2.gy;
          robot2.target.gz = robot2.gz;
        }
      } else if (robot2.progress < 1.0) {
        const t = (robot2.progress - 0.7) / 0.3;
        robot2.gx = 7.5;
        robot2.gy = 15;
        if (robot2.target) {
          robot2.target.gx = 7.5;
          robot2.target.gy = 15;
          robot2.target.gz = 6 + (1 - t) * 4;
        }
      } else {
        if (robot2.target) {
          // Put on pallet
          const pIndex = products.indexOf(robot2.target);
          if (pIndex !== -1) products.splice(pIndex, 1);
          pallet.items++;
          robot2.target = null;

          if (pallet.items >= pallet.maxItems) {
            pallet.isShipping = true;
            pallet.shippingProgress = 0;
          }
        }
        robot2.phase = 0;
        robot2.progress = 0;
      }
    }

    // Pallet Shipping / Maglev Train dispatch
    if (pallet.isShipping) {
      pallet.shippingProgress += dt * 0.7;
      if (pallet.shippingProgress > 1.2) {
        pallet.items = 0;
        pallet.isShipping = false;
        pallet.shippingProgress = 0;
        totalDispatched += pallet.maxItems;
        if (statCountEl) {
          statCountEl.textContent = String(totalDispatched).padStart(4, '0');
        }
        addSteam(4, 15, 6, 8);
      }
    }

    // AGV Pathing Update
    for (let agv of agvs) {
      agv.t += dt * 0.28;
      const nextIdx = (agv.currentIdx + 1) % agv.route.length;
      const pA = agv.route[agv.currentIdx];
      const pB = agv.route[nextIdx];

      if (agv.t >= 1) {
        agv.t = 0;
        agv.currentIdx = nextIdx;
      }
      agv.gx = pA.gx + (pB.gx - pA.gx) * agv.t;
      agv.gy = pA.gy + (pB.gy - pA.gy) * agv.t;

      // Calculate heading angle
      agv.angle = Math.atan2(pB.gy - pA.gy, pB.gx - pA.gx);
    }

    // Steam vents continuously breathing
    if (Math.random() < 0.2) {
      addSteam(14, 4, 30, 1);
      addSteam(2, 2, 22, 1);
    }

    updateParticles();
  }

  // --- RENDERING ROUTINES ---

  // Floor Grid & Caution Stripes
  function drawFloor() {
    const minX = 0, maxX = 16;
    const minY = 0, maxY = 18;

    // Base factory floor slab
    const pTop = iso(minX, minY, 0);
    const pRight = iso(maxX, minY, 0);
    const pBottom = iso(maxX, maxY, 0);
    const pLeft = iso(minX, maxY, 0);

    ctx.fillStyle = '#0b0f19';
    ctx.beginPath();
    ctx.moveTo(pTop.x, pTop.y);
    ctx.lineTo(pRight.x, pRight.y);
    ctx.lineTo(pBottom.x, pBottom.y);
    ctx.lineTo(pLeft.x, pLeft.y);
    ctx.closePath();
    ctx.fill();

    // Floor bevel / depth
    const bBottom = iso(maxX, maxY, -24);
    const bLeft = iso(minX, maxY, -24);
    const bRight = iso(maxX, minY, -24);

    ctx.fillStyle = '#05070c';
    ctx.beginPath();
    ctx.moveTo(pLeft.x, pLeft.y);
    ctx.lineTo(pBottom.x, pBottom.y);
    ctx.lineTo(bBottom.x, bBottom.y);
    ctx.lineTo(bLeft.x, bLeft.y);
    ctx.closePath();
    ctx.fill();

    ctx.fillStyle = '#080c14';
    ctx.beginPath();
    ctx.moveTo(pBottom.x, pBottom.y);
    ctx.lineTo(pRight.x, pRight.y);
    ctx.lineTo(bRight.x, bRight.y);
    ctx.lineTo(bBottom.x, bBottom.y);
    ctx.closePath();
    ctx.fill();

    // Grid lines & AGV track markings
    ctx.strokeStyle = '#151d2d';
    ctx.lineWidth = 1;
    for (let x = minX; x <= maxX; x++) {
      const p1 = iso(x, minY, 0);
      const p2 = iso(x, maxY, 0);
      ctx.beginPath();
      ctx.moveTo(p1.x, p1.y);
      ctx.lineTo(p2.x, p2.y);
      ctx.stroke();
    }
    for (let y = minY; y <= maxY; y++) {
      const p1 = iso(minX, y, 0);
      const p2 = iso(maxX, y, 0);
      ctx.beginPath();
      ctx.moveTo(p1.x, p1.y);
      ctx.lineTo(p2.x, p2.y);
      ctx.stroke();
    }

    // Glowing AGV guide lanes on floor
    drawLanes();
  }

  function drawLanes() {
    ctx.save();
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.15)';
    ctx.lineWidth = 3;
    ctx.setLineDash([8, 8]);

    // Outer AGV Loop
    const waypoints = [
      { x: 3, y: 3 },
      { x: 13, y: 3 },
      { x: 13, y: 16 },
      { x: 3, y: 16 }
    ];
    ctx.beginPath();
    const start = iso(waypoints[0].x, waypoints[0].y, 1);
    ctx.moveTo(start.x, start.y);
    for (let i = 1; i <= waypoints.length; i++) {
      const wp = waypoints[i % waypoints.length];
      const p = iso(wp.x, wp.y, 1);
      ctx.lineTo(p.x, p.y);
    }
    ctx.stroke();
    ctx.restore();
  }

  // Draw Conveyor Segment
  function drawConveyor(x1, y1, x2, y2, z = 12, widthScale = 0.7) {
    const isX = Math.abs(x2 - x1) > Math.abs(y2 - y1);
    const minGx = Math.min(x1, x2);
    const maxGx = Math.max(x1, x2);
    const minGy = Math.min(y1, y2);
    const maxGy = Math.max(y1, y2);

    const length = isX ? (maxGx - minGx + 1) : (maxGy - minGy + 1);

    // Support pillars
    for (let i = 0; i < length; i += 2) {
      const px = isX ? minGx + i : x1;
      const py = isX ? y1 : minGy + i;
      drawIsoBox(px + 0.35, py + 0.35, 0, 0.3, 0.3, z, '#1c2434', '#111722', '#161e2e');
    }

    // Conveyor Track Bed
    const sx = isX ? length : widthScale;
    const sy = isX ? widthScale : length;
    const ox = isX ? minGx : (x1 + (1 - widthScale) / 2);
    const oy = isX ? (y1 + (1 - widthScale) / 2) : minGy;

    drawIsoBox(ox, oy, z - 2, sx, sy, 2, '#232d40', '#141b27', '#1b2433');

    // Belt surface with moving slats
    const time = performance.now() * 0.003;
    const pA = iso(ox, oy, z);
    const pB = iso(ox + sx, oy, z);
    const pC = iso(ox + sx, oy + sy, z);
    const pD = iso(ox, oy + sy, z);

    ctx.fillStyle = '#0f141f';
    ctx.beginPath();
    ctx.moveTo(pA.x, pA.y);
    ctx.lineTo(pB.x, pB.y);
    ctx.lineTo(pC.x, pC.y);
    ctx.lineTo(pD.x, pD.y);
    ctx.closePath();
    ctx.fill();

    // Belt Slat Stripes
    ctx.strokeStyle = '#2f3d54';
    ctx.lineWidth = 1.5;
    const slats = 22;
    for (let i = 0; i < slats; i++) {
      const offset = ((i / slats) + time) % 1;
      let slatStart, slatEnd;
      if (isX) {
        const curX = ox + offset * sx;
        slatStart = iso(curX, oy, z + 0.2);
        slatEnd = iso(curX, oy + sy, z + 0.2);
      } else {
        const curY = oy + offset * sy;
        slatStart = iso(ox, curY, z + 0.2);
        slatEnd = iso(ox + sx, curY, z + 0.2);
      }
      ctx.beginPath();
      ctx.moveTo(slatStart.x, slatStart.y);
      ctx.lineTo(slatEnd.x, slatEnd.y);
      ctx.stroke();
    }

    // Glowing edge rails
    ctx.strokeStyle = '#00f0ff';
    ctx.lineWidth = 1.5;
    ctx.beginPath();
    if (isX) {
      const r1 = iso(ox, oy, z + 1);
      const r2 = iso(ox + sx, oy, z + 1);
      ctx.moveTo(r1.x, r1.y);
      ctx.lineTo(r2.x, r2.y);

      const r3 = iso(ox, oy + sy, z + 1);
      const r4 = iso(ox + sx, oy + sy, z + 1);
      ctx.moveTo(r3.x, r3.y);
      ctx.lineTo(r4.x, r4.y);
    } else {
      const r1 = iso(ox, oy, z + 1);
      const r2 = iso(ox, oy + sy, z + 1);
      ctx.moveTo(r1.x, r1.y);
      ctx.lineTo(r2.x, r2.y);

      const r3 = iso(ox + sx, oy, z + 1);
      const r4 = iso(ox + sx, oy + sy, z + 1);
      ctx.moveTo(r3.x, r3.y);
      ctx.lineTo(r4.x, r4.y);
    }
    ctx.stroke();
  }

  // Articulated Industrial Robotic Arm
  function drawRobotArm(arm, baseGx, baseGy) {
    const baseZ = 0;
    // Turret Base
    drawIsoBox(baseGx - 0.4, baseGy - 0.4, baseZ, 0.8, 0.8, 4, '#1e293b', '#0f172a', '#162032');
    drawIsoBox(baseGx - 0.3, baseGy - 0.3, baseZ + 4, 0.6, 0.6, 3, '#ffaa00', '#c27803', '#d98200');

    // Arm kinematics simulation
    const pBase = iso(baseGx, baseGy, baseZ + 7);

    // End effector target
    const targetZ = (arm.target && arm.phase > 0) ? arm.gz : 14;
    const effGx = (arm.target && arm.phase > 0) ? arm.gx : (baseGx + 0.8);
    const effGy = (arm.target && arm.phase > 0) ? arm.gy : baseGy;

    // Intermediate elbow joint
    const midGx = (baseGx + effGx) / 2;
    const midGy = (baseGy + effGy) / 2;
    const midGz = Math.max(baseZ + 18, targetZ + 10);

    const pMid = iso(midGx, midGy, midGz);
    const pEff = iso(effGx, effGy, targetZ + 3);

    // Hydraulic cylinders & Segments
    ctx.lineWidth = 5;
    ctx.strokeStyle = '#334155';
    ctx.lineCap = 'round';
    ctx.beginPath();
    ctx.moveTo(pBase.x, pBase.y);
    ctx.lineTo(pMid.x, pMid.y);
    ctx.stroke();

    ctx.lineWidth = 3.5;
    ctx.strokeStyle = '#ff9d00';
    ctx.beginPath();
    ctx.moveTo(pBase.x, pBase.y);
    ctx.lineTo(pMid.x, pMid.y);
    ctx.stroke();

    // Forearm
    ctx.lineWidth = 4;
    ctx.strokeStyle = '#1e293b';
    ctx.beginPath();
    ctx.moveTo(pMid.x, pMid.y);
    ctx.lineTo(pEff.x, pEff.y);
    ctx.stroke();

    ctx.lineWidth = 2.5;
    ctx.strokeStyle = '#00f0ff';
    ctx.beginPath();
    ctx.moveTo(pMid.x, pMid.y);
    ctx.lineTo(pEff.x, pEff.y);
    ctx.stroke();

    // Joints / Servo motors
    ctx.fillStyle = '#64748b';
    ctx.beginPath();
    ctx.arc(pBase.x, pBase.y, 4, 0, Math.PI * 2);
    ctx.arc(pMid.x, pMid.y, 4, 0, Math.PI * 2);
    ctx.fill();

    // Gripper Tool
    ctx.fillStyle = '#ffaa00';
    ctx.beginPath();
    ctx.arc(pEff.x, pEff.y, 3, 0, Math.PI * 2);
    ctx.fill();

    // Tiny suction/welding indicator
    if (arm.phase > 0) {
      ctx.fillStyle = '#00ffaa';
      ctx.shadowColor = '#00ffaa';
      ctx.shadowBlur = 6;
      ctx.beginPath();
      ctx.arc(pEff.x, pEff.y + 4, 1.8, 0, Math.PI * 2);
      ctx.fill();
      ctx.shadowBlur = 0;
    }
  }

    ctx.beginPath();

    drawIsoBox(agv.gx - sx/2, agv.gy - sy/2, z, sx, sy, sz, '#1e293b', '#0f172a', '#172033');

    // Blinking safety light
    const flash = Math.floor(performance.now() / 250) % 2 === 0;

    ctx.fillStyle = flash ? '#f59e0b' : '#78350f';
    ctx.beginPath();
    ctx.arc(lPos.x, lPos.y, 1.8, 0, Math.PI * 2);

  }

  // Pallet Staging & Outbound Maglev Track
  function drawShippingDock() {
    // Dock platform
    drawIsoBox(4, 14.2, 0, 5, 2.6, 2, '#161e2e', '#0b1019', '#111724');

    // Outbound Maglev Rails leading off left edge
    const rStart = iso(0, 15, 2.5);
    const rEnd = iso(4, 15, 2.5);
    ctx.strokeStyle = '#00f0ff';
    ctx.lineWidth = 2;
    ctx.shadowColor = '#00f0ff';
    ctx.shadowBlur = 6;
    ctx.beginPath();
    ctx.moveTo(rStart.x, rStart.y);
    ctx.lineTo(rEnd.x, rEnd.y);
    ctx.stroke();
    ctx.shadowBlur = 0;

    // Pallet Base
    let palletGx = pallet.gx;
    let palletGy = pallet.gy;
    let palletGz = pallet.gz;

    if (pallet.isShipping) {
      // Maglev transport speeding off to origin (0, 15)
      palletGx = 7.5 - pallet.shippingProgress * 8.5;

    drawIsoBox(palletGx - 0.7, palletGy - 0.7, palletGz, 1.4, 1.4, 0.6, '#334155', '#1e293b', '#253346');

    const count = pallet.isShipping ? pallet.maxItems : pallet.items;
    for (let i = 0; i < count; i++) {

      const oy = (i < 2 ? -0.3 : 0.3);
      drawManufacturedProduct({
        gx: palletGx + ox,

        gz: palletGz + 0.6,
        scanned: true,
        assembled: true,
        glowColor: '#00ff88',
        coreColor: '#00ffaa'
      });
    }

    // Departure Sensor Gate
    drawIsoBox(2.8, 14.2, 2, 0.4, 0.4, 18, '#1e293b', '#0f172a', '#141d2c');
    drawIsoBox(2.8, 15.6, 2, 0.4, 0.4, 18, '#1e293b', '#0f172a', '#141d2c');
    drawIsoBox(2.8, 14.2, 20, 0.4, 1.8, 2, '#00f0ff', '#0099aa', '#00b3c6');
  }

  // Draw Individual Product
  function drawManufacturedProduct(p) {
    const size = 0.55;
    const gz = p.gz || 12;

    if (!p.assembled) {
      // Raw unprocessed or newly sized block
      const topCol = p.glowColor || '#64748b';
      drawIsoBox(p.gx - size/2, p.gy - size/2, gz, size, size, 1.2, topCol, '#334155', '#475569');
      if (p.processedLaser) {
        // Laser core indicator
        const center = iso(p.gx, p.gy, gz + 1.3);
        ctx.fillStyle = '#00f0ff';
        ctx.beginPath();
        ctx.arc(center.x, center.y, 2, 0, Math.PI * 2);
        ctx.fill();
      }
    } else {
      // Completed Sci-Fi Quantum Battery / Core: Sleek Metallic Shell + Glowing Ring
      drawIsoBox(p.gx - size/2, p.gy - size/2, gz, size, size, 1.5, '#e2e8f0', '#475569', '#64748b');

      // Glowing power strip ring
      const ringPos = iso(p.gx, p.gy, gz + 0.8);
      ctx.fillStyle = p.glowColor;
      ctx.shadowColor = p.glowColor;
      ctx.shadowBlur = 6;
      ctx.beginPath();
      ctx.ellipse(ringPos.x, ringPos.y, 5, 2.5, 0, 0, Math.PI * 2);
      ctx.fill();
      ctx.shadowBlur = 0;

      // Top energy emitter
      const topCap = iso(p.gx, p.gy, gz + 1.55);
      ctx.fillStyle = p.coreColor;
      ctx.beginPath();
      ctx.arc(topCap.x, topCap.y, 1.5, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  // Draw Particles (Sparks & Steam)
  function renderParticles() {
    for (let p of particles) {
      const screenPos = iso(p.gx, p.gy, p.gz);
      if (p.type === 'spark') {
        ctx.fillStyle = p.color;
        ctx.shadowColor = p.color;
        ctx.shadowBlur = 5;
        ctx.beginPath();
        ctx.arc(screenPos.x, screenPos.y, 1.5 * p.life, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;
      } else if (p.type === 'steam') {
        ctx.fillStyle = `${p.color} ${p.life * 0.35})`;
        ctx.beginPath();
        ctx.arc(screenPos.x, screenPos.y, p.radius, 0, Math.PI * 2);
        ctx.fill();
      }
    }
  }

  // Master Render Pass
  function render() {
    ctx.clearRect(0, 0, width, height);

    // Deep ambient background gradient
    const bgGrad = ctx.createRadialGradient(width/2, height*0.4, 50, width/2, height*0.5, Math.max(width, height)*0.8);
    bgGrad.addColorStop(0, '#0e1626');
    bgGrad.addColorStop(1, '#05070c');
    ctx.fillStyle = bgGrad;
    ctx.fillRect(0, 0, width, height);

    // 1. Isometric Base Floor
    drawFloor();

    // 2. Heavy Stationary Infrastructure
    drawStorageRacks();
    drawShippingDock();

    // 3. Conveyors
    // Conveyor 1: Infeed
    drawConveyor(2, 2, 2, 8, 12);
    // Conveyor 2: Assembly Line
    drawConveyor(4, 8, 10, 8, 12);
    // Conveyor 3: Outfeed / Testing
    drawConveyor(10, 8, 10, 15, 12);

    // 4. Processing Stations
    drawLaserStation();
    drawStamper();
    drawScanner();

    // 5. AGVs
    for (let agv of agvs) {
      drawAGV(agv);
    }

    // 6. Products on belts & in air
    // Depth-sort products for clean visual overlap
    const renderableProducts = [...products];
    renderableProducts.sort((a, b) => (a.gx + a.gy) - (b.gx + b.gy));
    for (let p of renderableProducts) {
      drawManufacturedProduct(p);
    }

    // 7. Articulated Robotic Arms
    drawRobotArm(robot1, 3, 7.2);
    drawRobotArm(robot2, 9, 14.5);

    // 8. Dynamic SFX Particles (Welding sparks, steam exhaust, laser sheets)
    renderParticles();
  }

  // Animation Loop with precise delta timing
  function frame(timestamp) {
    const dt = Math.min((timestamp - lastTime) / 1000, 0.1);
    lastTime = timestamp;

    update(dt);
    render();

    requestAnimationFrame(frame);
  }

  requestAnimationFrame((ts) => {
    lastTime = ts;
    requestAnimationFrame(frame);
  });

})();
