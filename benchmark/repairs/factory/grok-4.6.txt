  let W = 0, H = 0, dpr = 1, TW = 54, TH = 27, ZH = 25;

  function resize() {
    dpr = Math.min(window.devicePixelRatio || 1, 2);
    W = window.innerWidth;
    H = window.innerHeight;

    canvas.height = Math.floor(H * dpr);

  function toS(x, y, z) {
    return {
      sx: W * 0.5 + ((x - y) - (CX - CY)) * TW * 0.5,
      sy: H * 0.30 + ((x + y) - (CX + CY)) * TH * 0.5 - z * ZH
    };
  }

  function clamp(v, a, b) { return v < a ? a : v > b ? b : v; }
  function lerp(a, b, t) { return a + (b - a) * t; }
  function smooth(t) { t = clamp(t, 0, 1); return t * t * (3 - 2 * t); }
  function hash(i, j, k) {
    const s = Math.sin(i * 12.9898 + j * 78.233 + (k || 0) * 45.164) * 43758.5453;
    return s - Math.floor(s);
  }

  const RQ = [];
  function add(x, y, z, fn) {
    RQ.push({ k: (x + y) * 20 + x * 0.02 - z * 2.4, fn: fn });
  }
  function flush() {
    RQ.sort(function (a, b) { return a.k - b.k; });
    for (let i = 0; i < RQ.length; i++) RQ[i].fn();
    RQ.length = 0;
  }

  function poly(pts, fill, strk, lw) {
    ctx.beginPath();
    ctx.moveTo(pts[0].sx, pts[0].sy);
    for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i].sx, pts[i].sy);
    ctx.closePath();
    if (fill) { ctx.fillStyle = fill; ctx.fill(); }
    if (strk) {
      ctx.strokeStyle = strk;
      ctx.lineWidth = lw || 0.7;
      ctx.lineJoin = "round";
      ctx.stroke();
    }
  }

  function box(x, y, z, w, d, h, top, right, left, strk) {
    const p000 = toS(x, y, z), p100 = toS(x + w, y, z), p110 = toS(x + w, y + d, z), p010 = toS(x, y + d, z);
    const p001 = toS(x, y, z + h), p101 = toS(x + w, y, z + h), p111 = toS(x + w, y + d, z + h), p011 = toS(x, y + d, z + h);
    poly([p100, p101, p111, p110], right, strk || "rgba(0,0,0,0.28)", 0.6);
    poly([p010, p011, p111, p110], left, strk || "rgba(0,0,0,0.28)", 0.6);
    poly([p001, p101, p111, p011], top, strk || "rgba(8,16,28,0.55)", 0.7);
  }

    });
  }

      const p11 = toS(x + w, y + d, 0.02);
      const p01 = toS(x, y + d, 0.02);
      poly([p00, p10, p11, p01], "rgba(0,0,0," + a + ")");
    });

  const LIFE = 21.5;
  const SPAWN = 1.52;

  function itemPose(age) {
    if (age < 0 || age > LIFE) return null;
    let i = 0;
    while (i + 1 < KF.length && KF[i + 1].t <= age) i++;
    if (i >= KF.length - 1) {
      const p = KF[KF.length - 1];
      return { x: p.x, y: p.y, z: p.z, age: age, i: i, a: 1 - (age - 20.5) / 1.0 };
    }
    const a = KF[i], b = KF[i + 1];
    const u = (age - a.t) / Math.max(0.0001, b.t - a.t);
    const zBig = Math.abs(b.z - a.z) > 0.25;
    const uu = zBig ? smooth(u) : u;
    const us = smooth(u);
    return {
      x: lerp(a.x, b.x, uu),
      y: lerp(a.y, b.y, uu),
      z: lerp(a.z, b.z, us),
      age: age,
      i: i,
      a: age > 20.6 ? 1 - (age - 20.6) / 0.9 : 1
    };
  }

  function eachItem(t, fn) {
    const n = Math.ceil(LIFE / SPAWN) + 3;
    const k0 = Math.floor(t / SPAWN);
    for (let k = k0 - n; k <= k0; k++) {
      if (k < 0) continue;
      const pose = itemPose(t - k * SPAWN);
      if (pose) fn(pose, k);
    }
  }

  function compPose(age) {
    if (age < 0 || age > CLIFE) return null;
    let i = 0;
    while (i + 1 < COMP.length && COMP[i + 1].t <= age) i++;
    if (i >= COMP.length - 1) return { x: COMP[i].x, y: COMP[i].y, z: COMP[i].z, age: age, a: 0.4 };
    const a = COMP[i], b = COMP[i + 1];
    const u = smooth((age - a.t) / Math.max(0.0001, b.t - a.t));
    return { x: lerp(a.x, b.x, u), y: lerp(a.y, b.y, u), z: lerp(a.z, b.z, u), age: age, a: 1 };
  }

  const particles = [];
  function emit(x, y, z, vx, vy, vz, life, col, sz, g) {
    if (particles.length > 280) particles.splice(0, 40);
    particles.push({ x: x, y: y, z: z, vx: vx, vy: vy, vz: vz, life: life, max: life, col: col, sz: sz, g: g });
  }

  const AGV = [
    { x: 5.5, y: 1.25 }, { x: 14.6, y: 1.25 }, { x: 14.6, y: 11.35 }, { x: 1.15, y: 11.35 }, { x: 1.15, y: 1.25 }
  ];
  function pathLen(path) {
    let L = 0;
    for (let i = 0; i < path.length; i++) {
      const a = path[i], b = path[(i + 1) % path.length];
      L += Math.hypot(b.x - a.x, b.y - a.y);
    }
    return L;
  }
  const AGVL = pathLen(AGV);
  function along(path, dist, total) {
    let d = ((dist % total) + total) % total;
    for (let i = 0; i < path.length; i++) {
      const a = path[i], b = path[(i + 1) % path.length];
      const seg = Math.hypot(b.x - a.x, b.y - a.y);
      if (d <= seg) {
        const u = d / Math.max(0.0001, seg);
        return { x: lerp(a.x, b.x, u), y: lerp(a.y, b.y, u), dx: b.x - a.x, dy: b.y - a.y };
      }
      d -= seg;
    }
    return { x: path[0].x, y: path[0].y, dx: 1, dy: 0 };
  }

  const PAL = [

  function stampHeight(t) {
    let best = 1;
    eachItem(t, function (p) {
      if (p.age > 4.30 && p.age < 5.95) {
        const loc = p.age - 4.30;
        if (loc < 0.14) best = Math.min(best, 1 - loc / 0.14);
        else if (loc < 1.48) best = Math.min(best, 0);
        else best = Math.min(best, (loc - 1.48) / 0.17);
      }
    });
    return clamp(best, 0, 1);
  }

  function assemblePulse(t) {
    let v = 0;
    eachItem(t, function (p) {
      if (p.age > 11.10 && p.age < 12.60) {
        const loc = p.age - 11.10;
        v = Math.max(v, 0.4 + 0.6 * Math.abs(Math.sin(loc * 18)));
      }
    });
    return v;
  }

  let lastStamp = -1;
  let lastWeld = -1;
  let produced = 12840;

  function update(t, dt) {
    eachItem(t, function (p, k) {
      if (p.age > 4.42 && p.age < 4.50 && k !== lastStamp) {
        lastStamp = k;
        for (let i = 0; i < 18; i++) {
          emit(p.x + (hash(i, k, 1) - 0.5) * 0.5, p.y + (hash(i, k, 2) - 0.5) * 0.5, p.z + 0.4,
            (hash(i, k, 3) - 0.5) * 1.6, (hash(i, k, 4) - 0.5) * 1.6, 1.4 + hash(i, k, 5) * 2.2,
            0.35 + hash(i, k, 6) * 0.35, hash(i, 7) > 0.5 ? "#ffd28a" : "#ff8a3c", 1.6 + hash(i, 8) * 2, -6);
        }
      }
      if (p.age > 11.2 && p.age < 12.5 && Math.random() < 0.35) {
        emit(p.x + (Math.random() - 0.5) * 0.4, p.y + (Math.random() - 0.5) * 0.4, p.z + 0.5,
          (Math.random() - 0.5) * 0.8, (Math.random() - 0.5) * 0.8, 0.6 + Math.random(),
          0.28, Math.random() > 0.5 ? "#7dd3fc" : "#a5f3fc", 1.4, -1);
      }
      if (p.age > 21.35 && p.age < 21.35 + dt + 0.02) produced++;
    });
    if (Math.random() < 0.6) {
      emit(2.15, 1.85, 3.6 + Math.random() * 0.2, (Math.random() - 0.5) * 0.15, (Math.random() - 0.5) * 0.15, 0.55 + Math.random() * 0.4, 1.8, "rgba(180,200,220,0.5)", 4 + Math.random() * 5, 0.2);
      emit(3.85, 1.85, 4.1, (Math.random() - 0.5) * 0.12, (Math.random() - 0.5) * 0.12, 0.5, 2.0, "rgba(170,190,210,0.45)", 5, 0.15);
    }
    for (let i = particles.length - 1; i >= 0; i--) {
      const p = particles[i];
      p.life -= dt;
      p.x += p.vx * dt;
      p.y += p.vy * dt;
      p.z += p.vz * dt;
      p.vz += p.g * dt;
      if (p.life <= 0) particles.splice(i, 1);
    }
  }

  function drawFloor() {
    for (let i = 0; i < 16; i++) {
      for (let j = 0; j < 12; j++) {
        const odd = (i + j) & 1;
        let top = odd ? C.floorA : C.floorB;
        if ((i === 8 && j === 3) || (i === 10 && j === 4) || (i === 10 && j === 9) || (i === 2 && j === 9)) top = "#1a2210";

          box(i + 0.18, j + 0.18, 0.06, 0.64, 0.64, 0.03, C.floorC, "#10182a", "#0c1320");

      }
    }
    const hz = [
      [7.4, 3.3, 1.6, 0.55], [8.4, 3.3, 1.5, 0.55], [9.9, 8.5, 1.7, 0.55], [2.3, 8.5, 1.6, 0.55]
    ];
    for (let h = 0; h < hz.length; h++) {
      const z = hz[h];
      for (let n = 0; n < 6; n++) {
        const col = n % 2 ? "#d4b23a" : "#1a1a1a";
        box(z[0] + n * 0.22, z[1], 0.06, 0.2, z[3], 0.025, col, col, col);
      }
    }
  }

  function drawConv(x, y, len, thick, axis, sign, t, z0) {
    z0 = z0 || 0.08;
    const pitch = 0.32;
    const n = Math.ceil(len / 1.0);
    for (let s = 0; s < n; s++) {
      const a = s;
      const seg = Math.min(1, len - s);
      let bx = x, by = y, bw = thick, bd = thick;
      if (axis === "x") { bx = x + a * sign; if (sign < 0) bx = x - a - seg; bw = seg; bd = thick; }
      else { by = y + a * sign; if (sign < 0) by = y - a - seg; bw = thick; bd = seg; }
      const cx = bx + bw * 0.5, cy = by + bd * 0.5;
      add(cx, cy, z0 + 0.2, (function (bx, by, bw, bd) {
        return function () {
          box(bx, by, z0, bw, bd, 0.34, C.belt, "#101522", "#0c1018");
          const off = ((t * 1.35 * sign) % pitch + pitch) % pitch;
          if (axis === "x") {
            const count = Math.ceil(bw / pitch) + 2;
            for (let k = -1; k < count; k++) {
              let sx = bx + k * pitch + (sign > 0 ? off : pitch - off);
              const sw = 0.14;
              if (sx < bx || sx + sw > bx + bw) continue;
              box(sx, by + 0.08, z0 + 0.34, sw, bd - 0.16, 0.04, k % 2 ? C.slat : "#3d4a63", "#222a3c", "#1a2230");
            }
          } else {
            const count = Math.ceil(bd / pitch) + 2;
            for (let k = -1; k < count; k++) {
              let sy = by + k * pitch + (sign > 0 ? off : pitch - off);
              const sd = 0.14;
              if (sy < by || sy + sd > by + bd) continue;
              box(bx + 0.08, sy, z0 + 0.34, bw - 0.16, sd, 0.04, k % 2 ? C.slat : "#3d4a63", "#222a3c", "#1a2230");
            }
          }
        };
      })(bx, by, bw, bd));
    }
    if (axis === "x") {
      qBox(x - (sign < 0 ? len : 0), y - 0.06, z0, len, 0.08, 0.42, C.railT, C.rail, C.darkL);
      qBox(x - (sign < 0 ? len : 0), y + thick - 0.02, z0, len, 0.08, 0.42, C.railT, C.rail, C.darkL);
    } else {
      qBox(x - 0.06, y - (sign < 0 ? len : 0), z0, 0.08, len, 0.42, C.railT, C.rail, C.darkL);
      qBox(x + thick - 0.02, y - (sign < 0 ? len : 0), z0, 0.08, len, 0.42, C.railT, C.rail, C.darkL);
    }
  }

  function drawProduct(p, k, finished, small) {
    const pal = finished ? DONE : PAL[Math.abs(k) % PAL.length];
    const s = small ? 0.28 : 0.42;
    const h = small ? 0.22 : 0.34;
    add(p.x, p.y, p.z + h * 0.5, function () {
      ctx.globalAlpha = clamp(p.a, 0, 1);
      qShadow.skip;
      box(p.x - s * 0.5, p.y - s * 0.5, p.z, s, s, h, pal[0], pal[1], pal[2], "rgba(0,0,0,0.4)");

      }
      if (finished) {
        ctx.save();
        const c = toS(p.x, p.y, p.z + h + 0.08);
        ctx.globalAlpha = 0.45 * clamp(p.a, 0, 1);
        ctx.fillStyle = "#5eead4";
        ctx.beginPath();
        ctx.arc(c.sx, c.sy, 5, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
      }
      ctx.globalAlpha = 1;
    });
  }

  function drawArm(bx, by, target, grip, col) {

    add(bx, by, 2.2, function () {
      const sh = toS(bx, by, baseH + 0.55);

      ctx.strokeStyle = "rgba(0,0,0,0.35)";

      ctx.lineWidth = 5.5;
      ctx.beginPath(); ctx.moveTo(mid.sx, mid.sy); ctx.lineTo(tg.sx, tg.sy); ctx.stroke();
      ctx.fillStyle = col;
      ctx.beginPath(); ctx.arc(sh.sx, sh.sy, 5.5, 0, Math.PI * 2); ctx.fill();
      ctx.beginPath(); ctx.arc(mid.sx, mid.sy, 4.5, 0, Math.PI * 2); ctx.fill();
      ctx.strokeStyle = grip > 0.5 ? "#5eead4" : "#ffb040";
      ctx.lineWidth = 2.2;
      const g = 6 + (1 - grip) * 5;
      ctx.beginPath();

  function armTarget(t, which) {
    let found = null;
    eachItem(t, function (p) {
      if (which === 1 && p.age > 3.18 && p.age < 6.95) found = p;
      if (which === 2 && p.age > 10.12 && p.age < 13.55) found = p;
    });
    if (found) return { x: found.x, y: found.y, z: found.z + 0.34, grip: 1 };
    if (which === 1) return { x: 7.45 + Math.sin(t * 0.9) * 0.28, y: 3.55, z: 2.05 + Math.sin(t * 1.4) * 0.12, grip: 0 };
    return { x: 9.55 + Math.sin(t * 0.7 + 1.2) * 0.22, y: 8.35, z: 2.05 + Math.sin(t * 1.1) * 0.1, grip: 0 };
  }

    bg.addColorStop(0, "#0b1220");
    bg.addColorStop(0.45, "#070b14");
    bg.addColorStop(1, "#05070d");

    drawConv(1.84, 3.20, 6.10, 0.78, "y", -1, t, 0.08);

    qBox(9.95, 3.55, 0.08, 0.9, 0.92, 0.28, C.darkT, C.darkR, C.darkL);

    qBox(8.55, 8.84, 0.08, 0.9, 0.78, 0.28, C.darkT, C.darkR, C.darkL);

    drawSilos(t);
    drawPress(t);
    drawAssembler(t);
    drawPacker(t);
    drawBooth(t);
    drawGantry(t);
    drawProps(t);
    drawStacks(t);

    drawArm(7.35, 2.85, armTarget(t, 1), armTarget(t, 1).grip, "#5eead4");
    drawArm(9.15, 8.05, armTarget(t, 2), armTarget(t, 2).grip, "#7dd3fc");

    eachItem(t, function (p, k) {
      drawProduct(p, k, p.age > 12.6, false);
    });

    const cn = Math.ceil(CLIFE / CSPAWN) + 2;
    const c0 = Math.floor(t / CSPAWN);
    for (let k = c0 - cn; k <= c0; k++) {
      if (k < 0) continue;
      const p = compPose(t - k * CSPAWN);
      if (p) drawProduct(p, k + 17, false, true);
    }

    drawAGV(t, 0, "#38bdf8");
    drawAGV(t, AGVL * 0.38, "#2dd4bf");
    drawAGV(t, AGVL * 0.72, "#818cf8");

    drawParticles();
    flush();
    drawLights(t);
    hud(t);
  }

  let last = performance.now();

    const t = now / 1000;
    const dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    update(t, dt);
    render(t);
    requestAnimationFrame(loop);
  }
  requestAnimationFrame(loop);
